###############################################################################
# Power Without War: situation events (namespace pww_sit)
#
# Fired only to human seats by common/scripted_effects/pww_sit_wolfram.txt,
# which sets the recipient variables (pww_sit_ev_id, pww_sit_ev_seat,
# pww_sit_ev_role, pww_sit_ev_outcome, pww_sit_ev_can_*) and the event
# targets pww_claimant, pww_holder, pww_patron, pww_lessee.
# Seat moves offered here are the fallback used while the diplomacy lane is
# not loaded (global flag pww_diplo_loaded); with it, seats act through the
# diplomacy panel and these popups only brief.
###############################################################################

add_namespace = pww_sit

# The Wolfram Question: opening brief per seat
country_event = {
	id = pww_sit.1
	title = pww_sit_wolfram_title
	desc = {
		text = pww_sit_ev_open_claimant
		trigger = { check_variable = { pww_sit_ev_seat = 0 } }
	}
	desc = {
		text = pww_sit_ev_open_holder
		trigger = { check_variable = { pww_sit_ev_seat = 1 } }
	}
	desc = {
		text = pww_sit_ev_open_patron
		trigger = { check_variable = { pww_sit_ev_seat = 2 } }
	}
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	# Claimant
	option = {
		name = pww_sit_opt_c_offer
		trigger = {
			check_variable = { pww_sit_ev_seat = 0 }
			NOT = { has_global_flag = pww_diplo_loaded }
		}
		set_temp_variable = { pww_in_bid_band = 2 }
		pww_sit_ev_bid_mine = yes
	}
	option = {
		name = pww_sit_opt_c_press
		trigger = {
			check_variable = { pww_sit_ev_seat = 0 }
			NOT = { has_global_flag = pww_diplo_loaded }
		}
		add_political_power = -25
		set_country_flag = pww_sit_human_press
		set_temp_variable = { pww_in_bid_band = 3 }
		pww_sit_ev_bid_mine = yes
	}
	option = {
		name = pww_sit_opt_c_aside
		trigger = {
			check_variable = { pww_sit_ev_seat = 0 }
			NOT = { has_global_flag = pww_diplo_loaded }
		}
	}
	# Holder
	option = {
		name = pww_sit_opt_h_hear
		trigger = {
			check_variable = { pww_sit_ev_seat = 1 }
			NOT = { has_global_flag = pww_diplo_loaded }
		}
	}
	# Patron
	option = {
		name = pww_sit_opt_p_pledge
		trigger = {
			check_variable = { pww_sit_ev_seat = 2 }
			NOT = { has_global_flag = pww_diplo_loaded }
		}
		pww_sit_ev_pledge_mine = yes
	}
	option = {
		name = pww_sit_opt_p_bid
		trigger = {
			check_variable = { pww_sit_ev_seat = 2 }
			NOT = { has_global_flag = pww_diplo_loaded }
		}
		set_temp_variable = { pww_in_bid_band = 2 }
		pww_sit_ev_bid_mine = yes
	}
	option = {
		name = pww_sit_opt_p_out
		trigger = {
			check_variable = { pww_sit_ev_seat = 2 }
			NOT = { has_global_flag = pww_diplo_loaded }
		}
	}
	# Diplomacy panel available
	option = {
		name = pww_sit_opt_ack
		trigger = { has_global_flag = pww_diplo_loaded }
	}
}

# Human holder decides a lease renewal.
country_event = {
	id = pww_sit.11
	title = pww_renew_wolfram_title
	desc = pww_renew_wolfram_desc_holder
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes
	timeout_days = 14

	option = {
		name = pww_renew_btn_reclaim
		ai_chance = { factor = 1 }
		set_temp_variable = { pww_in_outcome = 8 }
		pww_sit_ev_resolve_mine = yes
	}
	option = {
		name = pww_renew_btn_extend
		ai_chance = { factor = 0 }
		set_temp_variable = { pww_in_outcome = 6 }
		pww_sit_ev_resolve_mine = yes
	}
	option = {
		name = pww_renew_btn_rebid
		ai_chance = { factor = 0 }
		trigger = {
			OR = {
				AND = {
					check_variable = { global.pww_sit_lessee^pww_sit_ev_id = 0 }
					check_variable = { pww_sit_ev_can_p = 1 }
				}
				AND = {
					check_variable = { global.pww_sit_lessee^pww_sit_ev_id = 2 }
					check_variable = { pww_sit_ev_can_c = 1 }
				}
			}
		}
		set_temp_variable = { pww_in_outcome = 7 }
		pww_sit_ev_resolve_mine = yes
	}
}
