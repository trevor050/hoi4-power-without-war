###############################################################################
# Power Without War: scripted diplomatic actions (diplomacy lane)
#
# ROOT is the sender, THIS the target (vanilla template). Every send GUI
# stores its draft on ROOT; the AI acceptance reads only precomputed ROOT
# variables (cheap and deterministic), and ai_desire reads only intent flags
# set on the staggered tick (pww_diplo_ticks.txt), so the AI sends nothing
# without a concrete reason.
#
# Actions
#   pww_diplo_propose_agreement   negotiated supply agreement     10 PP
#   pww_diplo_lease_rights        negotiated named-state lease    25 PP
#   pww_diplo_poach_supplier      a better delivered deal than    15 PP
#                                 one of their mod supply deals
#   pww_diplo_demand_quiet        demand, back channel            25 PP
#   pww_diplo_demand_public       demand, public ultimatum        50 PP
#   pww_diplo_pledge_support      patron pledge (reserved deal)   25 PP
#   pww_diplo_quarantine          declare quarantine (coercion)   75 PP, 25 CP
#   pww_diplo_demand_settlement   after a nationalization         10 PP
#   pww_diplo_join_embargo        ask a third party to embargo    15 PP
#   pww_diplo_submit_bid          bid on a holder's auction lot    5 PP
# A pending flag (pww_diplo_pending, 30 days) stops an AI from tabling a
# second proposal before the first is answered: drafts live on the sender.
###############################################################################

scripted_diplomatic_actions = {

		allowed = { always = yes }
		visible = {
			NOT = { tag = ROOT }

		selectable = {

		send_scripted_gui = pww_diplo_send_offer_gui
		reset_send_effect = {
			set_temp_variable = { pww_dk_target = THIS }
			set_temp_variable = { pww_dk_act = 0 }
			ROOT = {
				if = {
					limit = {
						is_ai = yes
						has_country_flag = pww_diplo_ai_resend
						check_variable = { pww_do_target = pww_dk_target }
						check_variable = { pww_do_act = 0 }
					}
					clr_country_flag = pww_diplo_ai_resend
					pww_diplo_offer_changed = yes
				}
				else = {
					pww_diplo_offer_begin = yes
					# An AI sends the draft it prepared on its tick (kept by
					# offer_begin); it only drafts afresh when none is kept.
					if = {
						limit = {
							is_ai = yes
							check_variable = { pww_dob_keep = 0 }
						}
						if = {
							limit = { check_variable = { pww_diplo_ai_offer_to = pww_dk_target } }
							set_variable = { pww_do_res = pww_diplo_ai_offer_res }
						}
						pww_diplo_ai_build_bundle = yes
						pww_diplo_refresh_display = yes
						pww_diplo_bundle_mirror = yes
					}
				}
			}
		}
		can_be_sent = {
			# AI senders build the draft in reset_send_effect; ai_acceptance
			# rejects anything unworkable.
			OR = {
				ROOT = { is_ai = yes }
				AND = {
						ROOT = {
							check_variable = { pww_do_valid = 0 }
							check_variable = { pww_dd_closed = 0 }
							check_variable = { pww_do_act = 0 }
							check_variable = { pww_do_target = PREV }
						}
				}
			}
		}
		send_description = pww_diplo_offer_send_desc

		on_sent_effect = {
			if = {
				limit = { has_global_flag = pww_diplo_debug }
				log = "[GetDateText] PWW_DIPLO propose sent [ROOT.GetTag]->[THIS.GetTag] sale=[?ROOT.pww_diplo_ai_offer_sell|0] total=[?ROOT.pww_do_e_total|0]"
			}
			ROOT = {
				clr_country_flag = pww_diplo_ai_offer
				set_country_flag = { flag = pww_diplo_pending days = 30 value = 1 }
				pww_diplo_sit_bid_hook = yes
			}
		}
		complete_effect = {
			ROOT = {
				clr_country_flag = pww_diplo_pending
				pww_diplo_talk_accepted = yes
			}
		}
		reject_effect = {
			if = {
				limit = {
					is_ai = no
					ROOT = { is_ai = yes }
				}
				# A human receiver declined or countered an AI proposal.
				set_temp_variable = { pww_dac_counter = pww_dr_counter }
				ROOT = {
					clr_country_flag = pww_diplo_pending
					if = {
						limit = { check_variable = { pww_dac_counter > 0 } }
						pww_diplo_ai_consider_counter = yes
					}
					else = {
						set_temp_variable = { pww_dtc_t = pww_do_talk }
						set_temp_variable = { pww_dtc_status = 2 }
						if = {
							limit = {
								check_variable = { pww_dtc_t > -1 }
								check_variable = { global.pww_dt_status^pww_dtc_t = 0 }
							}
							pww_diplo_talk_close = yes
						}
					}
				}
			}
			else = {

					clr_country_flag = pww_diplo_pending

				}
