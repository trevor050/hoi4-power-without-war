###############################################################################
# Power Without War: stockpiles decisions.
#   pww_stock_build_depot  locks global.pww_stock_c_depot_civs (3) civilian
#                          factories for 120 days, then adds a depot level
#                          (pww_stock_depot_finish). AI countries take it
#                          through pww_stock_ai_depots (activate_decision), so
#                          ai_will_do is 0 here.
###############################################################################

pww_stock_category = {
	pww_stock_build_depot = {
		icon = generic_industry
		priority = 10
		allowed = { always = yes }
		visible = {
			has_global_flag = pww_stock_loaded
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = pww_stock_depot_max_tt
				check_variable = { pww_stock_depots < global.pww_stock_c_depot_max }
			}
			num_of_civilian_factories_available_for_projects > 2
		}
		fire_only_once = no
		cost = 15
		days_remove = 120
		modifier = {
			civilian_factory_use = 3
		}
		complete_effect = {
			hidden_effect = { pww_stock_depot_start = yes }
		}
		remove_effect = {
			custom_effect_tooltip = pww_stock_depot_done_tt
			hidden_effect = { pww_stock_depot_finish = yes }
		}
		ai_will_do = {
			factor = 0
		}
	}
}
