###############################################################################
# Power Without War: diplomacy lane, negotiation (talks, offers, replies)
#
# Storage and enums: header of pww_diplo_core.txt (TALKS, OFFER DRAFT).
#
# Trade talks (Propose Trade Deal, Lease Mines) negotiate a BUNDLE: items on
# both sides with real amounts. The bundle model, its valuation, signing and
# the AI drafts live in pww_diplo_bundle.txt; this file keeps the talk record,
# the replies and the old single-term evaluation that auction bids still use
# (pww_diplo_eval_offer / pww_diplo_sign_draft, read by pww_diplo_auctions.txt
# and pww_diplo_bids.txt only).
#
# How a negotiation runs
#   1. The sender opens Propose Trade Deal or Lease Mines. reset_send_effect
#      calls pww_diplo_offer_begin: it finds or opens the talk with that
#      target (seeding its hidden values once) and loads the bundle.
#   2. Every click calls pww_diplo_offer_changed: it re-evaluates the bundle
#      (pww_diplo_eval_bundle), so the GUI, can_be_sent and the AI
#      acceptance always read current numbers.
#   3. On a rejection pww_diplo_talk_reply runs in reject_effect: it reveals
#      one red line or hidden want (or reports the main obstacle), spends or
#      restores patience, and closes the talks after the third counter or
#      when patience runs out. An identical repeat gets the same answer and
#      reveals nothing.
#   4. On acceptance pww_diplo_sign_bundle turns the bundle into ledger
#      agreements (one per resource or lease, linked; payments and tranches on
#      the main one).
#   Reply line codes (global.pww_dt_line): 11-17 patience low, 21-25 patience
#   restored, 31-37 red line hit, 35 red line again, 40+want want revealed
#   (trade-talks wants, see pww_diplo_bundle.txt), 51-56 reliability doubt,
#   71-76 rival, 81-85 stalling (83 = same terms again), 90 signed, 91 failed
#   to register, 92 closed, 93 superseded by an auction, 94 opening, 95
#   expired, 96/97 an AI sender took / refused a human counter, 101-106 the
#   price is too low (sets global.pww_dt_floor: the "go below market" clue),
#   111-113 too little trust for a fixed contract (bond under
#   global.pww_rel_c_lock and the terms not clearly generous), 121 / 122
#   bound by a producers' agreement (under the agreed price / beyond its
#   quota: pww_do_cartfail).
#
# AUCTION BID VALUES (old single-term model, bids only). Civilian-factory
# equivalents:
#   resource value = units / 8 (fixed vanilla price, 8 units per factory)
#   payment value  = resource value x price factor (0.7 / 1.0 / 1.35);
#                    civilian factories are whole: rounded down below
#                    market, up above it, at least 1; the real count is
#                    what gets valued. A license is 1.5 x price factor.
#   The receiver discounts payment forms it cannot use freely (clearing
#   credit 0.75, equipment 0.85, political power 0.9) and weighs its own
#   desperation: a seller in need loses more (x 1 + desp/100), a desperate
#   buyer gains more (x 1 + desp/50). A lease costs the seller +30% of the
#   value (foreign rights on home soil), 15 points less with a fig leaf.
# Acceptance points of an auction bid (the AI accepts when the total is above
# zero). Trade-talks bundles use the market-floor model in the header of
# pww_diplo_bundle.txt (glut, scarcity-scaled reserve, fair part):
#   terms     (gain - loss) / resource value x 100, clamped -100..100: a
#             percentage, so price cards weigh the same on any deal size
#             (30% below market = -30, 35% above = +35)
#   need      buyer's desperation / 4; a factory-poor seller paid in
#             factories +6 (0 when factories are also its hidden want)
#   want      +15 when the draft carries their hidden want once it is
#             known (revealed, or acknowledged on a counter), +6 before;
#             payment-form wants count only when they are the side paid
#   sweet     sweeteners and fig leaves
#   red line  -1000 when the draft crosses it
#   trust     sender Reliability (frozen at opening), halved if non-committal
#   relations normalized opinion / 4 (frozen at opening)
#   patron    -20 when a patron underwrites them against this sender
#   reserve   their hidden reservation (2..25)
#   repeat    -1000 when the exact draft was already refused
# So a bid below market never clears a neutral, comfortable holder.
###############################################################################

###############################################################################
# Open or find a talk. Sender scope.
# in : pww_dk_target (country), pww_dk_act (0/1), pww_dk_ai_side (0/1)
# out: pww_dk_talk (id, -1 if refused by cooldown), pww_dk_new (1 if opened)
###############################################################################
pww_diplo_talk_find = {
	set_temp_variable = { pww_dk_talk = -1 }
	set_temp_variable = { pww_dk_brk = 0 }
	for_each_loop = {
		array = pww_dt_ids
		value = pww_dk_v
		break = pww_dk_brk
		if = {
			limit = {
				check_variable = { global.pww_dt_status^pww_dk_v = 0 }
				check_variable = { global.pww_dt_target^pww_dk_v = pww_dk_target }
				check_variable = { global.pww_dt_act^pww_dk_v = pww_dk_act }
			}
			set_temp_variable = { pww_dk_talk = pww_dk_v }
			set_temp_variable = { pww_dk_brk = 1 }
		}
	}
}

# Sender scope. in: pww_dk_target. out: pww_dk_cool (1 when a closed talk
# with that target is still cooling down), pww_dk_cool_talk,
# pww_dk_recent (1 when talks with that target opened in the last 90 days:
# AI intents skip such targets so a signed deal is not re-proposed at once).
pww_diplo_talk_cooling = {
	set_temp_variable = { pww_dk_cool = 0 }
	set_temp_variable = { pww_dk_recent = 0 }
	set_temp_variable = { pww_dk_cool_talk = -1 }
	set_temp_variable = { pww_dk_since = global.num_days }
	subtract_from_temp_variable = { pww_dk_since = 90 }
	for_each_loop = {
		array = pww_dt_ids
		value = pww_dk_rv
		if = {
			limit = {
				check_variable = { global.pww_dt_target^pww_dk_rv = pww_dk_target }
				check_variable = { global.pww_dt_opened^pww_dk_rv > pww_dk_since }
				NOT = { check_variable = { global.pww_dt_status^pww_dk_rv = 0 } }
			}
			set_temp_variable = { pww_dk_recent = 1 }
		}
	}
	for_each_loop = {
		array = pww_dt_ids
		value = pww_dk_cv
		if = {
			limit = {
				check_variable = { global.pww_dt_target^pww_dk_cv = pww_dk_target }
				check_variable = { global.pww_dt_status^pww_dk_cv = 2 }
				check_variable = { global.pww_dt_cool_until^pww_dk_cv > global.num_days }
			}
			set_temp_variable = { pww_dk_cool = 1 }
			set_temp_variable = { pww_dk_cool_talk = pww_dk_cv }
		}
	}
}

pww_diplo_talk_open = {
	pww_diplo_init = yes
	set_temp_variable = { pww_dk_new = 0 }
	pww_diplo_talk_find = yes
	if = {
		limit = { check_variable = { pww_dk_talk < 0 } }
		pww_diplo_talk_cooling = yes
		if = {
			limit = { check_variable = { pww_dk_cool = 0 } }
			pww_diplo_talk_create = yes
		}
	}
}

# Sender scope. Appends a new talk record and seeds its hidden values.
pww_diplo_talk_create = {
	set_temp_variable = { pww_dk_talk = global.pww_next_talk_id }
	add_to_variable = { global.pww_next_talk_id = 1 }
	set_temp_variable = { pww_dk_new = 1 }
	set_temp_variable = { pww_dk_expires = global.num_days }
	add_to_temp_variable = { pww_dk_expires = global.pww_diplo_c_talk_days }

	# Seed: campaign salt + the pair (core buckets are fixed per country) +
	# the talk id. Stable across save and reload.
	set_temp_variable = { pww_dk_seed = global.pww_diplo_salt }
	set_temp_variable = { pww_dk_b = pww_bucket }
	multiply_temp_variable = { pww_dk_b = 17 }
	add_to_temp_variable = { pww_dk_seed = pww_dk_b }
	var:pww_dk_target = { set_temp_variable = { pww_dk_b = pww_bucket } }
	multiply_temp_variable = { pww_dk_b = 31 }
	add_to_temp_variable = { pww_dk_seed = pww_dk_b }
	set_temp_variable = { pww_dk_b = pww_dk_talk }
	multiply_temp_variable = { pww_dk_b = 7 }
	add_to_temp_variable = { pww_dk_seed = pww_dk_b }
	modulo_temp_variable = { var = pww_dk_seed value = 97 }

	# The AI party being read: the target, or the sender when an AI tables
	# terms to a human.
	if = {
		limit = { check_variable = { pww_dk_ai_side = 1 } }
		set_temp_variable = { pww_dk_party = THIS }
		set_temp_variable = { pww_dk_other = pww_dk_target }
	}
	else = {
		set_temp_variable = { pww_dk_party = pww_dk_target }
		set_temp_variable = { pww_dk_other = THIS }
	}

	# Red line candidates
	set_temp_variable = { pww_dpk_found = 0 }
	clear_temp_array = pww_dk_cand
	if = {
		limit = { check_variable = { pww_dk_act = 1 } }
		add_to_temp_array = { pww_dk_cand = 0 }
		add_to_temp_array = { pww_dk_cand = 1 }
		add_to_temp_array = { pww_dk_cand = 3 }
		add_to_temp_array = { pww_dk_cand = 7 }
		var:pww_dk_party = {
			if = {
				limit = { NOT = { has_government = democratic } }
				add_to_temp_array = { pww_dk_cand = 0 }
				add_to_temp_array = { pww_dk_cand = 0 }
			}
		}
	}
	else = {
		add_to_temp_array = { pww_dk_cand = 1 }
		add_to_temp_array = { pww_dk_cand = 2 }
		add_to_temp_array = { pww_dk_cand = 3 }
		add_to_temp_array = { pww_dk_cand = 5 }
		add_to_temp_array = { pww_dk_cand = 7 }
	}
	var:pww_dk_party = {
		if = {
			limit = { has_government = democratic }
			add_to_temp_array = { pww_dk_cand = 3 }
		}
		set_temp_variable = { pww_dpk_claimant = pww_dk_other }
		pww_diplo_pledged_against = yes
		if = {
			limit = { check_variable = { pww_dpk_found = 1 } }
			add_to_temp_array = { pww_dk_cand = 6 }
			add_to_temp_array = { pww_dk_cand = 6 }
		}
	}
	set_temp_variable = { pww_dk_r = pww_dk_seed }
	multiply_temp_variable = { pww_dk_r = 5 }
	add_to_temp_variable = { pww_dk_r = 13 }
	modulo_temp_variable = { var = pww_dk_r value = pww_dk_cand^num }
	set_temp_variable = { pww_dk_redline = pww_dk_cand^pww_dk_r }

	# Hidden want candidates
	set_temp_variable = { pww_dk_coal_ok = 0 }
	clear_temp_array = pww_dk_cand
	add_to_temp_array = { pww_dk_cand = 3 }
	add_to_temp_array = { pww_dk_cand = 4 }
	add_to_temp_array = { pww_dk_cand = 5 }
	add_to_temp_array = { pww_dk_cand = 6 }
	var:pww_dk_other = {
		if = {
			limit = { num_of_military_factories > 9 }
			add_to_temp_array = { pww_dk_cand = 0 }
			add_to_temp_array = { pww_dk_cand = 0 }
		}
		if = {
			limit = { pww_core_produces_coal = yes }
			set_temp_variable = { pww_dk_coal_ok = 1 }
		}
	}
	var:pww_dk_party = {
		if = {
			limit = { num_of_civilian_factories < 25 }
			add_to_temp_array = { pww_dk_cand = 1 }
			add_to_temp_array = { pww_dk_cand = 1 }
		}
		if = {
			limit = {
				pww_core_has_deficit_coal = yes
				check_variable = { pww_dk_coal_ok = 1 }
			}
			add_to_temp_array = { pww_dk_cand = 2 }
			add_to_temp_array = { pww_dk_cand = 2 }
		}
		if = {
			limit = { check_variable = { pww_dpk_found = 1 } }
			add_to_temp_array = { pww_dk_cand = 8 }
		}
	}
	set_temp_variable = { pww_dk_r = pww_dk_seed }
	multiply_temp_variable = { pww_dk_r = 7 }
	add_to_temp_variable = { pww_dk_r = 29 }
	modulo_temp_variable = { var = pww_dk_r value = pww_dk_cand^num }
	set_temp_variable = { pww_dk_want = pww_dk_cand^pww_dk_r }

	# Frozen relations and trust (party's view of the other side)
	set_temp_variable = { pww_opinion_target = pww_dk_other }
	var:pww_dk_party = { pww_core_norm_opinion = yes }
	set_temp_variable = { pww_dk_rel = pww_norm_opinion }
	divide_temp_variable = { pww_dk_rel = 4 }
	round_temp_variable = pww_dk_rel
	var:pww_dk_other = {
		pww_ledger_rep_ensure = yes
		set_temp_variable = { pww_dk_trust = pww_reliability }
	}
	subtract_from_temp_variable = { pww_dk_trust = 50 }
	divide_temp_variable = { pww_dk_trust = 2 }
	round_temp_variable = pww_dk_trust

	# Patience 45..80 shifted by relations; reserve 2..25 (on a plain
	# at-market deal about one seed in four signs at once)
	set_temp_variable = { pww_dk_pat = pww_dk_seed }
	multiply_temp_variable = { pww_dk_pat = 11 }
	add_to_temp_variable = { pww_dk_pat = 3 }
	modulo_temp_variable = { var = pww_dk_pat value = 36 }
	add_to_temp_variable = { pww_dk_pat = 45 }
	set_temp_variable = { pww_dk_b = pww_dk_rel }
	divide_temp_variable = { pww_dk_b = 2 }
	add_to_temp_variable = { pww_dk_pat = pww_dk_b }
	# Trade bond (either direction, the stronger): old partners sit longer
	# at the table (+ bond / 4, up to +25).
	set_temp_variable = { pww_dbo_s = pww_dk_party }
	set_temp_variable = { pww_dbo_b = pww_dk_other }
	pww_diplo_bond = yes
	set_temp_variable = { pww_dk_bond = pww_dbo_v }
	set_temp_variable = { pww_dbo_s = pww_dk_other }
	set_temp_variable = { pww_dbo_b = pww_dk_party }
	pww_diplo_bond = yes
	if = {
		limit = { check_variable = { pww_dbo_v > pww_dk_bond } }
		set_temp_variable = { pww_dk_bond = pww_dbo_v }
	}
	divide_temp_variable = { pww_dk_bond = 4 }
	add_to_temp_variable = { pww_dk_pat = pww_dk_bond }
	clamp_temp_variable = { var = pww_dk_pat min = 20 max = 95 }
	round_temp_variable = pww_dk_pat
	set_temp_variable = { pww_dk_res = pww_dk_seed }
	multiply_temp_variable = { pww_dk_res = 13 }
	add_to_temp_variable = { pww_dk_res = 41 }
	modulo_temp_variable = { var = pww_dk_res value = 24 }
	add_to_temp_variable = { pww_dk_res = 2 }

	# Shared situation
	set_temp_variable = { pww_dg_a = THIS }
	set_temp_variable = { pww_dg_b = pww_dk_target }
	pww_diplo_find_shared_sit = yes

	add_to_array = { array = global.pww_dt_sender value = THIS }
	add_to_array = { array = global.pww_dt_target value = pww_dk_target }
	add_to_array = { array = global.pww_dt_act value = pww_dk_act }
	add_to_array = { array = global.pww_dt_status value = 0 }
	add_to_array = { array = global.pww_dt_opened value = global.num_days }
	add_to_array = { array = global.pww_dt_expires value = pww_dk_expires }
	add_to_array = { array = global.pww_dt_cool_until value = 0 }
	add_to_array = { array = global.pww_dt_round value = 0 }
	add_to_array = { array = global.pww_dt_patience value = pww_dk_pat }
	add_to_array = { array = global.pww_dt_redline value = pww_dk_redline }
	add_to_array = { array = global.pww_dt_want value = pww_dk_want }
	add_to_array = { array = global.pww_dt_rl_seen value = 0 }
	add_to_array = { array = global.pww_dt_want_seen value = 0 }
	add_to_array = { array = global.pww_dt_reserve value = pww_dk_res }
	add_to_array = { array = global.pww_dt_seed value = pww_dk_seed }
	add_to_array = { array = global.pww_dt_rel value = pww_dk_rel }
	add_to_array = { array = global.pww_dt_trust value = pww_dk_trust }
	add_to_array = { array = global.pww_dt_last_sig value = -1 }
	add_to_array = { array = global.pww_dt_last_ok value = 0 }
	add_to_array = { array = global.pww_dt_line value = 94 }
	add_to_array = { array = global.pww_dt_sit value = pww_dg_sit }
	add_to_array = { array = global.pww_dt_ag value = -1 }
	add_to_array = { array = global.pww_dt_ai_side value = pww_dk_ai_side }
	add_to_array = { array = global.pww_dt_floor value = 0 }
	add_to_array = { array = pww_dt_ids value = pww_dk_talk }
	var:pww_dk_target = { add_to_array = { array = pww_dt_in value = pww_dk_talk } }
	add_to_array = { array = global.pww_dt_live value = pww_dk_talk }

	set_temp_variable = { pww_dq_kind_in = 1 }
	set_temp_variable = { pww_dq_a_in = THIS }
	set_temp_variable = { pww_dq_b_in = pww_dk_target }
	set_temp_variable = { pww_dq_ref_in = pww_dk_talk }
	set_temp_variable = { pww_dq_ag_in = -1 }
	set_temp_variable = { pww_dq_sit_in = pww_dg_sit }
	set_temp_variable = { pww_dq_x_in = pww_dk_act }
	pww_diplo_queue_push = yes
	if = {
		limit = { has_global_flag = pww_diplo_debug }
		log = "[GetDateText] PWW_DIPLO talk open id=[?pww_dk_talk|0] [THIS.GetTag]->[?pww_dk_target.GetTag] act=[?pww_dk_act|0] seed=[?pww_dk_seed|0] redline=[?pww_dk_redline|0] want=[?pww_dk_want|0] patience=[?pww_dk_pat|0] reserve=[?pww_dk_res|0] rel=[?pww_dk_rel|0] trust=[?pww_dk_trust|0]"
	}
}

###############################################################################
# Begin an offer. Sender scope. in: pww_dk_target, pww_dk_act.
# Opens or resumes talks and loads the draft. A draft for the same target and
# act is kept between visits (the GUI remembers the last terms).
###############################################################################
pww_diplo_offer_begin = {
	set_temp_variable = { pww_dk_ai_side = 0 }
	if = {
		limit = {
			is_ai = yes
			var:pww_dk_target = { is_ai = no }
		}
		set_temp_variable = { pww_dk_ai_side = 1 }
	}
	pww_diplo_talk_open = yes
	set_temp_variable = { pww_dob_keep = 0 }
	if = {
		limit = {
			check_variable = { pww_do_target = pww_dk_target }
			check_variable = { pww_do_act = pww_dk_act }
			check_variable = { pww_do_talk = pww_dk_talk }
			check_variable = { pww_dob_k^num = 8 }
			has_variable = pww_do_n1
		}
		set_temp_variable = { pww_dob_keep = 1 }
	}
	set_variable = { pww_do_target = pww_dk_target }
	set_variable = { pww_do_act = pww_dk_act }
	set_variable = { pww_do_talk = pww_dk_talk }
	set_variable = { pww_dbm_side = -1 }
	set_variable = { pww_dbm_page = 0 }
	if = {
		limit = { check_variable = { pww_do_act = 2 } }
		# Poach Supplier: the incumbent may have changed since the last visit.
		set_temp_variable = { pww_dpq_ag = pww_do_pinc_ag }
		pww_diplo_poach_check = yes
		if = {
			limit = { check_variable = { pww_dpq_ok = 0 } }
			set_temp_variable = { pww_dob_keep = 0 }
		}
	}
	if = {
		limit = { check_variable = { pww_dob_keep = 0 } }
		if = {
			limit = { check_variable = { pww_do_act = 2 } }
			pww_diplo_poach_defaults = yes
		}
		else = {
			pww_diplo_bundle_defaults = yes
		}
	}
	pww_diplo_offer_changed = yes
}

# Sender scope (act 2). Picks the incumbent (pww_diplo_poach_pick) and drafts
# a better delivered deal: the resource they buy, as many units as now (at
# most half our output), for 90% of the incumbent's price a unit (market when
# it has none), less our shipping, paid in their free civilian factories and
# political power. One year, either side may end it.
pww_diplo_poach_defaults = {
	pww_diplo_bundle_clear = yes
	set_variable = { pww_do_len = 1 }
	set_variable = { pww_do_bind = 0 }
	pww_diplo_poach_pick = yes
	set_temp_variable = { pww_dpd_ag = pww_do_pinc_ag }
	if = {
		limit = { check_variable = { pww_dpd_ag > -1 } }
		set_temp_variable = { pww_dpd_r = global.pww_ag_res^pww_dpd_ag }
		set_temp_variable = { pww_dpd_u = global.pww_ag_amount^pww_dpd_ag }
		set_temp_variable = { pww_dg_res = pww_dpd_r }
		pww_diplo_read_res = yes
		set_temp_variable = { pww_dpd_cap = pww_dg_prod }
		multiply_temp_variable = { pww_dpd_cap = 0.5 }
		if = {
			limit = { check_variable = { pww_dpd_u > pww_dpd_cap } }
			set_temp_variable = { pww_dpd_u = pww_dpd_cap }
		}
		round_temp_variable = pww_dpd_u
		if = {
			limit = { check_variable = { pww_dpd_u < 1 } }
			set_temp_variable = { pww_dpd_u = 1 }
		}
		set_temp_variable = { pww_dba_side = 0 }
		set_temp_variable = { pww_dba_kind = 3 }
		set_temp_variable = { pww_dba_x = pww_dpd_r }
		pww_diplo_bundle_add = yes
		if = {
			limit = { check_variable = { pww_dba_new > -1 } }
			set_variable = { pww_dob_a^pww_dba_new = pww_dpd_u }
		}
		pww_diplo_eval_bundle = yes
		# Price factor: 90% of the incumbent's price per unit (vs market),
		# 5% less each round of the talk.
		set_temp_variable = { pww_dpd_m = 0.9 }
		set_temp_variable = { pww_dpd_t = pww_do_talk }
		if = {
			limit = {
				check_variable = { pww_dpd_t > -1 }
				check_variable = { pww_dpd_t < global.pww_next_talk_id }
			}
			set_temp_variable = { pww_dpd_rd = global.pww_dt_round^pww_dpd_t }
			multiply_temp_variable = { pww_dpd_rd = 0.05 }
			subtract_from_temp_variable = { pww_dpd_m = pww_dpd_rd }
		}
		set_temp_variable = { pww_dpd_pf = pww_dpd_m }
		set_temp_variable = { pww_dpd_ia = global.pww_ag_amount^pww_dpd_ag }
		if = {
			limit = { check_variable = { pww_dpd_ia > 0.999 } }
			set_temp_variable = { pww_dpd_pf = pww_do_p_pay }
			multiply_temp_variable = { pww_dpd_pf = 8 }
			divide_temp_variable = { pww_dpd_pf = pww_dpd_ia }
			multiply_temp_variable = { pww_dpd_pf = pww_dpd_m }
		}
		set_temp_variable = { pww_dsh_s = THIS }
		set_temp_variable = { pww_dsh_b = pww_do_target }
		set_temp_variable = { pww_dsh_r = pww_dpd_r }
		pww_diplo_ship = yes
		subtract_from_temp_variable = { pww_dpd_pf = pww_dsh_share }
		clamp_temp_variable = { var = pww_dpd_pf min = 0.3 max = 1.5 }
		set_temp_variable = { pww_dsl_ask = pww_do_mk_m0 }
		multiply_temp_variable = { pww_dsl_ask = pww_dpd_pf }
		set_temp_variable = { pww_dsl_rl = -1 }
		pww_diplo_bundle_ask_pay = yes
	}
}

# Sender scope. Re-evaluate and refresh every display mirror.
pww_diplo_offer_changed = {
	# Any edit clears the Balance it message (balance sets it after this).
	set_variable = { pww_dbl_msg = 0 }
	pww_diplo_eval_bundle = yes
	pww_diplo_refresh_display = yes
	pww_diplo_bundle_mirror = yes
	add_to_variable = { pww_diplo_dirty = 1 }
}

# Sender scope. Display mirrors for the send GUI (read as ROOT.*).
pww_diplo_refresh_display = {
	set_temp_variable = { pww_drd_t = pww_do_talk }
	set_variable = { pww_dd_rl = -1 }
	set_variable = { pww_dd_want = -1 }
	set_variable = { pww_dd_line = 0 }
	set_variable = { pww_dd_round = 0 }
	set_variable = { pww_dd_closed = 0 }
	set_variable = { pww_dd_floor = 0 }
	set_variable = { pww_dd_pat = 0 }
	if = {
		limit = {
			check_variable = { pww_drd_t > -1 }
			check_variable = { pww_drd_t < global.pww_next_talk_id }
		}
		if = {
			limit = { check_variable = { global.pww_dt_rl_seen^pww_drd_t = 1 } }
			set_variable = { pww_dd_rl = global.pww_dt_redline^pww_drd_t }
		}
		if = {
			limit = { check_variable = { global.pww_dt_want_seen^pww_drd_t = 1 } }
			set_variable = { pww_dd_want = global.pww_dt_want^pww_drd_t }
		}
		set_variable = { pww_dd_line = global.pww_dt_line^pww_drd_t }
		set_variable = { pww_dd_round = global.pww_dt_round^pww_drd_t }
		set_variable = { pww_dd_floor = global.pww_dt_floor^pww_drd_t }
		# Patience in words: 0 nearly gone, 1 running thin, 2 steady, 3 patient.
		# Bands follow the reply thresholds (< 35 turns curt, < 15 ends talks), so a fresh
		# talk (seeded 45..80, shifted by relations) reads steady unless the party is hostile.
		set_temp_variable = { pww_drd_p = global.pww_dt_patience^pww_drd_t }
		if = {
			limit = { check_variable = { pww_drd_p > 64 } }
			set_variable = { pww_dd_pat = 3 }
		}
		else_if = {
			limit = { check_variable = { pww_drd_p > 34 } }
			set_variable = { pww_dd_pat = 2 }
		}
		else_if = {
			limit = { check_variable = { pww_drd_p > 14 } }
			set_variable = { pww_dd_pat = 1 }
		}
		if = {
			limit = { NOT = { check_variable = { global.pww_dt_status^pww_drd_t = 0 } } }
			set_variable = { pww_dd_closed = 1 }
		}
	}
	else = {
		set_variable = { pww_dd_closed = 1 }
		set_variable = { pww_dd_line = 92 }
	}
	# Patron name for the rival envoy lines, on the target
	var:pww_do_target = {
		if = {
			limit = { check_variable = { pww_dpl_ids^num > 0 } }
			for_each_loop = {
				array = pww_dpl_ids
				value = pww_drd_p
				if = {
					limit = { check_variable = { global.pww_dpl_status^pww_drd_p = 0 } }
					set_variable = { pww_dd_patron = global.pww_dpl_patron^pww_drd_p }
				}
			}
		}
	}
}

###############################################################################
# Evaluate the draft. Sender scope; target = pww_do_target.
# Writes every derived pww_do_* variable (see the core header).
###############################################################################
pww_diplo_eval_offer = {
	set_temp_variable = { pww_dv_t = pww_do_target }
	set_temp_variable = { pww_dv_talk = pww_do_talk }
	set_variable = { pww_do_valid = 0 }
	set_variable = { pww_do_crosses = 0 }

	# Roles: the receiver (target) sells in a lease or when we buy.
	if = {
		limit = {
			OR = {
				check_variable = { pww_do_act = 1 }
				check_variable = { pww_do_dir = 0 }
			}
		}
		set_variable = { pww_do_seller = pww_dv_t }
		set_variable = { pww_do_buyer = THIS }
		set_temp_variable = { pww_dv_rcv_sells = 1 }
	}
	else = {
		set_variable = { pww_do_seller = THIS }
		set_variable = { pww_do_buyer = pww_dv_t }
		set_temp_variable = { pww_dv_rcv_sells = 0 }
	}
	set_temp_variable = { pww_dg_res = pww_do_res }
	var:pww_do_seller = { pww_diplo_read_res = yes }
	set_temp_variable = { pww_dv_s_prod = pww_dg_prod }
	set_temp_variable = { pww_dv_s_desp = pww_dg_desp }
	var:pww_do_buyer = { pww_diplo_read_res = yes }
	set_temp_variable = { pww_dv_b_desp = pww_dg_desp }

	# Units
	set_temp_variable = { pww_dv_units = 0 }
	if = {
		limit = { check_variable = { pww_do_act = 0 } }
		set_temp_variable = { pww_dv_share = 0.15 }
		if = {
			limit = { check_variable = { pww_do_amt = 1 } }
			set_temp_variable = { pww_dv_share = 0.3 }
		}
		else_if = {
			limit = { check_variable = { pww_do_amt = 2 } }
			set_temp_variable = { pww_dv_share = 0.5 }
		}
		else_if = {
			limit = { check_variable = { pww_do_amt = 3 } }
			set_temp_variable = { pww_dv_share = 0.75 }
		}
		set_temp_variable = { pww_dv_units = pww_dv_s_prod }
		multiply_temp_variable = { pww_dv_units = pww_dv_share }
		round_temp_variable = pww_dv_units
		if = {
			limit = { check_variable = { pww_dv_units < 1 } }
			set_temp_variable = { pww_dv_units = 1 }
		}
		if = {
			limit = { check_variable = { pww_dv_s_prod < 1 } }
			set_variable = { pww_do_valid = 1 }
		}
		clear_variable = pww_do_state
	}
	else = {
		set_temp_variable = { pww_dg_slot = pww_do_slot }
		var:pww_dv_t = { pww_diplo_pick_state = yes }
		set_variable = { pww_do_slot = pww_dg_slot }
		set_variable = { pww_do_nstates = pww_dg_nstates }
		if = {
			limit = { check_variable = { pww_dg_nstates < 1 } }
			set_variable = { pww_do_valid = 2 }
			clear_variable = pww_do_state
		}
		else = {
			set_variable = { pww_do_state = pww_dg_state }
			set_variable = { pww_do_state_s = pww_dg_state_s }
			var:pww_dv_t = { set_variable = { pww_do_state_s = pww_dg_state_s } }
			var:pww_dg_state = { pww_diplo_state_units = yes }
			set_temp_variable = { pww_dv_units = pww_dg_units }
		}
		if = {
			limit = { check_variable = { pww_do_dur = 0 } }
			set_variable = { pww_do_valid = 5 }
		}
	}
	set_variable = { pww_do_units = pww_dv_units }
	set_temp_variable = { pww_dv_vres = pww_dv_units }
	divide_temp_variable = { pww_dv_vres = 8 }

	# Term
	set_variable = { pww_do_term = 0 }
	if = {
		limit = { check_variable = { pww_do_dur = 1 } }
		set_variable = { pww_do_term = 180 }
	}
	else_if = {
		limit = { check_variable = { pww_do_dur = 2 } }
		set_variable = { pww_do_term = 365 }
	}
	else_if = {
		limit = { check_variable = { pww_do_dur = 3 } }
		set_variable = { pww_do_term = 1095 }
	}
	else_if = {
		limit = { check_variable = { pww_do_dur = 4 } }
		set_variable = { pww_do_term = 1825 }
	}
	if = {
		limit = {
			check_variable = { pww_do_dur = 0 }
			check_variable = { pww_do_pay > 1 }
		}
		set_variable = { pww_do_valid = 3 }
	}

	# Payment
	set_temp_variable = { pww_dv_pf = 1 }
	if = {
		limit = { check_variable = { pww_do_price = 0 } }
		set_temp_variable = { pww_dv_pf = 0.7 }
	}
	else_if = {
		limit = { check_variable = { pww_do_price = 2 } }
		set_temp_variable = { pww_dv_pf = 1.35 }
	}
	set_temp_variable = { pww_dv_vpay = pww_dv_vres }
	multiply_temp_variable = { pww_dv_vpay = pww_dv_pf }
	set_variable = { pww_do_civ = 0 }
	set_variable = { pww_do_pp = 0 }
	set_variable = { pww_do_cic = 0 }
	if = {
		limit = { check_variable = { pww_do_pay = 0 } }
		# Whole factories, rounded in the direction of the price card (down
		# below market, up above it) so the three prices stay distinct where
		# the deal is large enough. The evaluation uses the real count.
		set_temp_variable = { pww_dv_civ = pww_dv_vpay }
		if = {
			limit = { check_variable = { pww_do_price = 0 } }
			subtract_from_temp_variable = { pww_dv_civ = 0.499 }
		}
		else_if = {
			limit = { check_variable = { pww_do_price = 2 } }
			add_to_temp_variable = { pww_dv_civ = 0.499 }
		}
		round_temp_variable = pww_dv_civ
		if = {
			limit = { check_variable = { pww_dv_civ < 1 } }
			set_temp_variable = { pww_dv_civ = 1 }
		}
		set_variable = { pww_do_civ = pww_dv_civ }
		set_temp_variable = { pww_dv_vpay = pww_dv_civ }
	}
	else_if = {
		limit = { check_variable = { pww_do_pay = 1 } }
		set_temp_variable = { pww_dv_pp = pww_dv_vpay }
		multiply_temp_variable = { pww_dv_pp = 0.1 }
		if = {
			limit = { check_variable = { pww_dv_pp < 0.05 } }
			set_temp_variable = { pww_dv_pp = 0.05 }
		}
		set_variable = { pww_do_pp = pww_dv_pp }
	}
	else_if = {
		limit = { check_variable = { pww_do_pay = 4 } }
		set_temp_variable = { pww_dv_vpay = 1.5 }
		multiply_temp_variable = { pww_dv_vpay = pww_dv_pf }
	}
	else = {
		# Clearing credit or equipment: three tranches over the term, each
		# worth a third of the payment flow (5 CIC per factory-day).
		set_temp_variable = { pww_dv_cic = pww_dv_vpay }
		multiply_temp_variable = { pww_dv_cic = 5 }
		multiply_temp_variable = { pww_dv_cic = pww_do_term }
		divide_temp_variable = { pww_dv_cic = 3 }
		clamp_temp_variable = { var = pww_dv_cic min = 50 max = 5000 }
		round_temp_variable = pww_dv_cic
		set_variable = { pww_do_cic = pww_dv_cic }
	}

	# Sweetener validity: a coal bundle needs our own coal surplus.
	if = {
		limit = { check_variable = { pww_do_sweet = 1 } }
		set_temp_variable = { pww_dg_res = 6 }
		pww_diplo_read_res = yes
		if = {
			limit = { check_variable = { pww_dg_net < 2 } }
			set_variable = { pww_do_valid = 4 }
		}
	}
	# Same direction check for the sender selling: it must produce the resource.
	if = {
		limit = {
			check_variable = { pww_dv_rcv_sells = 0 }
			check_variable = { pww_dv_s_prod < 1 }
		}
		set_variable = { pww_do_valid = 6 }
	}

	# --- Acceptance parts (the receiver's view) -----------------------------
	set_temp_variable = { pww_dv_mult = 1 }
	if = {
		limit = { check_variable = { pww_do_pay = 1 } }
		set_temp_variable = { pww_dv_mult = 0.9 }
	}
	else_if = {
		limit = { check_variable = { pww_do_pay = 2 } }
		set_temp_variable = { pww_dv_mult = 0.75 }
	}
	else_if = {
		limit = { check_variable = { pww_do_pay = 3 } }
		set_temp_variable = { pww_dv_mult = 0.85 }
	}
	set_temp_variable = { pww_dv_fig_cut = 0 }
	if = {
		limit = { check_variable = { pww_do_fig > 0 } }
		set_temp_variable = { pww_dv_fig_cut = 0.15 }
	}
	if = {
		limit = { check_variable = { pww_dv_rcv_sells = 1 } }
		set_temp_variable = { pww_dv_gain = pww_dv_vpay }
		multiply_temp_variable = { pww_dv_gain = pww_dv_mult }
		set_temp_variable = { pww_dv_loss = pww_dv_s_desp }
		divide_temp_variable = { pww_dv_loss = 100 }
		add_to_temp_variable = { pww_dv_loss = 1 }
		if = {
			limit = { check_variable = { pww_do_act = 1 } }
			# Foreign rights on home soil: +30% of the value, a fig leaf
			# takes back 15 points.
			add_to_temp_variable = { pww_dv_loss = 0.3 }
			subtract_from_temp_variable = { pww_dv_loss = pww_dv_fig_cut }
		}
		multiply_temp_variable = { pww_dv_loss = pww_dv_vres }
	}
	else = {
		set_temp_variable = { pww_dv_gain = pww_dv_b_desp }
		divide_temp_variable = { pww_dv_gain = 50 }
		add_to_temp_variable = { pww_dv_gain = 1 }
		multiply_temp_variable = { pww_dv_gain = pww_dv_vres }
		set_temp_variable = { pww_dv_loss = pww_dv_vpay }
		multiply_temp_variable = { pww_dv_loss = pww_dv_mult }
	}
	# Terms are the receiver's surplus as a percentage of the goods' market
	# value, so a price card means the same on a one-factory deal as on a
	# ten-factory one: 30% below market is -30, 35% above is +35.
	set_temp_variable = { pww_dv_terms = pww_dv_gain }
	subtract_from_temp_variable = { pww_dv_terms = pww_dv_loss }
	set_temp_variable = { pww_dv_base = pww_dv_vres }
	if = {
		limit = { check_variable = { pww_dv_base < 0.25 } }
		set_temp_variable = { pww_dv_base = 0.25 }
	}
	divide_temp_variable = { pww_dv_terms = pww_dv_base }
	multiply_temp_variable = { pww_dv_terms = 100 }
	clamp_temp_variable = { var = pww_dv_terms min = -100 max = 100 }
	round_temp_variable = pww_dv_terms
	set_variable = { pww_do_e_terms = pww_dv_terms }

	set_variable = { pww_do_e_need = 0 }
	if = {
		limit = { check_variable = { pww_dv_rcv_sells = 0 } }
		set_temp_variable = { pww_dv_need = pww_dv_b_desp }
		divide_temp_variable = { pww_dv_need = 4 }
		round_temp_variable = pww_dv_need
		set_variable = { pww_do_e_need = pww_dv_need }
	}
	else_if = {
		limit = {
			check_variable = { pww_do_pay = 0 }
			var:pww_dv_t = { num_of_civilian_factories < 20 }
		}
		set_variable = { pww_do_e_need = 6 }
	}

	# Talk-dependent parts
	set_variable = { pww_do_e_want = 0 }
	set_variable = { pww_do_e_red = 0 }
	set_variable = { pww_do_e_trust = 0 }
	set_variable = { pww_do_e_rel = 0 }
	set_variable = { pww_do_e_reserve = -10 }
	set_variable = { pww_do_e_fair = 0 }
	set_variable = { pww_do_e_repeat = 0 }
	if = {
		limit = {
			check_variable = { pww_dv_talk > -1 }
			check_variable = { pww_dv_talk < global.pww_next_talk_id }
		}
		set_temp_variable = { pww_dv_w = global.pww_dt_want^pww_dv_talk }
		set_temp_variable = { pww_dv_r = global.pww_dt_redline^pww_dv_talk }
		pww_diplo_draft_meets_want = yes
		if = {
			limit = { check_variable = { pww_dv_meets = 1 } }
			# Full weight once they have told us (the haggle loop: counter,
			# reveal, redraft); a lucky guess before that counts for less.
			set_variable = { pww_do_e_want = 6 }
			if = {
				limit = { check_variable = { global.pww_dt_want_seen^pww_dv_talk = 1 } }
				set_variable = { pww_do_e_want = 15 }
			}
			# A factory-poor seller that wants factories counts them once.
			if = {
				limit = { check_variable = { pww_dv_w = 1 } }
				set_variable = { pww_do_e_need = 0 }
			}
		}
		pww_diplo_draft_crosses_redline = yes
		if = {
			limit = { check_variable = { pww_dv_cross = 1 } }
			set_variable = { pww_do_e_red = -1000 }
			set_variable = { pww_do_crosses = 1 }
		}
		set_variable = { pww_do_e_trust = global.pww_dt_trust^pww_dv_talk }
		if = {
			limit = { check_variable = { pww_do_dur = 0 } }
			divide_variable = { pww_do_e_trust = 2 }
			round_variable = pww_do_e_trust
		}
		set_variable = { pww_do_e_rel = global.pww_dt_rel^pww_dv_talk }
		set_variable = { pww_do_e_reserve = global.pww_dt_reserve^pww_dv_talk }
		multiply_variable = { pww_do_e_reserve = -1 }
	}

	# Sweeteners and fig leaves
	set_variable = { pww_do_e_sweet = 0 }
	if = {
		limit = { check_variable = { pww_do_sweet > 0 } }
		add_to_variable = { pww_do_e_sweet = 6 }
		if = {
			limit = {
				check_variable = { pww_do_sweet = 1 }
				var:pww_dv_t = { pww_core_has_deficit_coal = yes }
			}
			add_to_variable = { pww_do_e_sweet = 10 }
		}
	}
	if = {
		limit = {
			check_variable = { pww_do_fig > 0 }
			check_variable = { pww_do_act = 0 }
		}
		add_to_variable = { pww_do_e_sweet = 4 }
	}

	# Patron underwriting the target against us
	set_variable = { pww_do_e_patron = 0 }
	var:pww_dv_t = {
		set_temp_variable = { pww_dpk_claimant = PREV }
		pww_diplo_pledged_against = yes
	}
	if = {
		limit = { check_variable = { pww_dpk_found = 1 } }
		set_variable = { pww_do_e_patron = -20 }
	}

	pww_diplo_compute_sig = yes
	if = {
		limit = {
			check_variable = { pww_dv_talk > -1 }
			check_variable = { pww_dv_talk < global.pww_next_talk_id }
			check_variable = { global.pww_dt_last_sig^pww_dv_talk = pww_do_sig }
			check_variable = { global.pww_dt_last_ok^pww_dv_talk = 0 }
		}
		set_variable = { pww_do_e_repeat = -1000 }
	}

	set_variable = { pww_do_e_invalid = 0 }
	if = {
		limit = { NOT = { check_variable = { pww_do_valid = 0 } } }
		set_variable = { pww_do_e_invalid = -1000 }
	}

	set_variable = { pww_do_e_total = pww_do_e_terms }
	add_to_variable = { pww_do_e_total = pww_do_e_need }
	add_to_variable = { pww_do_e_total = pww_do_e_want }
	add_to_variable = { pww_do_e_total = pww_do_e_red }
	add_to_variable = { pww_do_e_total = pww_do_e_trust }
	add_to_variable = { pww_do_e_total = pww_do_e_rel }
	add_to_variable = { pww_do_e_total = pww_do_e_reserve }
	add_to_variable = { pww_do_e_total = pww_do_e_sweet }
	add_to_variable = { pww_do_e_total = pww_do_e_patron }
	add_to_variable = { pww_do_e_total = pww_do_e_repeat }
	add_to_variable = { pww_do_e_total = pww_do_e_invalid }
	set_variable = { pww_do_ok = 0 }
	if = {
		limit = { check_variable = { pww_do_e_total > 0 } }
		set_variable = { pww_do_ok = 1 }
	}

	# Bands from the sender's view
	if = {
		limit = { check_variable = { pww_dv_rcv_sells = 1 } }
		set_temp_variable = { pww_band_value = pww_dv_vpay }
		pww_core_value_to_band = yes
		set_variable = { pww_do_band_give = pww_band }
		set_temp_variable = { pww_band_value = pww_dv_vres }
		pww_core_value_to_band = yes
		set_variable = { pww_do_band_get = pww_band }
	}
	else = {
		set_temp_variable = { pww_band_value = pww_dv_vres }
		pww_core_value_to_band = yes
		set_variable = { pww_do_band_give = pww_band }
		set_temp_variable = { pww_band_value = pww_dv_vpay }
		pww_core_value_to_band = yes
		set_variable = { pww_do_band_get = pww_band }
	}
	set_variable = { pww_do_vres = pww_dv_vres }
	set_variable = { pww_do_vpay = pww_dv_vpay }
	set_variable = { pww_do_rcv_sells = pww_dv_rcv_sells }
}

# Sender scope. in: pww_dv_w (want), pww_dv_rcv_sells. out: pww_dv_meets.
# Payment-form wants count only when the receiver is the side being paid.
pww_diplo_draft_meets_want = {
	set_temp_variable = { pww_dv_meets = 0 }
	if = {
		limit = {
			OR = {
				AND = {
					check_variable = { pww_dv_w = 0 }
					check_variable = { pww_do_pay = 2 }
					check_variable = { pww_dv_rcv_sells = 1 }
				}
				AND = {
					check_variable = { pww_dv_w = 1 }
					check_variable = { pww_do_pay = 0 }
					check_variable = { pww_dv_rcv_sells = 1 }
				}
				AND = { check_variable = { pww_dv_w = 2 } check_variable = { pww_do_sweet = 1 } }
				AND = {
					check_variable = { pww_dv_w = 3 }
					OR = {
						AND = {
							check_variable = { pww_do_pay = 3 }
							check_variable = { pww_dv_rcv_sells = 1 }
						}
						check_variable = { pww_do_sweet = 3 }
					}
				}
				AND = {
					check_variable = { pww_dv_w = 4 }
					OR = {
						AND = {
							check_variable = { pww_do_pay = 4 }
							check_variable = { pww_dv_rcv_sells = 1 }
						}
						check_variable = { pww_do_sweet = 2 }
					}
				}
				AND = { check_variable = { pww_dv_w = 5 } check_variable = { pww_do_fig = 4 } }
				AND = {
					check_variable = { pww_dv_w = 6 }
					check_variable = { pww_do_fig > 0 }
					check_variable = { pww_do_fig < 4 }
				}
				AND = {
					check_variable = { pww_dv_w = 7 }
					check_variable = { pww_do_act = 0 }
					check_variable = { pww_do_amt < 2 }
				}
				AND = { check_variable = { pww_dv_w = 8 } check_variable = { pww_do_fig > 4 } }
			}
		}
		set_temp_variable = { pww_dv_meets = 1 }
	}
}

# Sender scope. in: pww_dv_r (red line), pww_dv_rcv_sells. out: pww_dv_cross.

# Sender scope. pww_do_sig: a number unique to the draft's terms.

	multiply_temp_variable = { pww_dsg = 4 }
	add_to_temp_variable = { pww_dsg = pww_do_sweet }
	multiply_temp_variable = { pww_dsg = 3 }
	add_to_temp_variable = { pww_dsg = pww_do_price }
	multiply_temp_variable = { pww_dsg = 5 }
	add_to_temp_variable = { pww_dsg = pww_do_pay }
	multiply_temp_variable = { pww_dsg = 5 }
	add_to_temp_variable = { pww_dsg = pww_do_dur }
	multiply_temp_variable = { pww_dsg = 8 }
	if = {
		limit = { check_variable = { pww_do_act = 1 } }
		set_temp_variable = { pww_dsg_s = pww_do_slot }
		modulo_temp_variable = { var = pww_dsg_s value = 8 }
		add_to_temp_variable = { pww_dsg = pww_dsg_s }

	}
	set_variable = { pww_do_sig = pww_dsg }
}

###############################################################################
# Reply to a refused draft. Sender scope (the draft is current).
# Reveals one red line or hidden want, moves patience, may close the talks.
###############################################################################
pww_diplo_talk_reply = {
	set_temp_variable = { pww_dr_t = pww_do_talk }
	# Refused because they cannot spare the resource (SCARCITY): tell the
	# capacity lane a buyer was turned away (reopens or expands output).
	if = {
		limit = {
			has_global_flag = pww_cap_loaded
			check_variable = { pww_do_shres > -1 }
			check_variable = { pww_do_scar > 0.499 }
		}
		set_temp_variable = { pww_in_res = pww_do_shres }
		var:pww_do_target = {
			meta_effect = {
				text = { pww_cap_note_turned = [Y] }
				Y = "yes"
			}
		}
	}
	if = {
		limit = {
			check_variable = { pww_dr_t > -1 }
			check_variable = { pww_dr_t < global.pww_next_talk_id }
			check_variable = { global.pww_dt_status^pww_dr_t = 0 }
		}
		set_temp_variable = { pww_dr_seed = global.pww_dt_seed^pww_dr_t }
		set_temp_variable = { pww_dr_pat = global.pww_dt_patience^pww_dr_t }
		set_temp_variable = { pww_dr_round = global.pww_dt_round^pww_dr_t }
		set_temp_variable = { pww_dr_line = 0 }
		if = {
			limit = {
				check_variable = { global.pww_dt_last_sig^pww_dr_t = pww_do_sig }
				check_variable = { pww_dr_round > 0 }
			}
			# The same terms again: the same answer, nothing new learned.
			set_temp_variable = { pww_dr_line = 83 }
			subtract_from_temp_variable = { pww_dr_pat = 10 }
		}
		else = {
			add_to_temp_variable = { pww_dr_round = 1 }
			set_variable = { global.pww_dt_last_sig^pww_dr_t = pww_do_sig }
			# A draft that already carries their want: the want counts as
			# known from now on (full weight next round).
			if = {
				limit = { check_variable = { pww_do_e_want > 0 } }
				set_variable = { global.pww_dt_want_seen^pww_dr_t = 1 }
			}
			set_temp_variable = { pww_dr_k = pww_dr_seed }
			add_to_temp_variable = { pww_dr_k = pww_dr_round }
			if = {
				limit = { check_variable = { pww_do_crosses = 1 } }
				if = {
					limit = { check_variable = { global.pww_dt_rl_seen^pww_dr_t = 0 } }
					set_variable = { global.pww_dt_rl_seen^pww_dr_t = 1 }
					subtract_from_temp_variable = { pww_dr_pat = 30 }
					modulo_temp_variable = { var = pww_dr_k value = 7 }
					set_temp_variable = { pww_dr_line = 31 }
					add_to_temp_variable = { pww_dr_line = pww_dr_k }
				}
				else = {
					subtract_from_temp_variable = { pww_dr_pat = 45 }
					set_temp_variable = { pww_dr_line = 35 }
				}
			}
			else_if = {
				# Not enough trust for a fixed contract (TRADE BONDS): the
				# specific reason comes before any want is revealed.
				limit = { check_variable = { pww_do_lockfail = 1 } }
				subtract_from_temp_variable = { pww_dr_pat = 5 }
				modulo_temp_variable = { var = pww_dr_k value = 3 }
				set_temp_variable = { pww_dr_line = 111 }
				add_to_temp_variable = { pww_dr_line = pww_dr_k }
			}
			else_if = {
				# Bound by a producers' agreement (CARTELS in
				# pww_diplo_bundle.txt): 121 under the agreed price, 122
				# beyond its quota.
				limit = { check_variable = { pww_do_cartfail > 0 } }
				subtract_from_temp_variable = { pww_dr_pat = 5 }
				set_temp_variable = { pww_dr_line = 120 }
				add_to_temp_variable = { pww_dr_line = pww_do_cartfail }
			}
			else_if = {
				# Reveal the hidden want, unless the draft already carries it
				# (then they acknowledge it below and name the real obstacle).
				limit = {
					check_variable = { global.pww_dt_want_seen^pww_dr_t = 0 }
					check_variable = { pww_do_e_want = 0 }
				}
				set_variable = { global.pww_dt_want_seen^pww_dr_t = 1 }
				subtract_from_temp_variable = { pww_dr_pat = 10 }
				set_temp_variable = { pww_dr_line = 40 }
				add_to_temp_variable = { pww_dr_line = global.pww_dt_want^pww_dr_t }
			}
			else_if = {
				limit = {
					check_variable = { pww_do_e_want > 0 }
					check_variable = { pww_do_e_terms > -1 }
				}
				add_to_temp_variable = { pww_dr_pat = 20 }
				modulo_temp_variable = { var = pww_dr_k value = 5 }
				set_temp_variable = { pww_dr_line = 21 }
				add_to_temp_variable = { pww_dr_line = pww_dr_k }
			}
			else = {
				subtract_from_temp_variable = { pww_dr_pat = 10 }
				if = {
					limit = { check_variable = { pww_do_e_patron < 0 } }
					modulo_temp_variable = { var = pww_dr_k value = 6 }
					set_temp_variable = { pww_dr_line = 71 }
					add_to_temp_variable = { pww_dr_line = pww_dr_k }
				}
				else_if = {
					limit = { check_variable = { pww_do_e_trust < -5 } }
					modulo_temp_variable = { var = pww_dr_k value = 6 }
					set_temp_variable = { pww_dr_line = 51 }
					add_to_temp_variable = { pww_dr_line = pww_dr_k }
				}
				else_if = {
					# The price is the main obstacle: the price-too-low family,
					# and the price floor becomes a known clue.
					limit = {
						check_variable = { pww_do_e_terms < 0 }
						check_variable = { pww_dr_pat > 14 }
					}
					modulo_temp_variable = { var = pww_dr_k value = 6 }
					set_temp_variable = { pww_dr_line = 101 }
					add_to_temp_variable = { pww_dr_line = pww_dr_k }
					set_variable = { global.pww_dt_floor^pww_dr_t = 1 }
				}
				else_if = {
					limit = { check_variable = { pww_dr_pat < 35 } }
					modulo_temp_variable = { var = pww_dr_k value = 7 }
					set_temp_variable = { pww_dr_line = 11 }
					add_to_temp_variable = { pww_dr_line = pww_dr_k }
				}
				else = {
					modulo_temp_variable = { var = pww_dr_k value = 3 }
					set_temp_variable = { pww_dr_line = 84 }
					add_to_temp_variable = { pww_dr_line = pww_dr_k }
					if = {
						limit = { check_variable = { pww_dr_line = 86 } }
						set_temp_variable = { pww_dr_line = 81 }
					}
				}
			}
		}
		clamp_temp_variable = { var = pww_dr_pat min = 0 max = 100 }
		set_variable = { global.pww_dt_patience^pww_dr_t = pww_dr_pat }
		set_variable = { global.pww_dt_round^pww_dr_t = pww_dr_round }
		set_variable = { global.pww_dt_last_ok^pww_dr_t = 0 }
		set_variable = { global.pww_dt_line^pww_dr_t = pww_dr_line }
		if = {
			limit = {
				OR = {
					check_variable = { pww_dr_pat < 1 }
					check_variable = { pww_dr_round > global.pww_diplo_c_max_counters }
				}
			}
			set_temp_variable = { pww_dtc_t = pww_dr_t }
			set_temp_variable = { pww_dtc_status = 2 }
			pww_diplo_talk_close = yes
		}
		if = {
			limit = { has_global_flag = pww_diplo_debug }
			log = "[GetDateText] PWW_DIPLO reply talk=[?pww_dr_t|0] round=[?pww_dr_round|0] patience=[?pww_dr_pat|0] line=[?pww_dr_line|0] sig=[?pww_do_sig|0] total=[?pww_do_e_total|0] terms=[?pww_do_e_terms|0] need=[?pww_do_e_need|0] want=[?pww_do_e_want|0] red=[?pww_do_e_red|0] trust=[?pww_do_e_trust|0] rel=[?pww_do_e_rel|0] patron=[?pww_do_e_patron|0] reserve=[?pww_do_e_reserve|0] fair=[?pww_do_e_fair|0] scar=[?pww_do_scar|2] repeat=[?pww_do_e_repeat|0] invalid=[?pww_do_valid|0] gap=[?pww_do_gap|2] len=[?pww_do_len|0] bind=[?pww_do_bind|0] status=[?global.pww_dt_status^pww_dr_t|0]"
		}
	}
	pww_diplo_refresh_display = yes
	pww_diplo_bundle_mirror = yes
	add_to_variable = { pww_diplo_dirty = 1 }
}

# Any scope. in: pww_dtc_t, pww_dtc_status (2 broken off, 3 expired,

		set_variable = { global.pww_dt_cool_until^pww_dtc_t = pww_dtc_cool }
		set_variable = { global.pww_dt_line^pww_dtc_t = 92 }
	}
	else_if = {
		limit = { check_variable = { pww_dtc_status = 4 } }
		set_variable = { global.pww_dt_line^pww_dtc_t = 93 }
	}
	else = {
		set_variable = { global.pww_dt_line^pww_dtc_t = 95 }
	}
	set_temp_variable = { pww_dlr_id = pww_dtc_t }
	pww_diplo_live_drop_talk = yes
	set_temp_variable = { pww_dq_kind_in = 3 }
	set_temp_variable = { pww_dq_a_in = global.pww_dt_sender^pww_dtc_t }
	set_temp_variable = { pww_dq_b_in = global.pww_dt_target^pww_dtc_t }
	set_temp_variable = { pww_dq_ref_in = pww_dtc_t }
	set_temp_variable = { pww_dq_sit_in = global.pww_dt_sit^pww_dtc_t }
	set_temp_variable = { pww_dq_x_in = pww_dtc_status }
	pww_diplo_queue_push = yes
}

###############################################################################
# Sign the current draft. Sender scope.
# out: pww_dsd_ok (1 signed), pww_dsd_ag (main agreement id or -1)
###############################################################################
pww_diplo_sign_draft = {
	set_temp_variable = { pww_dsd_ok = 0 }
	set_temp_variable = { pww_dsd_ag = -1 }
	pww_diplo_eval_offer = yes
	pww_diplo_ledger_clear = yes
	set_temp_variable = { pww_l_seller = pww_do_seller }
	set_temp_variable = { pww_l_buyer = pww_do_buyer }
	set_temp_variable = { pww_l_kind = pww_do_act }
	set_temp_variable = { pww_l_res = pww_do_res }
	if = {
		limit = { check_variable = { pww_do_act = 0 } }
		set_temp_variable = { pww_l_amount = pww_do_units }
	}
	else = {
		set_temp_variable = { pww_l_state = pww_do_state }
	}
	if = {
		limit = { check_variable = { pww_do_dur > 0 } }
		set_temp_variable = { pww_l_committal = 1 }
	}
	set_temp_variable = { pww_l_term_days = pww_do_term }
	set_temp_variable = { pww_l_civ = pww_do_civ }
	set_temp_variable = { pww_l_pp = pww_do_pp }
	# Payer side for tranches and token payments: the sender's side.
	set_temp_variable = { pww_dsd_side = 0 }
	if = {
		limit = { check_variable = { pww_do_buyer = THIS } }
		set_temp_variable = { pww_dsd_side = 1 }
	}
	if = {
		limit = { check_variable = { pww_do_fig = 1 } }
		if = {
			limit = { check_variable = { pww_dsd_side = 1 } }
			add_to_temp_variable = { pww_l_civ = 1 }
		}
		else = {
			subtract_from_temp_variable = { pww_l_civ = 1 }
		}
	}
	if = {
		limit = {
			check_variable = { pww_do_fig = 4 }
			check_variable = { pww_do_term > 0 }
		}
		set_temp_variable = { pww_l_revert_days = pww_do_term }
		divide_temp_variable = { pww_l_revert_days = 2 }
		round_temp_variable = pww_l_revert_days
	}
	set_temp_variable = { pww_dsd_tranches = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { pww_do_pay > 1 }
				check_variable = { pww_do_sweet > 1 }
			}
		}
		set_temp_variable = { pww_dsd_tranches = 1 }
	}
	if = {
		limit = { check_variable = { pww_dsd_tranches = 0 } }
		set_temp_variable = { pww_l_activate = 1 }
	}
	pww_ledger_create_agreement = yes
	if = {
		limit = { check_variable = { pww_l_result = 1 } }
		set_temp_variable = { pww_dsd_ag = pww_l_new_id }
		if = {
			limit = { check_variable = { pww_dsd_tranches = 1 } }
			pww_diplo_add_draft_tranches = yes
			set_temp_variable = { pww_l_ag = pww_dsd_ag }
			set_temp_variable = { pww_l_expect_rev = 0 }
			pww_ledger_activate = yes
		}
		if = {
			limit = { check_variable = { global.pww_ag_status^pww_dsd_ag = 1 } }
			set_temp_variable = { pww_dsd_ok = 1 }
		}
		else = {
			# Activation refused: cancel the draft so nothing dangles.
			set_temp_variable = { pww_dsd_fail = pww_l_fail }
			set_temp_variable = { pww_l_ag = pww_dsd_ag }
			set_temp_variable = { pww_l_cause = 18 }
			pww_ledger_terminate = yes
			set_temp_variable = { pww_l_fail = pww_dsd_fail }
		}
	}
	# Coal bundle: a second, unpaid coal supply from the sender for the term.
	if = {
		limit = {
			check_variable = { pww_dsd_ok = 1 }
			check_variable = { pww_do_sweet = 1 }
		}
		set_temp_variable = { pww_dg_res = 6 }
		pww_diplo_read_res = yes
		set_temp_variable = { pww_dsd_coal = pww_dg_prod }
		multiply_temp_variable = { pww_dsd_coal = 0.1 }
		round_temp_variable = pww_dsd_coal
		if = {
			limit = { check_variable = { pww_dsd_coal < 1 } }
			set_temp_variable = { pww_dsd_coal = 1 }
		}
		pww_diplo_ledger_clear = yes
		set_temp_variable = { pww_l_seller = THIS }
		set_temp_variable = { pww_l_buyer = pww_do_target }
		set_temp_variable = { pww_l_res = 6 }
		set_temp_variable = { pww_l_amount = pww_dsd_coal }
		if = {
			limit = { check_variable = { pww_do_dur > 0 } }
			set_temp_variable = { pww_l_committal = 1 }
		}
		set_temp_variable = { pww_l_term_days = pww_do_term }
		set_temp_variable = { pww_l_activate = 1 }
		pww_ledger_create_agreement = yes
	}
	if = {
		limit = { has_global_flag = pww_diplo_debug }
		log = "[GetDateText] PWW_DIPLO sign [THIS.GetTag]->[?pww_do_target.GetTag] act=[?pww_do_act|0] res=[?pww_do_res|0] units=[?pww_do_units|1] pay=[?pww_do_pay|0] civ=[?pww_do_civ|0] term=[?pww_do_term|0] ok=[?pww_dsd_ok|0] ag=[?pww_dsd_ag|0] fail=[?pww_l_fail|0]"
	}
}

# Sender scope; pww_dsd_ag is a draft agreement. Adds the payment and
# sweetener tranches (nothing irreversible is paid in full on signing).
pww_diplo_add_draft_tranches = {
	set_temp_variable = { pww_dat_third = pww_do_term }
	divide_temp_variable = { pww_dat_third = 3 }
	round_temp_variable = pww_dat_third
	if = {
		limit = { check_variable = { pww_dat_third < 30 } }
		set_temp_variable = { pww_dat_third = 30 }
	}
	# The buyer pays for the resource in every case.
	if = {
		limit = {
			OR = {

				check_variable = { pww_do_pay = 3 }
			}
		}
		set_temp_variable = { pww_dat_kind = 3 }
		set_temp_variable = { pww_dat_amt = pww_do_cic }
		if = {
			limit = { check_variable = { pww_do_pay = 3 } }
			set_temp_variable = { pww_dat_kind = 0 }
			multiply_temp_variable = { pww_dat_amt = 2 }
		}
		for_loop_effect = {
			start = 0
			end = 3

			set_temp_variable = { pww_l_ag = pww_dsd_ag }
			set_temp_variable = { pww_l_t_payer = 1 }
			set_temp_variable = { pww_l_t_kind = pww_dat_kind }

			set_temp_variable = { pww_l_t_amount = pww_dat_amt }
			set_temp_variable = { pww_l_t_uses = 1 }
			set_temp_variable = { pww_l_t_milestone = pww_dat_m }
			set_temp_variable = { pww_l_t_offset = pww_dat_third }
			if = {
				limit = { check_variable = { pww_dat_m = 0 } }

			}
			set_temp_variable = { pww_l_t_escrow = 0 }

	}

		# Technical mission on signing, the license after service.
		set_temp_variable = { pww_l_ag = pww_dsd_ag }
		set_temp_variable = { pww_l_t_payer = 1 }
		set_temp_variable = { pww_l_t_kind = 2 }
		set_temp_variable = { pww_l_t_item = 0 }
		set_temp_variable = { pww_l_t_amount = 0.1 }
		if = {
			limit = { check_variable = { pww_do_price = 2 } }
			set_temp_variable = { pww_l_t_amount = 0.2 }
		}
		set_temp_variable = { pww_l_t_uses = 1 }
		set_temp_variable = { pww_l_t_milestone = 0 }
		set_temp_variable = { pww_l_t_offset = 0 }
		set_temp_variable = { pww_l_t_escrow = 0 }
		pww_ledger_add_tranche = yes
		set_temp_variable = { pww_l_ag = pww_dsd_ag }
		set_temp_variable = { pww_l_t_payer = 1 }
		set_temp_variable = { pww_l_t_kind = 1 }
		set_temp_variable = { pww_l_t_item = 2 }
		set_temp_variable = { pww_l_t_amount = 0 }

		set_temp_variable = { pww_l_t_escrow = 0 }
		pww_ledger_add_tranche = yes
	}
	# Sweeteners paid by the sender's side after a month of service.
	if = {
		limit = { check_variable = { pww_do_sweet = 2 } }
		set_temp_variable = { pww_l_ag = pww_dsd_ag }
		set_temp_variable = { pww_l_t_payer = pww_dsd_side }
		set_temp_variable = { pww_l_t_kind = 2 }
		set_temp_variable = { pww_l_t_item = 0 }

		set_temp_variable = { pww_l_t_escrow = 0 }
		pww_ledger_add_tranche = yes
	}
	else_if = {
		limit = { check_variable = { pww_do_sweet = 3 } }
		set_temp_variable = { pww_l_ag = pww_dsd_ag }
		set_temp_variable = { pww_l_t_payer = pww_dsd_side }
		set_temp_variable = { pww_l_t_kind = 0 }

		}
	}
}

###############################################################################
# AI preparation (sender scope, AI). in: pww_dk_target, pww_dk_act,
# pww_dap_res. Engine fact (v1.20, verified in tests/diplomacy): an AI sends
# a scripted diplomatic action only when the target's ai_acceptance, read on
# the draft as it stands BEFORE reset_send_effect, is positive. So the AI
# prepares its draft on its tick and plays the legation rounds in script:
# each refused draft goes through pww_diplo_talk_reply exactly like a sent
# one (one red line or hidden want revealed, patience spent, three counters
# at most), then the AI redrafts with what it learned. A draft the target
# would sign stays on the sender and the engine sends it through the real
# action. out: pww_dap_ready (1 = a signable draft is in place)
###############################################################################
pww_diplo_ai_prepare_offer = {
	set_temp_variable = { pww_dap_ready = 0 }
	pww_diplo_offer_begin = yes
	if = {
		limit = { check_variable = { pww_do_talk > -1 } }
		set_variable = { pww_do_res = pww_dap_res }
		pww_diplo_ai_build_bundle = yes
		set_temp_variable = { pww_dap_brk = 0 }
		for_loop_effect = {
			start = 0
			end = 5
			value = pww_dap_i
			break = pww_dap_brk
			set_temp_variable = { pww_dap_t = pww_do_talk }
			if = {
				limit = { check_variable = { pww_do_ok = 1 } }
				set_temp_variable = { pww_dap_ready = 1 }
				set_temp_variable = { pww_dap_brk = 1 }
			}
			else_if = {
				limit = { check_variable = { global.pww_dt_status^pww_dap_t = 0 } }
				pww_diplo_talk_reply = yes
				if = {
					limit = { check_variable = { global.pww_dt_status^pww_dap_t = 0 } }
					pww_diplo_ai_build_bundle = yes
					# Nothing new to table: stop rather than burn their patience.
					if = {
						limit = {
							check_variable = { global.pww_dt_last_sig^pww_dap_t = pww_do_sig }
							check_variable = { pww_do_ok = 0 }
						}
						set_temp_variable = { pww_dap_brk = 1 }
					}
				}
			}
			else = {
				set_temp_variable = { pww_dap_brk = 1 }
			}
		}
	}
	pww_diplo_refresh_display = yes
	pww_diplo_bundle_mirror = yes
}

# Receiver mirror (sender scope, at delivery): the receive window reads ROOT =
# the sender, from a snapshot in its own namespace (pww_diplo_bundle_snapshot_recv:
# pww_drx_*), never the live mirror a send window reads. Nothing is written on
# the target's display variables, so an incoming proposal cannot repaint a
# send window the receiver has open. pww_dr_counter (the receiver's counter
# mark) starts clear. The counter itself: pww_diplo_ai_consider_counter
# (pww_diplo_bundle.txt).
###############################################################################
pww_diplo_mirror_for_receiver = {
	pww_diplo_eval_bundle = yes
	pww_diplo_refresh_display = yes
	pww_diplo_bundle_mirror = yes
	set_variable = { pww_dr_act = pww_do_act }

	var:pww_do_target = {
		set_variable = { pww_dr_from = pww_dmr_s }
		set_variable = { pww_dr_act = PREV.pww_dr_act }
		set_variable = { pww_dr_counter = 0 }
		add_to_variable = { pww_diplo_dirty = 1 }
	}
	pww_diplo_bundle_snapshot_recv = yes
}
