###############################################################################
# Power Without War: producer cartels, core (storage, conference, membership,
# collapse, the International Rubber Regulation Agreement anchor, ticks)
#
# Related files
#   pww_cartel_quota.txt       member tick: ledger sales scan, quota, storage
#                              request / fallback withholding, cheating
#                              (AI side sales), detection, cohesion drift
#   pww_cartel_outsiders.txt   invitations (AI acceptance), poaching (tempt),
#                              importer pressure, flooding, joining, AI
#                              outsiders, human decision entry points
#   scripted_triggers/pww_cartel_triggers.txt
#   on_actions/pww_cartel_on_actions.txt
#   dynamic_modifiers/pww_cartel_dynamic_modifiers.txt (pww_cartel_flows)
#   opinion_modifiers/pww_cartel_opinion_modifiers.txt
#   events/pww_cartel_events.txt (namespace pww_cartel, human countries only)
#   decisions/pww_cartel_decisions.txt (+ categories), generated by
#     tools/pww_cartel_gen.py together with the decision loc
#   localisation/english/pww_cartel_l_english.yml
#
# A cartel is its own record (not a pww_sit_* situation: the situations
# framework is a three-seat contest over named states; a cartel is many
# members holding a price). Global flag pww_cartel_loaded.
#
# ===========================================================================
# RESOURCES: 1 aluminium, 2 rubber, 3 tungsten, 4 steel, 5 chromium, 6 coal.
# Oil (0) is excluded, as in the stockpiles lane (vanilla turns spare oil
# into fuel). At most one live cartel per resource.
#
# ENUMS
#   cartel status  0 conference (invitations out, decision on conf_day)
#                  1 active      2 collapsed (price war)   3 conference failed
#   end cause      0 none, 1 cohesion broke, 2 too few members, 3 members at
#                  war with each other, 4 conference failed
#   member status  0 invited, 1 member, 2 refused, 3 expelled, 4 left
#                  (resigned or courted away), 5 cartel ended
#   leave cause    0 resigned, 1 courted away by an outsider, 2 at war with
#                  the cartel, 3 no longer produces the resource
#
# ===========================================================================
# STORAGE (parallel global arrays, appended in pww_cartel_alloc, never
# reordered; id == index). Read global.pww_cart_<field>^<id var>.
#   res, status, leader (country), anchor (0 generic, 1 the 1934 rubber
#   agreement), price (cartel price, factor of the open market rate of
#   8 units per civilian factory: 1.10 = 10% above market), cohesion
#   (0..100), quota_pct (share of a member's unsold output it may still
#   sell through mod deals), open_day, conf_day, start_day, end_day,
#   pw_until (price war ends), cause (end cause), n (members), caught
#   (num_days of the last detection), rev (+1 on change)
# Members (flat parallel arrays global.pww_cm_<field>^<entry id>):
#   cart (cartel id), who (country), status, join_day, quota (units/day the
#   member may sell through mod deals; -1 until the cartel is active), base
#   (its mod-deal sales when it joined, grandfathered), sold (current mod-
#   deal sales of the resource), over (sales above quota), under (units in
#   deals priced below the cartel price, signed after joining), cheat (over
#   + under, capped at sold), susp (the cartel's suspicion of this member,
#   0..100), warns (0..2), store (units/day the quota asks it to keep off
#   the market), hold (units/day withheld by the fallback), held (unit-days
#   kept off the market, fallback pool; decays), tempt (outside temptation,
#   decays), unsold (last reading), ucap (unsold when it joined: the cap on
#   what the quota keeps off the market), last_tick, rev
# Per resource (global arrays indexed 0..6): global.pww_cart_by_res (live
#   cartel id or -1; "live" = conference or active), global.pww_cart_cd_res
#   (num_days before which no new cartel forms), global.pww_cart_glut_n /
#   global.pww_cart_ripe (formation scan: glut producers, 1 = ripe).
#   global.pww_cart_coh_res / _upf_res / _qpct_res (cohesion, agreed price
#   as units per factory, quota % of the newest cartel for that resource:
#   pww_cartel_sync_coh, for decision tooltips).
# Live list: global.pww_cart_live (cartel ids in status 0 or 1).
# Country: array pww_cart_my (member entry ids where status is 0 or 1),
#   flags pww_cartel_mem_<res> (status 1 in a live cartel), pww_cartel_inv_
#   <res> (invited, undecided), pww_cartel_glut_<res> (last scan: glut
#   producer), pww_cartel_pricewar_<res> (timed, ex-members after a
#   collapse), pww_cartel_flooding_<res> (timed, outsider flooding);
#   vars pww_cartel_standing (0..100, start 60; below 30 nobody invites you
#   and you cannot join), pww_cartel_hold_<res> / pww_cartel_dump_<res>
#   (dynamic modifier inputs), pww_cartel_store_<res> (storage request for
#   the stockpiles lane), pww_cartel_held_<res> (fallback pool).
# Dirty variable for UI: global.pww_cart_rev.
#
# UI CONTRACT (one row per cartel the viewer is in or watching): read
#   global.pww_cart_{res,status,leader,price,cohesion,n} by id; members by
#   looping global.pww_cm_cart (== id) with status 1: who, quota, sold,
#   store, susp, warns. Viewer's own rows: country array pww_cart_my.
#   Loc: pww_cartel_res_<res>, pww_cartel_status_<status>,
#   pww_cartel_mstatus_<status>, pww_cartel_cohesion_band_<0..4> and
#   pww_cartel_susp_band_<0..4> (temp in pww_band_value 0..100 through
#   pww_cartel_band, out pww_band).
#
# ===========================================================================
# PUBLIC EFFECTS (inputs in temp variables; out pww_out_result 1/0 and
# pww_out_fail)
#   pww_cartel_init = yes
#   pww_cartel_open_conference = yes   in: pww_in_res (1..6), pww_in_leader
#       out: pww_out_cartel_id (-1). fail 1 bad res, 2 a live cartel
#       exists, 3 cooldown, 4 leader does not produce it, 5 leader standing
#   pww_cartel_join = yes        in: pww_in_cartel_id, pww_in_country
#       fail 1 bad cartel, 2 does not produce, 3 already in, 4 standing
#   pww_cartel_leave = yes       in: pww_in_cm (member entry), pww_in_cause
#   pww_cartel_expel = yes       in: pww_in_cm
#   pww_cartel_collapse = yes    in: pww_in_cartel_id, pww_in_cause
#   pww_cartel_find_member = yes (country) in: pww_in_res; out: pww_out_cm
#       (this country's entry in the live cartel for res, status 0/1, or -1)
#   Outsiders (pww_cartel_outsiders.txt): pww_cartel_tempt, pww_cartel_press,
#   pww_cartel_flood.
#
# CALLS OUT
#   ledger (hard dependency, always loaded): pww_ledger_create_agreement
#     (AI side sales, poaching deals), pww_ledger_withdraw (ending side
#     deals after a warning). Agreements are read from pww_ag_ids.
#   stockpiles (guarded by has_global_flag = pww_stock_loaded): the cartel
#     writes country var pww_cartel_store_<res> (units/day the quota asks
#     storage to take) and reads pww_stock_in_<res> (what storage took).
#     The remainder is withheld by the fallback. Without the stockpiles lane
#     the whole request is withheld: the fallback puts it on
#     country_resource_cost_<res> (dynamic modifier pww_cartel_flows; the
#     planters tap less) and keeps a virtual pool pww_cartel_held_<res>
#     that decays 2% a month. On a collapse each ex-member dumps its pool
#     over global.pww_cartel_c_dump_days through country_resource_<res>.
#
# TUNABLES (set by pww_cartel_init only when unset)
#   pww_cartel_c_quota_pct 30   pww_cartel_c_price 1.10   (anchor 25 / 1.15)
#   pww_cartel_c_undercut 0.95  (a deal below price x this is undercutting)
#   pww_cartel_c_cohesion 65    (anchor 75)   pww_cartel_c_collapse 20
#   pww_cartel_c_conf_days 30   pww_cartel_c_cooldown 365
#   pww_cartel_c_pricewar_days 180   pww_cartel_c_dump_days 90
#   pww_cartel_c_glut_share 0.2 pww_cartel_c_glut_min 10
#   pww_cartel_c_max_live 2 (AI conferences only while fewer are live)
#   pww_cartel_c_glut_producers 3    pww_cartel_c_desp_buyers 1 (at most
#     this many buyers at desperation 50+ for a conference to be ripe)
#   pww_cartel_c_form_chance 10 (% a month the biggest AI glut producer
#     calls a conference)  pww_cartel_c_held_decay 0.02 (per month)
#   pww_cartel_c_ai_form 1, pww_cartel_c_ai_cheat 1, pww_cartel_c_ai_poach 1
#     (0 disables), pww_cartel_c_irra 1 (0: no 1934 anchor)
#   pww_cartel_c_detect_force 0 (tests: 1 = a cheat is always caught)
#
# TEST HOOKS (country vars): pww_cartel_test_res + pww_cartel_test_unsold
#   override the unsold reading for that resource.
# DEBUG: global flag pww_cartel_debug logs PWW_CARTEL lines; opens,
#   activations, detections, expulsions and collapses always log.
#
# TEMP VARIABLES: pww_in_* / pww_out_* public; internals pww_k<xx>_*.
###############################################################################

pww_cartel_init = {
	if = {
		limit = { NOT = { has_global_flag = pww_cartel_initialized } }
		set_global_flag = pww_cartel_initialized
		set_global_flag = pww_cartel_loaded
		set_variable = { global.pww_cart_version = 1 }
		set_variable = { global.pww_next_cart_id = 0 }
		set_variable = { global.pww_next_cm_id = 0 }
		set_variable = { global.pww_cart_rev = 1 }
		set_variable = { global.pww_cart_day = -1 }
		set_variable = { global.pww_cart_last_monthly = -100 }
		for_loop_effect = {
			start = 0
			end = 7
			add_to_array = { array = global.pww_cart_by_res value = -1 }
			add_to_array = { array = global.pww_cart_cd_res value = 0 }
			add_to_array = { array = global.pww_cart_glut_n value = 0 }
			add_to_array = { array = global.pww_cart_ripe value = 0 }
		}
		log = "[GetDateText] PWW_CARTEL init v1"
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_quota_pct } }
		set_variable = { global.pww_cartel_c_quota_pct = 30 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_price } }
		set_variable = { global.pww_cartel_c_price = 1.10 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_undercut } }
		set_variable = { global.pww_cartel_c_undercut = 0.95 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_cohesion } }
		set_variable = { global.pww_cartel_c_cohesion = 65 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_collapse } }
		set_variable = { global.pww_cartel_c_collapse = 20 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_conf_days } }
		set_variable = { global.pww_cartel_c_conf_days = 30 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_cooldown } }
		set_variable = { global.pww_cartel_c_cooldown = 365 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_pricewar_days } }
		set_variable = { global.pww_cartel_c_pricewar_days = 180 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_dump_days } }
		set_variable = { global.pww_cartel_c_dump_days = 90 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_glut_share } }
		set_variable = { global.pww_cartel_c_glut_share = 0.2 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_glut_min } }
		set_variable = { global.pww_cartel_c_glut_min = 10 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_glut_producers } }
		set_variable = { global.pww_cartel_c_glut_producers = 3 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_desp_buyers } }
		set_variable = { global.pww_cartel_c_desp_buyers = 1 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_form_chance } }
		set_variable = { global.pww_cartel_c_form_chance = 10 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_max_live } }
		set_variable = { global.pww_cartel_c_max_live = 2 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_held_decay } }
		set_variable = { global.pww_cartel_c_held_decay = 0.02 }
	}
	if = {
		limit = { NOT = { has_variable = global.pww_cartel_c_ai_form } }
		set_variable = { global.pww_cartel_c_ai_form = 1 }

	}

# Country scope (on_daily, every country).
pww_cartel_on_daily = {
	if = {
		limit = { has_global_flag = pww_initialized }
		if = {
			limit = { NOT = { has_global_flag = pww_cartel_initialized } }
			pww_cartel_init = yes
		}
		if = {
			limit = { NOT = { check_variable = { global.pww_cart_day = global.num_days } } }
			set_variable = { global.pww_cart_day = global.num_days }
			pww_cartel_daily_global = yes
		}
		if = {
			limit = {
				check_variable = { pww_cart_my^num > 0 }
				check_variable = { pww_bucket = global.pww_day_mod30 }
			}
			pww_cartel_member_tick_country = yes
		}
		if = {
			limit = {
				has_variable = pww_cartel_dump_until
				NOT = { check_variable = { pww_cartel_dump_until > global.num_days } }
			}
			pww_cartel_end_dump = yes
		}
	}
}

# Country scope (on_monthly, every country). The first caller at least 28
# days after the last run does the monthly global work.
pww_cartel_on_monthly = {
	if = {
		limit = { has_global_flag = pww_cartel_initialized }
		set_temp_variable = { pww_kmo_since = global.num_days }
		subtract_from_temp_variable = { pww_kmo_since = global.pww_cart_last_monthly }
		if = {
			limit = { check_variable = { pww_kmo_since > 27 } }
			set_variable = { global.pww_cart_last_monthly = global.num_days }
			pww_cartel_monthly = yes
		}
	}
}

# Any scope, once a day. Anchor, conference deadlines, price war ends.
pww_cartel_daily_global = {
	if = {
		limit = {
			NOT = { has_global_flag = pww_cartel_irra_done }
			check_variable = { global.pww_cartel_c_irra = 1 }
		}
		set_global_flag = pww_cartel_irra_done
		pww_cartel_irra_anchor = yes
	}
	for_each_loop = {
		array = global.pww_cart_res
		value = pww_kdg_res
		index = pww_kdg_id
		if = {
			limit = {
				check_variable = { global.pww_cart_status^pww_kdg_id = 0 }
				NOT = { check_variable = { global.pww_cart_conf_day^pww_kdg_id > global.num_days } }
			}
			set_temp_variable = { pww_kc_cid = pww_kdg_id }
			pww_cartel_conference_resolve = yes
		}
		else_if = {
			limit = {
				check_variable = { global.pww_cart_status^pww_kdg_id = 2 }
				check_variable = { global.pww_cart_pw_until^pww_kdg_id > 0 }
				NOT = { check_variable = { global.pww_cart_pw_until^pww_kdg_id > global.num_days } }
			}
			set_variable = { global.pww_cart_pw_until^pww_kdg_id = 0 }
			add_to_variable = { global.pww_cart_rev = 1 }
			log = "[GetDateText] PWW_CARTEL price war over: cartel [?pww_kdg_id] res [?pww_kdg_res]"
		}
	}
}

# Any scope, monthly. Cohesion, AI outsiders, formation scan.
pww_cartel_monthly = {
	for_each_loop = {
		array = global.pww_cart_res
		value = pww_kmn_res
		index = pww_kmn_id
		if = {
			limit = { check_variable = { global.pww_cart_status^pww_kmn_id = 1 } }
			set_temp_variable = { pww_kco_cid = pww_kmn_id }
			pww_cartel_cohesion_tick = yes
			if = {
				limit = {
					check_variable = { global.pww_cart_status^pww_kmn_id = 1 }
					check_variable = { global.pww_cartel_c_ai_poach = 1 }
				}
				set_temp_variable = { pww_kao_cid = pww_kmn_id }
				pww_cartel_ai_outsiders = yes
			}
		}
	}
	pww_cartel_formation_scan = yes
}

# in: pww_ka_res, pww_ka_leader, pww_ka_anchor, pww_ka_status
# out: pww_ka_id
pww_cartel_alloc = {
	set_temp_variable = { pww_ka_id = global.pww_next_cart_id }
	add_to_array = { array = global.pww_cart_res value = pww_ka_res }
	add_to_array = { array = global.pww_cart_status value = pww_ka_status }
	add_to_array = { array = global.pww_cart_leader value = pww_ka_leader }
	add_to_array = { array = global.pww_cart_anchor value = pww_ka_anchor }
	set_temp_variable = { pww_ka_v = global.pww_cartel_c_price }
	set_temp_variable = { pww_ka_c = global.pww_cartel_c_cohesion }
	set_temp_variable = { pww_ka_q = global.pww_cartel_c_quota_pct }
	if = {
		limit = { check_variable = { pww_ka_anchor = 1 } }
		set_temp_variable = { pww_ka_v = 1.15 }
		set_temp_variable = { pww_ka_c = 75 }
		set_temp_variable = { pww_ka_q = 25 }
	}
	add_to_array = { array = global.pww_cart_price value = pww_ka_v }
	add_to_array = { array = global.pww_cart_cohesion value = pww_ka_c }
	add_to_array = { array = global.pww_cart_quota_pct value = pww_ka_q }
	add_to_array = { array = global.pww_cart_open_day value = global.num_days }
	set_temp_variable = { pww_ka_cd = global.num_days }
	add_to_temp_variable = { pww_ka_cd = global.pww_cartel_c_conf_days }
	add_to_array = { array = global.pww_cart_conf_day value = pww_ka_cd }
	add_to_array = { array = global.pww_cart_start_day value = 0 }
	add_to_array = { array = global.pww_cart_end_day value = 0 }
	add_to_array = { array = global.pww_cart_pw_until value = 0 }
	add_to_array = { array = global.pww_cart_cause value = 0 }
	add_to_array = { array = global.pww_cart_n value = 0 }
	add_to_array = { array = global.pww_cart_caught value = 0 }
	add_to_array = { array = global.pww_cart_rev value = 1 }
	add_to_variable = { global.pww_next_cart_id = 1 }
	add_to_array = { array = global.pww_cart_live value = pww_ka_id }
	set_variable = { global.pww_cart_by_res^pww_ka_res = pww_ka_id }
	set_temp_variable = { pww_kcy_cid = pww_ka_id }
	pww_cartel_sync_coh = yes
	add_to_variable = { global.pww_cart_rev = 1 }
}

# Country scope. Seeds the standing.
pww_cartel_ensure_country = {
	if = {
		limit = { NOT = { has_variable = pww_cartel_standing } }
		set_variable = { pww_cartel_standing = 60 }
	}
}

# Country scope. in: pww_kf_res, pww_kf_flag (0 member flag, 1 invited
# flag), pww_kf_on (1 set, 0 clear).
pww_cartel_set_flag = {
	if = {
		limit = { check_variable = { pww_kf_flag = 0 } }
		if = {
			limit = { check_variable = { pww_kf_on = 1 } }
			if = { limit = { check_variable = { pww_kf_res = 1 } } set_country_flag = pww_cartel_mem_aluminium }
			else_if = { limit = { check_variable = { pww_kf_res = 2 } } set_country_flag = pww_cartel_mem_rubber }

# in: pww_kcx_cid. Recounts members (status 1) into global.pww_cart_n.
pww_cartel_recount = {

	for_each_loop = {
		array = global.pww_cm_cart
		value = pww_kcx_c
		index = pww_kcx_i

			limit = {
				check_variable = { pww_kcx_c = pww_kcx_cid }

# in: pww_kcy_cid. Mirrors a cartel into per-resource global arrays that
# decision tooltips read by resource (loc cannot look up the cartel id):
# global.pww_cart_coh_res (cohesion), _upf_res (agreed price as units per
# civilian factory, rounded down to 0.1), _qpct_res (quota share of unsold
# output, %). The newest cartel for the resource writes them.
pww_cartel_sync_coh = {
	if = {
		limit = { NOT = { check_variable = { global.pww_cart_coh_res^num > 6 } } }
		clear_array = global.pww_cart_coh_res
		resize_array = { array = global.pww_cart_coh_res value = 0 size = 7 }
		clear_array = global.pww_cart_upf_res
		resize_array = { array = global.pww_cart_upf_res value = 0 size = 7 }
		clear_array = global.pww_cart_qpct_res
		resize_array = { array = global.pww_cart_qpct_res value = 0 size = 7 }
	}
	set_temp_variable = { pww_kcy_res = global.pww_cart_res^pww_kcy_cid }
	set_variable = { global.pww_cart_coh_res^pww_kcy_res = global.pww_cart_cohesion^pww_kcy_cid }
	set_variable = { global.pww_cart_qpct_res^pww_kcy_res = global.pww_cart_quota_pct^pww_kcy_cid }
	set_temp_variable = { pww_kcy_u = 80 }
	divide_temp_variable = { pww_kcy_u = global.pww_cart_price^pww_kcy_cid }
	subtract_from_temp_variable = { pww_kcy_u = 0.499 }
	round_temp_variable = pww_kcy_u
	divide_temp_variable = { pww_kcy_u = 10 }
	set_variable = { global.pww_cart_upf_res^pww_kcy_res = pww_kcy_u }
}

# in: pww_in_res (1..6), pww_in_leader. out: pww_out_cartel_id, pww_out_result,
# pww_out_fail. Invites every glut producer from the last formation scan
# (flag pww_cartel_glut_<res>) with standing 30+, not at war with the leader.
pww_cartel_open_conference = {
	set_temp_variable = { pww_out_cartel_id = -1 }
	set_temp_variable = { pww_out_result = 0 }
	set_temp_variable = { pww_out_fail = 0 }
	set_temp_variable = { pww_koc_res = pww_in_res }
	set_temp_variable = { pww_koc_l = pww_in_leader }
	if = {
		limit = {
			OR = {
				check_variable = { pww_koc_res < 1 }
				check_variable = { pww_koc_res > 6 }
			}
		}
		set_temp_variable = { pww_out_fail = 1 }
	}
	else_if = {
		limit = { check_variable = { global.pww_cart_by_res^pww_koc_res > -1 } }
		set_temp_variable = { pww_out_fail = 2 }
	}
	else_if = {
		limit = { check_variable = { global.pww_cart_cd_res^pww_koc_res > global.num_days } }
		set_temp_variable = { pww_out_fail = 3 }
	}
	if = {
		limit = { check_variable = { pww_out_fail = 0 } }
		var:pww_koc_l = {
			pww_cartel_ensure_country = yes
			set_temp_variable = { pww_kr_res = pww_koc_res }
			pww_cartel_read_res = yes
			if = {
				limit = { check_variable = { pww_kr_produces = 0 } }
				set_temp_variable = { pww_out_fail = 4 }
			}
			else_if = {
				limit = { check_variable = { pww_cartel_standing < 30 } }
				set_temp_variable = { pww_out_fail = 5 }
			}
		}
	}
	if = {
		limit = { check_variable = { pww_out_fail = 0 } }
		set_temp_variable = { pww_ka_res = pww_koc_res }
		set_temp_variable = { pww_ka_leader = pww_koc_l }
		set_temp_variable = { pww_ka_anchor = 0 }
		set_temp_variable = { pww_ka_status = 0 }
		pww_cartel_alloc = yes
		set_temp_variable = { pww_koc_cid = pww_ka_id }
		set_temp_variable = { pww_km_cid = pww_koc_cid }
		set_temp_variable = { pww_km_who = pww_koc_l }
		set_temp_variable = { pww_km_status = 1 }
		pww_cartel_alloc_member = yes
		set_temp_variable = { pww_out_cartel_id = pww_koc_cid }
		set_temp_variable = { pww_out_result = 1 }
		log = "[GetDateText] PWW_CARTEL conference called: cartel [?pww_koc_cid] res [?pww_koc_res] by [?pww_koc_l.GetTag]"
		# Invitations
		every_country = {
			limit = {
				exists = yes
				NOT = { check_variable = { pww_koc_l = THIS } }
				OR = {
					AND = { check_variable = { pww_koc_res = 1 } has_country_flag = pww_cartel_glut_aluminium }
					AND = { check_variable = { pww_koc_res = 2 } has_country_flag = pww_cartel_glut_rubber }
					AND = { check_variable = { pww_koc_res = 3 } has_country_flag = pww_cartel_glut_tungsten }
					AND = { check_variable = { pww_koc_res = 4 } has_country_flag = pww_cartel_glut_steel }
					AND = { check_variable = { pww_koc_res = 5 } has_country_flag = pww_cartel_glut_chromium }
					AND = { check_variable = { pww_koc_res = 6 } has_country_flag = pww_cartel_glut_coal }
				}
				NOT = { var:pww_koc_l = { has_war_with = PREV } }
			}
			pww_cartel_ensure_country = yes
			if = {
				limit = { check_variable = { pww_cartel_standing > 29.999 } }
				set_temp_variable = { pww_kiv_me = THIS }
				pww_cartel_invite_this = yes
			}
		}
		set_temp_variable = { pww_kcx_cid = pww_koc_cid }
		pww_cartel_recount = yes
		# A human leader hears when the conference sits (pww_cartel.2).
	}

# Country scope (the invitee). in: pww_koc_cid, pww_koc_res, pww_kiv_me.
pww_cartel_invite_this = {

	set_temp_variable = { pww_km_status = 0 }
	pww_cartel_alloc_member = yes
	set_temp_variable = { pww_kiv_cm = pww_km_id }
	if = {
		limit = { is_ai = yes }
		set_temp_variable = { pww_kvs_cid = pww_koc_cid }
		pww_cartel_ai_invite_score = yes
		if = {
			limit = { check_variable = { pww_kvs_score > 49.999 } }
			set_variable = { global.pww_cm_status^pww_kiv_cm = 1 }
		}
		else = {
			set_variable = { global.pww_cm_status^pww_kiv_cm = 2 }
			remove_from_array = { array = pww_cart_my value = pww_kiv_cm }
		}
		if = {
			limit = { has_global_flag = pww_cartel_debug }
			log = "[GetDateText] PWW_CARTEL invite [?THIS.GetTag] cartel [?pww_koc_cid] score [?pww_kvs_score|1] -> status [?global.pww_cm_status^pww_kiv_cm]"

	}
	else = {
		set_temp_variable = { pww_kf_res = pww_koc_res }
		set_temp_variable = { pww_kf_flag = 1 }
		set_temp_variable = { pww_kf_on = 1 }

}

# in: pww_kc_cid. The conference sits: two or more members -> active.
pww_cartel_conference_resolve = {
	set_temp_variable = { pww_kcx_cid = pww_kc_cid }
	pww_cartel_recount = yes
	# Undecided human invitees are treated as refusals
	for_each_loop = {
		array = global.pww_cm_cart
		value = pww_kcr_c
		index = pww_kcr_i
		if = {
			limit = {
				check_variable = { pww_kcr_c = pww_kc_cid }
				check_variable = { global.pww_cm_status^pww_kcr_i = 0 }
			}
			set_temp_variable = { pww_in_cm = pww_kcr_i }
			set_temp_variable = { pww_in_accept = 0 }
			set_temp_variable = { pww_kcr_who = global.pww_cm_who^pww_kcr_i }
			var:pww_kcr_who = { pww_cartel_answer_invite = yes }
		}
	}
	if = {
		limit = { check_variable = { global.pww_cart_n^pww_kc_cid > 1 } }
		set_temp_variable = { pww_kac_cid = pww_kc_cid }
		pww_cartel_activate = yes
	}
	else = {
		set_variable = { global.pww_cart_status^pww_kc_cid = 3 }
		set_variable = { global.pww_cart_cause^pww_kc_cid = 4 }
		set_variable = { global.pww_cart_end_day^pww_kc_cid = global.num_days }
		set_temp_variable = { pww_kcr_res = global.pww_cart_res^pww_kc_cid }
		set_variable = { global.pww_cart_by_res^pww_kcr_res = -1 }
		set_temp_variable = { pww_kcr_cd = global.num_days }
		add_to_temp_variable = { pww_kcr_cd = 180 }
		set_variable = { global.pww_cart_cd_res^pww_kcr_res = pww_kcr_cd }
		remove_from_array = { array = global.pww_cart_live value = pww_kc_cid }
		pww_cartel_release_members = yes
		set_temp_variable = { pww_kcr_l = global.pww_cart_leader^pww_kc_cid }
		var:pww_kcr_l = {
			if = {
				limit = { is_ai = no }
				set_variable = { pww_cartel_ev_cid = pww_kc_cid }
				country_event = { id = pww_cartel.3 }
			}
		}
		add_to_variable = { global.pww_cart_rev = 1 }
		log = "[GetDateText] PWW_CARTEL conference failed: cartel [?pww_kc_cid]"
	}
}

# in: pww_kam_cm. Member status 1 in an active cartel: quota, flag.
# Quota = its mod-deal sales now + quota_pct% of its unsold output.
pww_cartel_activate_member = {
	set_temp_variable = { pww_kam_cid = global.pww_cm_cart^pww_kam_cm }
	set_temp_variable = { pww_kam_res = global.pww_cart_res^pww_kam_cid }
	set_temp_variable = { pww_kam_who = global.pww_cm_who^pww_kam_cm }
	var:pww_kam_who = {
		set_temp_variable = { pww_ks_cm = pww_kam_cm }
		set_temp_variable = { pww_ks_res = pww_kam_res }
		set_temp_variable = { pww_ks_end = 0 }
		set_temp_variable = { pww_ks_join = 99999 }
		pww_cartel_scan_sales = yes
		set_temp_variable = { pww_kr_res = pww_kam_res }
		pww_cartel_read_res = yes
		set_temp_variable = { pww_kam_q = pww_kr_unsold }
		multiply_temp_variable = { pww_kam_q = global.pww_cart_quota_pct^pww_kam_cid }
		divide_temp_variable = { pww_kam_q = 100 }
		add_to_temp_variable = { pww_kam_q = pww_ks_sold }
		round_temp_variable = pww_kam_q
		if = {
			limit = { check_variable = { pww_kam_q < pww_ks_sold } }
			set_temp_variable = { pww_kam_q = pww_ks_sold }
		}
		set_variable = { global.pww_cm_quota^pww_kam_cm = pww_kam_q }
		set_variable = { global.pww_cm_base^pww_kam_cm = pww_ks_sold }
		set_variable = { global.pww_cm_sold^pww_kam_cm = pww_ks_sold }
		set_variable = { global.pww_cm_join_day^pww_kam_cm = global.num_days }
		set_variable = { global.pww_cm_unsold^pww_kam_cm = pww_kr_unsold }
		set_variable = { global.pww_cm_ucap^pww_kam_cm = pww_kr_unsold }
		set_variable = { global.pww_cm_store^pww_kam_cm = 0 }
		set_variable = { global.pww_cm_last_tick^pww_kam_cm = 0 }
		set_temp_variable = { pww_kf_res = pww_kam_res }
		set_temp_variable = { pww_kf_flag = 0 }
		set_temp_variable = { pww_kf_on = 1 }
		pww_cartel_set_flag = yes
		# First quota pass right away (storage request / fallback hold)
		set_temp_variable = { pww_kt_cm2 = pww_kam_cm }
		pww_cartel_apply_quota = yes
		log = "[GetDateText] PWW_CARTEL member [?THIS.GetTag] cartel [?pww_kam_cid] quota [?pww_kam_q|1] (deal sales [?pww_ks_sold|1], unsold [?pww_kr_unsold|1], store [?global.pww_cm_store^pww_kam_cm|1], hold [?global.pww_cm_hold^pww_kam_cm|1])"
	}
	add_to_variable = { global.pww_cm_rev^pww_kam_cm = 1 }
}

# in: pww_kpl_cid. The member with the largest output leads.
pww_cartel_pick_leader = {
	set_temp_variable = { pww_kpl_res = global.pww_cart_res^pww_kpl_cid }
	set_temp_variable = { pww_kpl_best = -1 }
	set_temp_variable = { pww_kpl_who = global.pww_cart_leader^pww_kpl_cid }
	# The incumbent keeps the chair while still a member (anchor: Britain)
	set_temp_variable = { pww_kpl_keep = 0 }
	for_each_loop = {
		array = global.pww_cm_cart
		value = pww_kpl_c
		index = pww_kpl_i
		if = {
			limit = {
				check_variable = { pww_kpl_c = pww_kpl_cid }
				check_variable = { global.pww_cm_status^pww_kpl_i = 1 }
				check_variable = { global.pww_cm_who^pww_kpl_i = pww_kpl_who }
			}
			set_temp_variable = { pww_kpl_keep = 1 }
		}
	}
	if = {
		limit = { check_variable = { pww_kpl_keep = 0 } }
		for_each_loop = {
			array = global.pww_cm_cart
			value = pww_kpl_c
			index = pww_kpl_i
			if = {
				limit = {
					check_variable = { pww_kpl_c = pww_kpl_cid }
					check_variable = { global.pww_cm_status^pww_kpl_i = 1 }
				}
				set_temp_variable = { pww_kpl_w = global.pww_cm_who^pww_kpl_i }
				var:pww_kpl_w = {
					set_temp_variable = { pww_kr_res = pww_kpl_res }
					pww_cartel_read_res = yes
				}
				if = {
					limit = { check_variable = { pww_kr_prod > pww_kpl_best } }
					set_temp_variable = { pww_kpl_best = pww_kr_prod }
					set_temp_variable = { pww_kpl_who = pww_kpl_w }
				}
			}
		}
		set_variable = { global.pww_cart_leader^pww_kpl_cid = pww_kpl_who }
	}
}

# in: pww_in_cm. Status 3, standing -30, opinion of every member, -20

		limit = { check_variable = { global.pww_cm_status^pww_kex_cm = 1 } }

			pww_cartel_clear_member_country = yes

			subtract_from_variable = { pww_cartel_standing = 30 }

		for_each_loop = {

				}
			}
		}
		var:pww_kex_who = {
			if = {
				limit = { is_ai = no }

			set_temp_variable = { pww_kpl_cid = pww_kex_cid }

	}
}

# in: pww_kck_cid. Active cartel with cohesion under the collapse line or
# fewer than two members falls into a price war.
pww_cartel_check_collapse = {
	if = {
		limit = { check_variable = { global.pww_cart_status^pww_kck_cid = 1 } }
		if = {
			limit = { check_variable = { global.pww_cart_n^pww_kck_cid < 2 } }
			set_temp_variable = { pww_in_cartel_id = pww_kck_cid }
			set_temp_variable = { pww_in_cause = 2 }
			pww_cartel_collapse = yes
		}
		else_if = {
			limit = { check_variable = { global.pww_cart_cohesion^pww_kck_cid < global.pww_cartel_c_collapse } }
			set_temp_variable = { pww_in_cartel_id = pww_kck_cid }
			set_temp_variable = { pww_in_cause = 1 }
			pww_cartel_collapse = yes
		}
	}
}

# in: pww_in_cartel_id, pww_in_cause. Active -> collapsed (price war). Every
# member: status 5, quota effects cleared, its fallback pool dumped over
# pww_cartel_c_dump_days, timed flag pww_cartel_pricewar_<res>.
pww_cartel_collapse = {
	set_temp_variable = { pww_kco2_cid = pww_in_cartel_id }
	if = {
		limit = { check_variable = { global.pww_cart_status^pww_kco2_cid = 1 } }
		set_variable = { global.pww_cart_status^pww_kco2_cid = 2 }
		set_variable = { global.pww_cart_cause^pww_kco2_cid = pww_in_cause }
		set_variable = { global.pww_cart_end_day^pww_kco2_cid = global.num_days }
		set_temp_variable = { pww_kco2_pw = global.num_days }
		add_to_temp_variable = { pww_kco2_pw = global.pww_cartel_c_pricewar_days }
		set_variable = { global.pww_cart_pw_until^pww_kco2_cid = pww_kco2_pw }
		set_temp_variable = { pww_kco2_res = global.pww_cart_res^pww_kco2_cid }
		set_variable = { global.pww_cart_by_res^pww_kco2_res = -1 }
		set_temp_variable = { pww_kco2_cd = global.num_days }
		add_to_temp_variable = { pww_kco2_cd = global.pww_cartel_c_cooldown }
		set_variable = { global.pww_cart_cd_res^pww_kco2_res = pww_kco2_cd }
		remove_from_array = { array = global.pww_cart_live value = pww_kco2_cid }
		for_each_loop = {
			array = global.pww_cm_cart
			value = pww_kco2_c
			index = pww_kco2_i
			if = {
				limit = {
					check_variable = { pww_kco2_c = pww_kco2_cid }
					check_variable = { global.pww_cm_status^pww_kco2_i = 1 }
				}
				set_variable = { global.pww_cm_status^pww_kco2_i = 5 }
				set_temp_variable = { pww_kco2_who = global.pww_cm_who^pww_kco2_i }
				var:pww_kco2_who = {
					remove_from_array = { array = pww_cart_my value = pww_kco2_i }
					set_temp_variable = { pww_kcl_res = pww_kco2_res }
					pww_cartel_clear_member_country = yes
					# Dump the fallback pool
					set_temp_variable = { pww_kr_res = pww_kco2_res }
					pww_cartel_read_res = yes
					set_temp_variable = { pww_kco2_d = pww_kr_held }
					divide_temp_variable = { pww_kco2_d = global.pww_cartel_c_dump_days }
					set_temp_variable = { pww_kw_res = pww_kco2_res }
					set_temp_variable = { pww_kw_hold = 0 }
					set_temp_variable = { pww_kw_store = 0 }
					set_temp_variable = { pww_kw_dump = pww_kco2_d }
					set_temp_variable = { pww_kw_held = 0 }
					pww_cartel_write_res = yes
					set_variable = { pww_cartel_dump_until = global.num_days }
					add_to_variable = { pww_cartel_dump_until = global.pww_cartel_c_dump_days }
					if = { limit = { check_variable = { pww_kco2_res = 1 } } set_country_flag = { flag = pww_cartel_pricewar_aluminium days = 180 value = 1 } }
					else_if = { limit = { check_variable = { pww_kco2_res = 2 } } set_country_flag = { flag = pww_cartel_pricewar_rubber days = 180 value = 1 } }
					else_if = { limit = { check_variable = { pww_kco2_res = 3 } } set_country_flag = { flag = pww_cartel_pricewar_tungsten days = 180 value = 1 } }
					else_if = { limit = { check_variable = { pww_kco2_res = 4 } } set_country_flag = { flag = pww_cartel_pricewar_steel days = 180 value = 1 } }
					else_if = { limit = { check_variable = { pww_kco2_res = 5 } } set_country_flag = { flag = pww_cartel_pricewar_chromium days = 180 value = 1 } }
					else_if = { limit = { check_variable = { pww_kco2_res = 6 } } set_country_flag = { flag = pww_cartel_pricewar_coal days = 180 value = 1 } }
					if = {
						limit = { is_ai = no }
						set_variable = { pww_cartel_ev_cid = pww_kco2_cid }
						country_event = { id = pww_cartel.5 }
					}
				}
			}
		}
		set_variable = { global.pww_cart_n^pww_kco2_cid = 0 }
		add_to_variable = { global.pww_cart_rev = 1 }
		log = "[GetDateText] PWW_CARTEL collapse: cartel [?pww_kco2_cid] res [?pww_kco2_res] cause [?pww_in_cause] price war until day [?pww_kco2_pw]"
	}
}

# in: pww_kc_cid. Failed conference: every status 0/1 entry ends (5).
pww_cartel_release_members = {
	for_each_loop = {
		array = global.pww_cm_cart
		value = pww_krm_c
		index = pww_krm_i
		if = {
			limit = {
				check_variable = { pww_krm_c = pww_kc_cid }
				check_variable = { global.pww_cm_status^pww_krm_i < 2 }
			}
			set_variable = { global.pww_cm_status^pww_krm_i = 5 }
			set_temp_variable = { pww_krm_who = global.pww_cm_who^pww_krm_i }
			set_temp_variable = { pww_kcl_res = global.pww_cart_res^pww_kc_cid }
			var:pww_krm_who = {
				remove_from_array = { array = pww_cart_my value = pww_krm_i }
				pww_cartel_clear_member_country = yes
			}
		}
	}
	set_variable = { global.pww_cart_n^pww_kc_cid = 0 }
}

###############################################################################
# Formation scan (monthly). For each resource without a live cartel and out
# of cooldown: flag glut producers (unsold >= glut_min and >= glut_share of
# output), count buyers at desperation 50+. Ripe = enough glut producers and
# few desperate buyers. The biggest AI glut producer then calls a
# conference with pww_cartel_c_form_chance; a human glut producer gets the
# decision (flag pww_cartel_glut_<res>, global.pww_cart_ripe^res). At most
# one AI conference per scan, and none while pww_cartel_c_max_live cartels
# are live (HoI4 is oversupplied in almost everything, so ripeness alone
# would found a cartel per resource in the first month).
###############################################################################
pww_cartel_formation_scan = {
	set_temp_variable = { pww_kfs_called = 0 }
	for_loop_effect = {
		start = 1
		end = 7
		value = pww_kfs_res
		set_variable = { global.pww_cart_ripe^pww_kfs_res = 0 }
		set_variable = { global.pww_cart_glut_n^pww_kfs_res = 0 }
		set_temp_variable = { pww_kfs_desp_n = 0 }
		set_temp_variable = { pww_kfs_best = 0 }
		set_temp_variable = { pww_kfs_best_who = -1 }
		every_country = {
			limit = { exists = yes }
			set_temp_variable = { pww_kf_res = pww_kfs_res }
			if = { limit = { check_variable = { pww_kfs_res = 1 } } clr_country_flag = pww_cartel_glut_aluminium }
			else_if = { limit = { check_variable = { pww_kfs_res = 2 } } clr_country_flag = pww_cartel_glut_rubber }
			else_if = { limit = { check_variable = { pww_kfs_res = 3 } } clr_country_flag = pww_cartel_glut_tungsten }
			else_if = { limit = { check_variable = { pww_kfs_res = 4 } } clr_country_flag = pww_cartel_glut_steel }
			else_if = { limit = { check_variable = { pww_kfs_res = 5 } } clr_country_flag = pww_cartel_glut_chromium }
			else_if = { limit = { check_variable = { pww_kfs_res = 6 } } clr_country_flag = pww_cartel_glut_coal }
			set_temp_variable = { pww_kr_res = pww_kfs_res }
			pww_cartel_read_res = yes
			if = {
				limit = { check_variable = { pww_kr_desp > 49.999 } }
				add_to_temp_variable = { pww_kfs_desp_n = 1 }
			}
			set_temp_variable = { pww_kfs_need = pww_kr_prod }
			multiply_temp_variable = { pww_kfs_need = global.pww_cartel_c_glut_share }
			if = {
				limit = {
					check_variable = { pww_kr_produces = 1 }
					has_war = no
					check_variable = { pww_kr_unsold > global.pww_cartel_c_glut_min }
					check_variable = { pww_kr_unsold > pww_kfs_need }
				}
				if = { limit = { check_variable = { pww_kfs_res = 1 } } set_country_flag = pww_cartel_glut_aluminium }
				else_if = { limit = { check_variable = { pww_kfs_res = 2 } } set_country_flag = pww_cartel_glut_rubber }
				else_if = { limit = { check_variable = { pww_kfs_res = 3 } } set_country_flag = pww_cartel_glut_tungsten }
				else_if = { limit = { check_variable = { pww_kfs_res = 4 } } set_country_flag = pww_cartel_glut_steel }
				else_if = { limit = { check_variable = { pww_kfs_res = 5 } } set_country_flag = pww_cartel_glut_chromium }
				else_if = { limit = { check_variable = { pww_kfs_res = 6 } } set_country_flag = pww_cartel_glut_coal }
				add_to_variable = { global.pww_cart_glut_n^pww_kfs_res = 1 }
				pww_cartel_ensure_country = yes
				if = {
					limit = {
						is_ai = yes
						check_variable = { pww_kr_unsold > pww_kfs_best }
						check_variable = { pww_cartel_standing > 29.999 }
					}
					set_temp_variable = { pww_kfs_best = pww_kr_unsold }
					set_temp_variable = { pww_kfs_best_who = THIS }
				}
			}
		}
		if = {
			limit = {
				check_variable = { global.pww_cart_by_res^pww_kfs_res = -1 }
				NOT = { check_variable = { global.pww_cart_cd_res^pww_kfs_res > global.num_days } }
				NOT = { check_variable = { global.pww_cart_glut_n^pww_kfs_res < global.pww_cartel_c_glut_producers } }
				NOT = { check_variable = { pww_kfs_desp_n > global.pww_cartel_c_desp_buyers } }
			}
			set_variable = { global.pww_cart_ripe^pww_kfs_res = 1 }
			if = {
				limit = {
					check_variable = { global.pww_cartel_c_ai_form = 1 }
					check_variable = { pww_kfs_best_who > -1 }
					check_variable = { pww_kfs_called = 0 }
					check_variable = { global.pww_cart_live^num < global.pww_cartel_c_max_live }
				}
				set_temp_variable_to_random = { var = pww_kfs_roll min = 0 max = 100 integer = yes }
				if = {
					limit = { check_variable = { pww_kfs_roll < global.pww_cartel_c_form_chance } }
					set_temp_variable = { pww_in_res = pww_kfs_res }
					set_temp_variable = { pww_in_leader = pww_kfs_best_who }
					pww_cartel_open_conference = yes
					set_temp_variable = { pww_kfs_called = 1 }
				}
			}
		}
		if = {
			limit = { has_global_flag = pww_cartel_debug }
			log = "[GetDateText] PWW_CARTEL scan res [?pww_kfs_res]: glut producers [?global.pww_cart_glut_n^pww_kfs_res] desperate buyers [?pww_kfs_desp_n] ripe [?global.pww_cart_ripe^pww_kfs_res] cooldown until [?global.pww_cart_cd_res^pww_kfs_res] best AI [?pww_kfs_best_who.GetTag] ([?pww_kfs_best|1])"
		}
	}
}
