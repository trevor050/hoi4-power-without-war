###############################################################################
# Power Without War: diplomacy lane, ticks
#
# pww_diplo_on_daily (country, from on_actions/pww_diplo_on_actions.txt):
#   lazy init; the global sweep once per day (talk expiry, auctions,
#   pledges, embargo compliance, ledger and coercion queue polls, every
#   5 days for the heavier parts); the country's staggered tick on its core
#   bucket day (global.pww_day_mod30).
#
# AI intents (pww_diplo_country_tick, AI countries only). ai_desire reads
# these flags and variables only, so it stays cheap. Every intent has a
# concrete reason:
#   pww_diplo_ai_offer      (+ pww_diplo_ai_offer_to, _res): desperation for
#                           a resource another active trader has spare (the
#                           supplier with the best delivered deal: spare
#                           units it can use x (1 - 2 x shipping share));
#                           an agreement it buys under expiring within 60
#                           days; or a situation bid intent
#                           (pww_sit_ai_wants_bid). Also set by a human
#                           counter the AI accepts (re-table).
#                           pww_diplo_ai_offer_sell 1 marks a SALE (below).
#   pww_diplo_ai_lease      (+ _lease_to, _lease_res): frantic major whose
#                           supplier has producing states
#   pww_diplo_ai_pledge     (+ _pledge_to, _pledge_vs): a situation pledge
#                           intent (pww_sit_ai_wants_pledge), or a holder it
#                           guarantees or shares a faction with called for a
#                           patron while under a demand
#   pww_diplo_ai_demand     (+ _demand_to): frantic major, coercion loaded,
#                           its trade talks with that producer broke off
#   pww_diplo_ai_quarantine (+ _quarantine_to): its live demand on a
#                           coastal target that hardened, with a fleet
#   pww_diplo_ai_settle     (+ _settle_to): a nationalization against it
#   pww_diplo_ai_embargo    (+ _embargo_to member, _embargo_target): it
#                           embargoes a target for a live demand and a
#                           faction member or subject does not
# Flags are timed (30 days) and cleared when the action is sent.
#
# SALES (glut sellers approach buyers): an AI with no offer intent whose
# unsold output of a resource is worth a factory or more (pww_dg_unsold >= 8
# and >= 15% of its output; a cartel member only with quota room or when it
# cheats) tables Propose Trade Deal to the best delivered buyer (demand:
# open-market imports plus any shortfall still open, at least 4 units, half
# a factory; score min(demand, unsold) x (1 - 2 x shipping share), no
# undeliverable route) among the active traders and human players,
# not at war or embargo either way, neither side's opinion of the other
# below -25 (normalized), no talks between them in the last 90 days or
# cooling. The draft (pww_diplo_ai_build_sale) sells up to that demand
# from the unsold output at 95% of market (90% when 40% or more of its output
# is unsold), 5% less each round, paid in their free civilian factories then
# political power. Rate limits: the seller waits 60 days after a proposal
# (120 for a small glut: under 40% unsold and under 40 units; 30 for a
# planned economy, capacity lane pww_cap_sys 1: it never idles output); a human
# receives at most one every 30 days (pww_diplo_ai_sale_last on the human);
# AI-to-AI sale attempts (sent or dropped) are capped at 6 per 30-day window
# (global.pww_ds_win / global.pww_ds_n). A draft the buyer would not take is
# dropped and its talk closed as superseded.
#
# POACHING (opportunistic AIs, pww_diplo_ai_poach_intent): an AI with no offer
# or sale draft pending and a large glut (unsold >= 40 units and >= 30% of
# its output) of a resource looks for the largest poachable mod supply deal
# in it (active, either side may end it, not ours) and drafts Poach Supplier
# to that buyer (pww_diplo_poach_defaults: 90% of the incumbent's price,
# less 5% a round, less shipping). Ready: flag pww_diplo_ai_poach (+ _to);
# either way a 120-day cooldown (pww_diplo_ai_poach_cd). AI-to-AI attempts
# count against the sales window (global.pww_ds_n).
###############################################################################

pww_diplo_on_daily = {
	if = {
		limit = { NOT = { has_global_flag = pww_diplo_initialized } }
		pww_diplo_init = yes
	}
	if = {
		limit = { NOT = { check_variable = { global.pww_diplo_last_day = global.num_days } } }
		set_variable = { global.pww_diplo_last_day = global.num_days }
		pww_diplo_global_sweep = yes
	}
	if = {
		limit = {
			has_variable = pww_bucket
			check_variable = { pww_bucket = global.pww_day_mod30 }
			NOT = { check_variable = { pww_diplo_tick_day = global.num_days } }
		}
		set_variable = { pww_diplo_tick_day = global.num_days }
		pww_diplo_country_tick = yes
	}
	# A situation seat's fresh intent is picked up within a week instead of
	# on the next bucket day (the situation waits pww_sit_c_diplo_grace).
	else_if = {
		limit = {
			is_ai = yes
			OR = {
				AND = {
					has_country_flag = pww_sit_ai_wants_bid
					NOT = { has_country_flag = pww_diplo_ai_offer }
				}
				AND = {
					has_country_flag = pww_sit_ai_wants_pledge
					NOT = { has_country_flag = pww_diplo_ai_pledge }
				}
			}
			NOT = { has_country_flag = pww_diplo_sit_early }
			NOT = { check_variable = { pww_diplo_tick_day = global.num_days } }
		}
		set_country_flag = { flag = pww_diplo_sit_early days = 7 value = 1 }
		set_variable = { pww_diplo_tick_day = global.num_days }
		pww_diplo_country_tick = yes
	}
}

###############################################################################
# Global sweep (any scope)
###############################################################################
pww_diplo_global_sweep = {
	# Talk expiry
	clear_temp_array = pww_dgs_exp
	for_each_loop = {
		array = global.pww_dt_live
		value = pww_dgs_t
		if = {
			limit = { check_variable = { global.pww_dt_expires^pww_dgs_t < global.num_days } }
			add_to_temp_array = { pww_dgs_exp = pww_dgs_t }
		}
	}
	for_each_loop = {
		array = pww_dgs_exp
		value = pww_dgs_t
		set_temp_variable = { pww_dtc_t = pww_dgs_t }
		set_temp_variable = { pww_dtc_status = 3 }
		pww_diplo_talk_close = yes
	}
	pww_diplo_auctions_sweep = yes
	set_temp_variable = { pww_dgs_m = global.num_days }
	modulo_temp_variable = { var = pww_dgs_m value = 5 }
	if = {
		limit = { check_variable = { pww_dgs_m = 0 } }
		pww_diplo_pledges_sweep = yes
		pww_diplo_embargo_sweep = yes
		pww_diplo_ledger_queue_poll = yes
		pww_diplo_coer_queue_poll = yes
		pww_diplo_auto_auctions = yes
	}
}

###############################################################################
# AI holders turn competing talks into an auction: two or more distinct
# senders in open talks to buy the same resource from one AI holder.
###############################################################################
pww_diplo_auto_auctions = {
	for_each_loop = {
		array = global.pww_dt_live
		value = pww_daut_t
		set_temp_variable = { pww_daut_h = global.pww_dt_target^pww_daut_t }
		set_temp_variable = { pww_daut_s = global.pww_dt_sender^pww_daut_t }
		if = {
			limit = {
				check_variable = { global.pww_dt_status^pww_daut_t = 0 }
				var:pww_daut_h = {
					is_ai = yes
					NOT = { has_country_flag = pww_diplo_auctioning }
				}
				var:pww_daut_s = { check_variable = { pww_do_target = pww_daut_h } }
			}
			set_temp_variable = { pww_daut_r = -1 }
			var:pww_daut_s = {
				if = {
					limit = {
						check_variable = { pww_do_talk = pww_daut_t }
						OR = {
							check_variable = { pww_do_act = 1 }
							check_variable = { pww_do_dir = 0 }
						}
					}
					set_temp_variable = { pww_daut_r = pww_do_res }
				}
			}
			if = {
				limit = { check_variable = { pww_daut_r > -1 } }
				# A second sender after the same resource from this holder?
				set_temp_variable = { pww_daut_other = -1 }
				for_each_loop = {
					array = global.pww_dt_live
					value = pww_daut_u
					set_temp_variable = { pww_daut_s2 = global.pww_dt_sender^pww_daut_u }
					if = {
						limit = {
							NOT = { check_variable = { pww_daut_u = pww_daut_t } }
							check_variable = { global.pww_dt_status^pww_daut_u = 0 }
							check_variable = { global.pww_dt_target^pww_daut_u = pww_daut_h }
							NOT = { check_variable = { pww_daut_s2 = pww_daut_s } }
							var:pww_daut_s2 = {
								check_variable = { pww_do_talk = pww_daut_u }
								check_variable = { pww_do_res = pww_daut_r }
							}
						}
						set_temp_variable = { pww_daut_other = pww_daut_u }
					}
				}
				if = {
					limit = { check_variable = { pww_daut_other > -1 } }
					var:pww_daut_h = { set_country_flag = { flag = pww_diplo_auctioning days = 60 value = 1 } }
					set_temp_variable = { pww_dg_a = pww_daut_s }
					set_temp_variable = { pww_dg_b = pww_daut_h }
					pww_diplo_find_shared_sit = yes
					set_temp_variable = { pww_in_holder = pww_daut_h }
					set_temp_variable = { pww_in_resource = pww_daut_r }
					set_temp_variable = { pww_in_sit_id = pww_dg_sit }
					set_temp_variable = { pww_in_days = 0 }
					pww_diplo_auction_open = yes
					set_temp_variable = { pww_in_auction_id = pww_out_auction_id }
					pww_diplo_auction_write_default_lots = yes
					# Supersede the competing talks: bids replace them.
					set_temp_variable = { pww_dtc_t = pww_daut_t }
					set_temp_variable = { pww_dtc_status = 4 }
					pww_diplo_talk_close = yes
					set_temp_variable = { pww_dtc_t = pww_daut_other }
					set_temp_variable = { pww_dtc_status = 4 }
					pww_diplo_talk_close = yes
				}
			}
		}
	}
}

###############################################################################
# Country tick (staggered, about monthly per country)
###############################################################################
pww_diplo_country_tick = {
	if = {
		limit = { check_variable = { pww_dpl_ids^num > 0 } }
		pww_diplo_pledge_strength = yes
	}
	if = {
		limit = {
			is_ai = yes
			has_global_flag = pww_initialized
			has_war = no
		}
		pww_diplo_ai_intents = yes
	}
}

pww_diplo_ai_intents = {
	set_temp_variable = { pww_dai_me = THIS }
	# --- Offer: situation bid intent first, then expiry, then desperation
	if = {
		limit = { NOT = { has_country_flag = pww_diplo_ai_offer } }
		set_temp_variable = { pww_dai_to = -1 }
		set_temp_variable = { pww_dai_res = -1 }
		if = {
			limit = {
				has_country_flag = pww_sit_ai_wants_bid
				has_variable = pww_sit_ai_sit
			}
			set_temp_variable = { pww_dai_sit = pww_sit_ai_sit }
			set_temp_variable = { pww_dai_to = global.pww_sit_holder^pww_dai_sit }
			set_temp_variable = { pww_dai_res = global.pww_sit_res^pww_dai_sit }
			if = {
				limit = { check_variable = { pww_dai_to = THIS } }
				set_temp_variable = { pww_dai_to = -1 }
			}
		}
		if = {
			limit = { check_variable = { pww_dai_to = -1 } }
			for_each_loop = {
				array = pww_ag_ids
				value = pww_dai_a
				if = {
					limit = {
						check_variable = { global.pww_ag_status^pww_dai_a = 1 }
						check_variable = { global.pww_ag_buyer^pww_dai_a = THIS }
						check_variable = { global.pww_ag_kind^pww_dai_a = 0 }
						check_variable = { global.pww_ag_end^pww_dai_a > 0 }
					}
					set_temp_variable = { pww_dai_left = global.pww_ag_end^pww_dai_a }
					subtract_from_temp_variable = { pww_dai_left = global.num_days }
					if = {
						limit = { check_variable = { pww_dai_left < 60 } }
						set_temp_variable = { pww_dai_to = global.pww_ag_seller^pww_dai_a }
						set_temp_variable = { pww_dai_res = global.pww_ag_res^pww_dai_a }
					}
				}
			}
		}
		if = {
			limit = {
				check_variable = { pww_dai_to = -1 }
				pww_core_is_desperate_any = yes
				check_variable = { pww_desp_max_res > -1 }
			}
			set_temp_variable = { pww_dai_res = pww_desp_max_res }
			# Delivered cost picks the supplier: spare units we can use, less
			# twice the shipping share (a 20% sea route counts 60%).
			set_temp_variable = { pww_dg_res = pww_dai_res }
			pww_diplo_read_market = yes
			set_temp_variable = { pww_dai_need = pww_dg_demand }
			if = {
				limit = { check_variable = { pww_dai_need < 2 } }
				set_temp_variable = { pww_dai_need = 2 }
			}
			set_temp_variable = { pww_dai_best = 0 }
			for_each_loop = {
				array = global.pww_active_traders
				value = pww_dai_c
				if = {
					limit = {
						NOT = { check_variable = { pww_dai_c = pww_dai_me } }
						var:pww_dai_c = {
							exists = yes
							NOT = { has_war_with = var:pww_dai_me }
						}
					}
					set_temp_variable = { pww_dg_res = pww_dai_res }
					var:pww_dai_c = { pww_diplo_read_res = yes }
					if = {
						limit = { check_variable = { pww_dg_net > 1.999 } }
						set_temp_variable = { pww_dai_sc = pww_dg_net }
						if = {
							limit = { check_variable = { pww_dai_sc > pww_dai_need } }
							set_temp_variable = { pww_dai_sc = pww_dai_need }
						}
						set_temp_variable = { pww_dsh_s = pww_dai_c }
						set_temp_variable = { pww_dsh_b = pww_dai_me }
						set_temp_variable = { pww_dsh_r = pww_dai_res }
						pww_diplo_ship = yes
						set_temp_variable = { pww_dai_m = pww_dsh_share }
						multiply_temp_variable = { pww_dai_m = -2 }
						add_to_temp_variable = { pww_dai_m = 1 }
						clamp_temp_variable = { var = pww_dai_m min = 0.05 max = 1 }
						multiply_temp_variable = { pww_dai_sc = pww_dai_m }
						if = {
							limit = {
								check_variable = { pww_dai_sc > pww_dai_best }
								NOT = { check_variable = { pww_dsh_route = 2 } }
							}
							set_temp_variable = { pww_dk_target = pww_dai_c }
							pww_diplo_talk_cooling = yes
							if = {
								limit = {
									check_variable = { pww_dk_cool = 0 }
									check_variable = { pww_dk_recent = 0 }
								}
								set_temp_variable = { pww_dai_best = pww_dai_sc }
								set_temp_variable = { pww_dai_to = pww_dai_c }
							}
						}
					}
				}
			}
		}
		if = {
			limit = { check_variable = { pww_dai_to > -1 } }
			set_variable = { pww_diplo_ai_offer_to = pww_dai_to }
			set_variable = { pww_diplo_ai_offer_res = pww_dai_res }
			set_variable = { pww_diplo_ai_offer_sell = 0 }
			set_temp_variable = { pww_dk_target = pww_dai_to }
			set_temp_variable = { pww_dk_act = 0 }
			set_temp_variable = { pww_dap_res = pww_dai_res }
			pww_diplo_ai_prepare_offer = yes
			if = {
				limit = { check_variable = { pww_dap_ready = 1 } }
				set_country_flag = { flag = pww_diplo_ai_offer days = 30 value = 1 }
			}
		}
	}
	# --- Sale: unsold output looks for a buyer (SALES above)
	if = {
		limit = {
			NOT = { has_country_flag = pww_diplo_ai_offer }
			NOT = { has_country_flag = pww_diplo_ai_sell_cd }
			NOT = { has_country_flag = pww_diplo_ai_poach }
		}
		pww_diplo_ai_sale_intent = yes
	}
	# --- Poach: a large glut looks for a supply deal to take over (POACHING)
	if = {
		limit = {
			NOT = { has_country_flag = pww_diplo_ai_offer }
			NOT = { has_country_flag = pww_diplo_ai_poach }
			NOT = { has_country_flag = pww_diplo_ai_poach_cd }
			check_variable = { global.pww_next_agreement_id > 0 }
		}
		pww_diplo_ai_poach_intent = yes
	}
	# --- Lease: frantic major, its chosen supplier has producing states
	if = {
		limit = {
			NOT = { has_country_flag = pww_diplo_ai_lease }
			is_major = yes
			pww_core_is_frantic_any = yes
			has_variable = pww_diplo_ai_offer_to
		}
		set_temp_variable = { pww_dg_res = pww_diplo_ai_offer_res }
		set_temp_variable = { pww_dg_slot = 0 }
		var:pww_diplo_ai_offer_to = { pww_diplo_pick_state = yes }
		# Only when no supply draft is ready: one tabled proposal at a time.
		if = {
			limit = { check_variable = { pww_dg_nstates > 0 } }
			set_variable = { pww_diplo_ai_lease_to = pww_diplo_ai_offer_to }
			set_variable = { pww_diplo_ai_lease_res = pww_diplo_ai_offer_res }
		}
	}
	if = {
		limit = {
			NOT = { has_country_flag = pww_diplo_ai_lease }
			NOT = { has_country_flag = pww_diplo_ai_offer }
			is_major = yes
			pww_core_is_frantic_any = yes
			has_variable = pww_diplo_ai_lease_to
		}
		set_temp_variable = { pww_dk_target = pww_diplo_ai_lease_to }
		set_temp_variable = { pww_dk_act = 1 }
		set_temp_variable = { pww_dap_res = pww_diplo_ai_lease_res }
		pww_diplo_ai_prepare_offer = yes
		if = {
			limit = { check_variable = { pww_dap_ready = 1 } }
			set_country_flag = { flag = pww_diplo_ai_lease days = 30 value = 1 }
		}
	}
	# --- Pledge: situation intent, or a called-for patron duty
	if = {
		limit = { NOT = { has_country_flag = pww_diplo_ai_pledge } }
		set_temp_variable = { pww_dai_ph = -1 }
		set_temp_variable = { pww_dai_pv = 0 }
		if = {
			limit = {
				has_country_flag = pww_sit_ai_wants_pledge
				has_variable = pww_sit_ai_sit
			}
			set_temp_variable = { pww_dai_sit = pww_sit_ai_sit }
			set_temp_variable = { pww_dai_ph = global.pww_sit_holder^pww_dai_sit }
			set_temp_variable = { pww_dai_pv = global.pww_sit_claimant^pww_dai_sit }
		}
		else = {
			every_country = {
				limit = {
					has_country_flag = pww_diplo_called_patron
					check_variable = { pww_dd_pressers^num > 0 }
					OR = {
						var:pww_dai_me = { has_guaranteed = PREV }
						is_in_faction_with = var:pww_dai_me
					}
				}
				set_temp_variable = { pww_dai_ph = THIS }
				set_temp_variable = { pww_dai_pv = pww_dd_pressers^0 }
			}
		}
		if = {
			limit = { check_variable = { pww_dai_ph > -1 } }
			set_variable = { pww_diplo_ai_pledge_to = pww_dai_ph }
			set_variable = { pww_diplo_ai_pledge_vs = pww_dai_pv }
			# The engine reads ai_acceptance on the standing draft before
			# reset_send_effect: build it now.
			set_temp_variable = { pww_dk_target = pww_dai_ph }
			pww_diplo_pledge_begin = yes
			set_temp_variable = { pww_dkp_vs = pww_dai_pv }
			var:pww_dai_ph = {
				for_each_loop = {
					array = pww_dd_pressers
					value = pww_dkp_c
					index = pww_dkp_i
					if = {
						limit = { check_variable = { pww_dkp_c = pww_dkp_vs } }
						PREV = { set_variable = { pww_dp_ci = pww_dkp_i } }
					}
				}
			}
			pww_diplo_pledge_changed = yes
			if = {
				limit = { check_variable = { pww_dp_valid = 0 } }
				set_country_flag = { flag = pww_diplo_ai_pledge days = 30 value = 1 }
			}
		}
	}
	# --- Settlement
	if = {
		limit = {
			has_country_flag = pww_dset_open
			check_variable = { pww_dset_hosts^num > 0 }
			NOT = { has_country_flag = pww_diplo_ai_settle }
		}
		set_variable = { pww_diplo_ai_settle_to = pww_dset_hosts^0 }
		set_country_flag = { flag = pww_diplo_ai_settle days = 30 value = 1 }
	}
	# --- Demand: frantic major whose trade talks with a producer broke off
	if = {
		limit = {
			has_global_flag = pww_coer_loaded
			is_major = yes
			pww_core_is_frantic_any = yes
			NOT = { has_country_flag = pww_diplo_ai_demand }
		}
		set_temp_variable = { pww_dai_dt = -1 }
		for_each_loop = {
			array = pww_dt_ids
			value = pww_dai_t
			if = {
				limit = {
					check_variable = { global.pww_dt_status^pww_dai_t = 2 }
					check_variable = { global.pww_dt_cool_until^pww_dai_t > global.num_days }
				}
				set_temp_variable = { pww_dai_dt = global.pww_dt_target^pww_dai_t }
			}
		}
		if = {
			limit = {
				check_variable = { pww_dai_dt > -1 }
				var:pww_dai_dt = {
					is_major = no
					NOT = { has_war_with = var:pww_dai_me }
				}
			}
			set_temp_variable = { pww_dai_has = 0 }
			for_each_loop = {
				array = global.pww_coer_live
				value = pww_dai_d
				if = {
					limit = { check_variable = { global.pww_cd_claimant^pww_dai_d = pww_dai_me } }
					set_temp_variable = { pww_dai_has = 1 }
				}
			}
			# A target's ai_acceptance of a demand means folding, so the
			# engine would never let the AI send it: AI demands go straight
			# to the coercion lane (same draft and call as the action).
			if = {
				limit = { check_variable = { pww_dai_has = 0 } }
				set_variable = { pww_diplo_ai_demand_to = pww_dai_dt }
				set_country_flag = { flag = pww_diplo_ai_demand days = 30 value = 1 }
				set_temp_variable = { pww_dk_target = pww_dai_dt }
				set_temp_variable = { pww_dk_channel = 0 }
				if = {
					limit = { has_government = democratic }
					set_temp_variable = { pww_dk_channel = 1 }
				}
				pww_diplo_demand_begin = yes
				if = {
					limit = { check_variable = { pww_dm_valid = 0 } }
					pww_diplo_demand_sent = yes
				}
			}
		}
	}
	# --- Quarantine: a hardened target of our live demand, coast and fleet
	if = {
		limit = {
			has_global_flag = pww_coer_loaded
			NOT = { has_country_flag = pww_diplo_ai_quarantine }
			has_navy_size = { size > 19 }
		}
		for_each_loop = {
			array = global.pww_coer_live
			value = pww_dai_d
			if = {
				limit = {
					check_variable = { global.pww_cd_claimant^pww_dai_d = pww_dai_me }
					check_variable = { global.pww_cd_hardened^pww_dai_d = 1 }
					check_variable = { global.pww_cd_quar^pww_dai_d = 0 }
				}
				set_temp_variable = { pww_dai_qt = global.pww_cd_target^pww_dai_d }
				if = {
					limit = { var:pww_dai_qt = { check_variable = { pww_coastal_states^num > 0 } } }
					set_variable = { pww_diplo_ai_quarantine_to = pww_dai_qt }
					set_country_flag = { flag = pww_diplo_ai_quarantine days = 30 value = 1 }
				}
			}
		}
	}
	# --- Join our embargo: a member of our bloc still trades with a target we
	# embargo for a live demand.
	if = {
		limit = {
			has_global_flag = pww_coer_loaded
			NOT = { has_country_flag = pww_diplo_ai_embargo }
		}
		for_each_loop = {
			array = global.pww_coer_live
			value = pww_dai_d
			if = {
				limit = {
					check_variable = { global.pww_cd_claimant^pww_dai_d = pww_dai_me }
					check_variable = { global.pww_cd_emb^pww_dai_d = 1 }
				}
				set_temp_variable = { pww_dai_et = global.pww_cd_target^pww_dai_d }
				set_temp_variable = { pww_dai_em = 0 }
				every_country = {
					limit = {
						exists = yes
						NOT = { check_variable = { pww_dai_et = THIS } }
						OR = {
							is_in_faction_with = var:pww_dai_me
							is_subject_of = var:pww_dai_me
						}
						NOT = { is_embargoing = var:pww_dai_et }
					}
					set_temp_variable = { pww_dai_em = THIS }
				}
				if = {
					limit = { check_variable = { pww_dai_em > 0 } }
					set_variable = { pww_diplo_ai_embargo_to = pww_dai_em }
					set_variable = { pww_diplo_ai_embargo_target = pww_dai_et }
					set_temp_variable = { pww_dk_target = pww_dai_em }
					pww_diplo_embargo_begin = yes
					for_each_loop = {
						array = pww_de_list
						value = pww_dke_c
						index = pww_dke_i
						if = {
							limit = { check_variable = { pww_dke_c = pww_dai_et } }
							set_variable = { pww_de_i = pww_dke_i }
						}
					}
					pww_diplo_embargo_changed = yes
					set_country_flag = { flag = pww_diplo_ai_embargo days = 30 value = 1 }
				}
			}
		}
	}
	# --- Auctions: bid on open lots of holders we need
	for_each_loop = {
		array = global.pww_dau_live
		value = pww_dai_au
		if = {
			limit = {
				check_variable = { global.pww_dau_status^pww_dai_au = 0 }
				NOT = { check_variable = { global.pww_dau_holder^pww_dai_au = pww_dai_me } }
			}
			set_temp_variable = { pww_dg_res = global.pww_dau_res^pww_dai_au }
			pww_diplo_read_res = yes
			if = {
				limit = {
					OR = {
						check_variable = { pww_dg_desp > 24.999 }
						is_major = yes
					}
				}
				set_temp_variable = { pww_in_auction_id = pww_dai_au }
				pww_diplo_auction_ai_bid = yes
			}
		}
	}
}

###############################################################################
# AI sale intent (AI country scope; SALES in the header).
###############################################################################
pww_diplo_ai_sale_intent = {
	set_temp_variable = { pww_dsi_me = THIS }
	set_temp_variable = { pww_dsi_res = -1 }
	set_temp_variable = { pww_dsi_best = 7.999 }
	set_temp_variable = { pww_dsi_ratio = 0 }
	for_loop_effect = {
		start = 0
		end = 7
		value = pww_dsi_r
		set_temp_variable = { pww_dg_res = pww_dsi_r }
		pww_diplo_read_market = yes
		if = {
			limit = {
				check_variable = { pww_dg_unsold > pww_dsi_best }
				check_variable = { pww_dg_prod > 0.999 }
			}
			set_temp_variable = { pww_dsi_q = pww_dg_unsold }
			divide_temp_variable = { pww_dsi_q = pww_dg_prod }
			# A cartel member sells only within its quota (or when it cheats).
			set_temp_variable = { pww_dsi_u = pww_dg_unsold }
			pww_diplo_sale_cartel = yes
			if = {
				limit = {
					check_variable = { pww_dsl_cheat = 0 }
					check_variable = { pww_dsl_room < 4 }
				}
				set_temp_variable = { pww_dsi_q = 0 }
			}
			if = {
				limit = { check_variable = { pww_dsi_q > 0.149 } }
				set_temp_variable = { pww_dsi_best = pww_dsi_u }
				set_temp_variable = { pww_dsi_res = pww_dsi_r }
				set_temp_variable = { pww_dsi_ratio = pww_dsi_q }
			}
		}
	}
	if = {
		limit = { check_variable = { pww_dsi_res > -1 } }
		# AI-to-AI window
		set_temp_variable = { pww_dsi_w = global.num_days }
		subtract_from_temp_variable = { pww_dsi_w = global.pww_ds_win }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = global.pww_ds_win }
					check_variable = { pww_dsi_w > 29 }
				}
			}
			set_variable = { global.pww_ds_win = global.num_days }
			set_variable = { global.pww_ds_n = 0 }
		}
		set_temp_variable = { pww_dsi_to = -1 }
		set_temp_variable = { pww_dsi_short = 0 }
		set_temp_variable = { pww_dsi_score = 0 }
		every_country = {
			limit = {
				exists = yes
				NOT = { check_variable = { pww_dsi_me = THIS } }
				OR = {
					is_ai = no
					is_in_array = { array = global.pww_active_traders value = THIS }
				}
				NOT = { has_war_with = var:pww_dsi_me }
				NOT = { is_embargoing = var:pww_dsi_me }
				var:pww_dsi_me = { NOT = { is_embargoing = PREV } }
			}
			set_temp_variable = { pww_dg_res = pww_dsi_res }
			pww_diplo_read_market = yes
			set_temp_variable = { pww_dsi_sh = pww_dg_demand }
			# Delivered score: what it would take, less twice the shipping share.
			set_temp_variable = { pww_dsi_sc = pww_dsi_sh }
			if = {
				limit = { check_variable = { pww_dsi_sc > pww_dsi_best } }
				set_temp_variable = { pww_dsi_sc = pww_dsi_best }
			}
			set_temp_variable = { pww_dsh_s = pww_dsi_me }
			set_temp_variable = { pww_dsh_b = THIS }
			set_temp_variable = { pww_dsh_r = pww_dsi_res }
			pww_diplo_ship = yes
			set_temp_variable = { pww_dsi_m = pww_dsh_share }
			multiply_temp_variable = { pww_dsi_m = -2 }
			add_to_temp_variable = { pww_dsi_m = 1 }
			clamp_temp_variable = { var = pww_dsi_m min = 0.05 max = 1 }
			multiply_temp_variable = { pww_dsi_sc = pww_dsi_m }
			if = {
				limit = {
					check_variable = { pww_dsi_sh > 3.999 }
					check_variable = { pww_dsi_sc > pww_dsi_score }
					NOT = { check_variable = { pww_dsh_route = 2 } }
				}
				set_temp_variable = { pww_dsi_ok = 1 }
				if = {
					limit = { is_ai = no }
					if = {
						limit = { has_variable = pww_diplo_ai_sale_last }
						set_temp_variable = { pww_dsi_since = global.num_days }
						subtract_from_temp_variable = { pww_dsi_since = pww_diplo_ai_sale_last }
						if = {
							limit = { check_variable = { pww_dsi_since < 30 } }
							set_temp_variable = { pww_dsi_ok = 0 }
						}
					}
				}
				else_if = {
					limit = { check_variable = { global.pww_ds_n > 5 } }
					set_temp_variable = { pww_dsi_ok = 0 }
				}
				set_temp_variable = { pww_opinion_target = pww_dsi_me }
				pww_core_norm_opinion = yes
				if = {
					limit = { check_variable = { pww_norm_opinion < -25 } }
					set_temp_variable = { pww_dsi_ok = 0 }
				}
				set_temp_variable = { pww_dsi_them = THIS }
				var:pww_dsi_me = {
					set_temp_variable = { pww_opinion_target = pww_dsi_them }
					pww_core_norm_opinion = yes
					set_temp_variable = { pww_dk_target = pww_dsi_them }
					pww_diplo_talk_cooling = yes
				}
				if = {
					limit = {
						OR = {
							check_variable = { pww_norm_opinion < -25 }
							check_variable = { pww_dk_cool = 1 }
							check_variable = { pww_dk_recent = 1 }
						}
					}
					set_temp_variable = { pww_dsi_ok = 0 }
				}
				if = {
					limit = { check_variable = { pww_dsi_ok = 1 } }
					set_temp_variable = { pww_dsi_short = pww_dsi_sh }
					set_temp_variable = { pww_dsi_score = pww_dsi_sc }
					set_temp_variable = { pww_dsi_to = THIS }
				}
			}
		}
		if = {
			limit = { check_variable = { pww_dsi_to > -1 } }
			set_variable = { pww_diplo_ai_offer_to = pww_dsi_to }
			set_variable = { pww_diplo_ai_offer_res = pww_dsi_res }
			set_variable = { pww_diplo_ai_offer_sell = 1 }
			set_temp_variable = { pww_dk_target = pww_dsi_to }
			set_temp_variable = { pww_dk_act = 0 }
			set_temp_variable = { pww_dap_res = pww_dsi_res }
			pww_diplo_ai_prepare_offer = yes
			# Every attempt on an AI counts against the window.
			if = {
				limit = { var:pww_dsi_to = { is_ai = yes } }
				add_to_variable = { global.pww_ds_n = 1 }
			}
			if = {
				limit = { check_variable = { pww_dap_ready = 1 } }
				set_country_flag = { flag = pww_diplo_ai_offer days = 30 value = 1 }
				if = {
					limit = { var:pww_dsi_to = { is_ai = no } }
					var:pww_dsi_to = { set_variable = { pww_diplo_ai_sale_last = global.num_days } }
				}
			}
			else = {
				# Nothing they would take: drop the draft and its talk (no
				# cooldown: superseded).
				set_variable = { pww_diplo_ai_offer_sell = 0 }
				set_temp_variable = { pww_dtc_t = pww_do_talk }
				set_temp_variable = { pww_dtc_status = 4 }
				if = {
					limit = {
						check_variable = { pww_dtc_t > -1 }
						check_variable = { global.pww_dt_status^pww_dtc_t = 0 }
					}
					pww_diplo_talk_close = yes
				}
			}
			# Wait before the next attempt either way. A planned economy
			# (capacity lane: pww_cap_sys 1) never idles output and the state
			# trading monopoly sells its surplus eagerly: 30 days.
			if = {
				limit = { check_variable = { pww_cap_sys = 1 } }
				set_country_flag = { flag = pww_diplo_ai_sell_cd days = 30 value = 1 }
			}
			else_if = {
				limit = {
					OR = {
						check_variable = { pww_dsi_ratio > 0.399 }
						check_variable = { pww_dsi_best > 39.999 }
					}
				}
				set_country_flag = { flag = pww_diplo_ai_sell_cd days = 60 value = 1 }
			}
			else = {
				set_country_flag = { flag = pww_diplo_ai_sell_cd days = 120 value = 1 }
			}
			if = {
				limit = { has_global_flag = pww_diplo_debug }
				log = "[GetDateText] PWW_DIPLO sale intent [THIS.GetTag]->[?pww_dsi_to.GetTag] res=[?pww_dsi_res|0] unsold=[?pww_dsi_best|0] ratio=[?pww_dsi_ratio|2] their_short=[?pww_dsi_short|0] ready=[?pww_dap_ready|0] ok=[?pww_do_ok|0] total=[?pww_do_e_total|0] we_give=[?pww_dob_a^0|0] they_give=[?pww_dob_k^4|0]x[?pww_dob_a^4|2]+[?pww_dob_k^5|0]x[?pww_dob_a^5|2] market=[?pww_do_mk_m0|2]/[?pww_do_mk_m1|2]"
			}
		}
	}
}

###############################################################################
# AI poach intent (AI country scope; POACHING in the header).
###############################################################################
pww_diplo_ai_poach_intent = {
	set_temp_variable = { pww_dpi_me = THIS }
	set_temp_variable = { pww_dpi_res = -1 }
	set_temp_variable = { pww_dpi_best = 39.999 }
	for_loop_effect = {
		start = 1
		end = 7
		value = pww_dpi_r
		set_temp_variable = { pww_dg_res = pww_dpi_r }
		pww_diplo_read_market = yes
		if = {
			limit = {
				check_variable = { pww_dg_unsold > pww_dpi_best }
				check_variable = { pww_dg_prod > 0.999 }
			}
			set_temp_variable = { pww_dpi_q = pww_dg_unsold }
			divide_temp_variable = { pww_dpi_q = pww_dg_prod }
			pww_diplo_sale_cartel = yes
			if = {
				limit = {
					check_variable = { pww_dpi_q > 0.299 }
					OR = {
						check_variable = { pww_dsl_cheat = 1 }
						check_variable = { pww_dsl_price < 0.001 }
					}
				}
				set_temp_variable = { pww_dpi_best = pww_dg_unsold }
				set_temp_variable = { pww_dpi_res = pww_dpi_r }
			}
		}
	}
	if = {
		limit = { check_variable = { pww_dpi_res > -1 } }
		set_country_flag = { flag = pww_diplo_ai_poach_cd days = 120 value = 1 }
		# The largest poachable deal in it (scan the ledger; the AI-to-AI
		# window caps attempts).
		set_temp_variable = { pww_dpi_ag = -1 }
		set_temp_variable = { pww_dpi_amt = 3.999 }
		for_loop_effect = {
			start = 0
			end = global.pww_next_agreement_id
			value = pww_dpi_a
			if = {
				limit = {
					check_variable = { global.pww_ag_status^pww_dpi_a = 1 }
					check_variable = { global.pww_ag_kind^pww_dpi_a = 0 }
					check_variable = { global.pww_ag_committal^pww_dpi_a = 0 }
					check_variable = { global.pww_ag_res^pww_dpi_a = pww_dpi_res }
					NOT = { check_variable = { global.pww_ag_seller^pww_dpi_a = pww_dpi_me } }
					NOT = { check_variable = { global.pww_ag_buyer^pww_dpi_a = pww_dpi_me } }
					check_variable = { global.pww_ag_amount^pww_dpi_a > pww_dpi_amt }
				}
				set_temp_variable = { pww_dpi_b = global.pww_ag_buyer^pww_dpi_a }
				if = {
					limit = {
						var:pww_dpi_b = {
							exists = yes
							NOT = { has_war_with = var:pww_dpi_me }
							NOT = { is_embargoing = var:pww_dpi_me }
						}
						NOT = { is_embargoing = var:pww_dpi_b }
					}
					set_temp_variable = { pww_dk_target = pww_dpi_b }
					pww_diplo_talk_cooling = yes
					if = {
						limit = {
							check_variable = { pww_dk_cool = 0 }
							check_variable = { pww_dk_recent = 0 }
						}
						set_temp_variable = { pww_dpi_amt = global.pww_ag_amount^pww_dpi_a }
						set_temp_variable = { pww_dpi_ag = pww_dpi_a }
					}
				}
			}
		}
		set_temp_variable = { pww_dpi_ok = 1 }
		if = {
			limit = { check_variable = { pww_dpi_ag > -1 } }
			set_temp_variable = { pww_dpi_to = global.pww_ag_buyer^pww_dpi_ag }
			if = {
				limit = { var:pww_dpi_to = { is_ai = yes } }
				set_temp_variable = { pww_dsi_w = global.num_days }
				subtract_from_temp_variable = { pww_dsi_w = global.pww_ds_win }
				if = {
					limit = {
						OR = {
							NOT = { has_variable = global.pww_ds_win }
							check_variable = { pww_dsi_w > 29 }
						}
					}
					set_variable = { global.pww_ds_win = global.num_days }
					set_variable = { global.pww_ds_n = 0 }
				}
				if = {
					limit = { check_variable = { global.pww_ds_n > 5 } }
					set_temp_variable = { pww_dpi_ok = 0 }
				}
				else = {
					add_to_variable = { global.pww_ds_n = 1 }
				}
			}
		}
		else = {
			set_temp_variable = { pww_dpi_ok = 0 }
		}
		if = {
			limit = { check_variable = { pww_dpi_ok = 1 } }
			set_variable = { pww_do_ppref = pww_dpi_res }
			set_variable = { pww_diplo_ai_poach_to = pww_dpi_to }
			set_temp_variable = { pww_dk_target = pww_dpi_to }
			set_temp_variable = { pww_dk_act = 2 }
			set_temp_variable = { pww_dap_res = pww_dpi_res }
			pww_diplo_ai_prepare_offer = yes
			if = {
				limit = { check_variable = { pww_dap_ready = 1 } }
				set_country_flag = { flag = pww_diplo_ai_poach days = 30 value = 1 }
			}
			else = {
				set_temp_variable = { pww_dtc_t = pww_do_talk }
				set_temp_variable = { pww_dtc_status = 4 }
				if = {
					limit = {
						check_variable = { pww_dtc_t > -1 }
						check_variable = { global.pww_dt_status^pww_dtc_t = 0 }
					}
					pww_diplo_talk_close = yes
				}
			}
			set_variable = { pww_do_ppref = -1 }
			if = {
				limit = { has_global_flag = pww_diplo_debug }
				log = "[GetDateText] PWW_DIPLO poach intent [THIS.GetTag]->[?pww_dpi_to.GetTag] res=[?pww_dpi_res|0] agreement=[?pww_dpi_ag|0] incumbent=[?pww_do_p_inc.GetTag] ready=[?pww_dap_ready|0] total=[?pww_do_e_total|0] terms=[?pww_do_e_terms|0] mult=[?pww_do_p_mult|2]"
			}
		}
	}
