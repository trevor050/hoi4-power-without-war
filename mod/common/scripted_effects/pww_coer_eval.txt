###############################################################################
# Power Without War: coercion engine, weekly evaluation
#
# Every live demand is evaluated once a week, on the day where
# num_days mod 7 equals its pww_cd_dow (ids spread over the week).
#
# Pressure P = committed force (ramped) + non-force levers + target weakness.
# Resistance R = 10 + allies, patrons, alternatives, pride, regime, demand
# size, humiliation, reputation, alarm, the target's own moves.
# Net = P - R maps to the target state with hysteresis, one step per week:
#   >= t1 Strained, >= t2 Wavering, >= t3 Cornered, >= t4 Ready (going down
#   needs net below the threshold minus pww_coer_c_hyst).
# Caps:
#   Force alone tops out at Wavering: above it needs lever points
#   >= pww_coer_c_lever_min (reason force_cap shows while it binds).
#   Quiet demands cap P at pww_coer_c_quiet_cap (reason quiet_ceiling).
#   Ready only after pww_coer_c_corner_days in Cornered, and never while a
#   route to war is open and no fig leaf is on the table.
# Routes to war (only from Cornered, never on refusal or expiry):
#   desperate: an embargo, quarantine or cutoff is on and the target's
#     desperation >= 80; calculated: the target hardened, has a patron
#     pledge >= 50, a major guarantor, a faction, or an army >= 60% of the
#     claimant's. After pww_coer_c_war_days in Cornered with no fig leaf
#     offered, the target gets a war goal on the claimant and AI war
#     strategies (pww_coer_fire_war_route); the demand ends with status 4.
# Force proxies (one force per demand, see pww_coer_commit_force):
#   naval  claimant ships in the demand's approach regions (ships_in_area,
#          ships in port count in the port's land region) minus half the
#          target's ships there; 3 points per ship, max 60.
#   air    claimant planes based in the demand's air regions
#          (num_planes_stationed_in_regions), 1 point per 3 planes, max 45.
#   army   claimant divisions in its states bordering the target
#          (divisions_in_state) minus 30% of the target's divisions,
#          2.5 points per division, max 55.
#   Ramp: +700/ramp_days per week while present, -1400/ramp_days when absent
#   (teleporting a fleet in for a week does nothing). Points = raw * ramp.
###############################################################################

# Any scope. Runs once per game day from the first on_daily.
pww_coer_daily_global = {
	set_variable = { global.pww_coer_last_day = global.num_days }
	if = {
		limit = { check_variable = { global.pww_coer_live^num > 0 } }
		set_temp_variable = { pww_cdg_dow = global.num_days }
		modulo_temp_variable = { var = pww_cdg_dow value = 7 }
		clear_temp_array = pww_cdg_due
		for_each_loop = {
			array = global.pww_coer_live
			value = pww_cdg_v
			if = {
				limit = { check_variable = { global.pww_cd_dow^pww_cdg_v = pww_cdg_dow } }
				add_to_temp_array = { pww_cdg_due = pww_cdg_v }
			}
		}
		for_each_loop = {
			array = pww_cdg_due
			value = pww_cdg_v
			set_temp_variable = { pww_ce_id = pww_cdg_v }
			pww_coer_eval_demand = yes
		}
	}
}

# Any scope. In: pww_ce_id.
pww_coer_eval_demand = {
	set_temp_variable = { pww_c_id = pww_ce_id }
	set_temp_variable = { pww_ce_cl = global.pww_cd_claimant^pww_ce_id }
	set_temp_variable = { pww_ce_tg = global.pww_cd_target^pww_ce_id }
	set_temp_variable = { pww_ce_st = global.pww_cd_status^pww_ce_id }
	if = {
		limit = {
			OR = {
				var:pww_ce_cl = { exists = no }
				var:pww_ce_tg = { exists = no }
				var:pww_ce_cl = { var:pww_ce_tg = { has_war_with = PREV } }
			}
		}
		set_temp_variable = { pww_ce_end_status = 6 }
		pww_coer_end_demand = yes
	}
	else_if = {
		limit = { check_variable = { pww_ce_st = 5 } }
		if = {
			limit = { NOT = { check_variable = { global.pww_cd_prov_until^pww_ce_id > global.num_days } } }
			# The provisional ran out: a scheduled second decision
			set_variable = { global.pww_cd_status^pww_ce_id = 0 }
			set_temp_variable = { pww_ce_dl = global.num_days }
			add_to_temp_variable = { pww_ce_dl = 60 }
			set_variable = { global.pww_cd_deadline^pww_ce_id = pww_ce_dl }
			set_variable = { global.pww_cd_prov_until^pww_ce_id = 0 }
			set_variable = { global.pww_cd_framp^pww_ce_id = 0 }
			set_temp_variable = { pww_cqp_kind = 7 }
			set_temp_variable = { pww_cqp_dem = pww_ce_id }
			set_temp_variable = { pww_cqp_val = 2 }
			pww_coer_queue_push = yes
		}
	}
	else_if = {
		limit = { check_variable = { global.num_days > global.pww_cd_deadline^pww_ce_id } }
		# Decision date passed without a fold: lapses. Never a war by itself.
		set_temp_variable = { pww_ce_end_status = 2 }
		pww_coer_end_demand = yes
	}
	else = {
		pww_coer_eval_live = yes
	}
	set_variable = { global.pww_cd_last_eval^pww_ce_id = global.num_days }
}

# Internal. pww_c_id = pww_ce_id, pww_ce_cl, pww_ce_tg. One evaluation.
pww_coer_eval_live = {
	# ------------------------------------------------------------ leak
	if = {
		limit = {
			check_variable = { global.pww_cd_channel^pww_ce_id = 0 }
			check_variable = { global.pww_cd_leak_day^pww_ce_id > 0 }
			NOT = { check_variable = { global.pww_cd_leak_day^pww_ce_id > global.num_days } }
		}
		pww_coer_make_public = yes
		set_temp_variable = { pww_cqp_kind = 4 }
		set_temp_variable = { pww_cqp_dem = pww_ce_id }
		set_temp_variable = { pww_cqp_val = 1 }
		pww_coer_queue_push = yes
		# Going public: a democracy rallies, a dictatorship's pride is engaged
		set_temp_variable = { pww_cev_id = 9 }
		if = {
			limit = { var:pww_ce_tg = { has_government = democratic } }
			set_temp_variable = { pww_cev_id = 7 }
		}
		set_temp_variable = { pww_cev_side = 1 }
		pww_coer_set_evt = yes
		if = {
			limit = {
				NOT = { has_global_flag = pww_coer_no_popups }
				var:pww_ce_cl = { is_ai = no }
			}
			set_temp_variable = { pww_cpu_id = pww_ce_id }
			var:pww_ce_tg = { var:pww_ce_cl = { pww_coer_popup_prep = yes country_event = { id = pww_coer.4 } } }
		}
		if = {
			limit = { has_global_flag = pww_coer_debug }
			log = "[GetDateText] PWW_COER leak id=[?pww_ce_id] now public"
		}
	}

	# ------------------------------------------------------------ setup
	clear_temp_array = pww_ce_fid
	clear_temp_array = pww_ce_fw
	clear_temp_array = pww_ce_hid
	clear_temp_array = pww_ce_hw
	set_temp_variable = { pww_ce_p = 0 }
	set_temp_variable = { pww_ce_r = 10 }
	set_temp_variable = { pww_ce_lever = 0 }
	set_temp_variable = { pww_ca_lever = 0 }
	set_temp_variable = { pww_ca_show = 0 }
	set_temp_variable = { pww_ce_s = global.pww_cd_tstate^pww_ce_id }
	set_temp_variable = { pww_ce_ch = global.pww_cd_channel^pww_ce_id }
	set_temp_variable = { pww_ce_ty = global.pww_cd_type^pww_ce_id }
	set_temp_variable = { pww_ce_ft = global.pww_cd_ftype^pww_ce_id }
	set_temp_variable = { pww_ce_quar = global.pww_cd_quar^pww_ce_id }
	set_temp_variable = { pww_ce_fig = global.pww_cd_fig^pww_ce_id }
	set_temp_variable = { pww_ce_tfig = global.pww_cd_tfig^pww_ce_id }
	set_temp_variable = { pww_ce_age = global.num_days }
	subtract_from_temp_variable = { pww_ce_age = global.pww_cd_issued^pww_ce_id }
	set_temp_variable = { pww_ce_left = global.pww_cd_deadline^pww_ce_id }
	subtract_from_temp_variable = { pww_ce_left = global.num_days }
	set_temp_variable = { pww_ce_cred = 50 }
	var:pww_ce_cl = {
		if = {
			limit = { has_variable = pww_credibility }
			set_temp_variable = { pww_ce_cred = pww_credibility }
		}
	}
	set_temp_variable = { pww_ce_tdem = 0 }
	var:pww_ce_tg = {
		if = {
			limit = { has_government = democratic }
			set_temp_variable = { pww_ce_tdem = 1 }
		}
		set_temp_variable = { pww_ce_tstab = has_stability }
		set_temp_variable = { pww_ce_tws = has_war_support }
		set_temp_variable = { pww_ce_tdesp = pww_desp_max }
		set_temp_variable = { pww_ce_tenergy = energy_ratio }
	}
	set_temp_variable = { pww_ce_emb = 0 }
	if = {
		limit = { var:pww_ce_tg = { var:pww_ce_cl = { is_embargoing = PREV } } }
		set_temp_variable = { pww_ce_emb = 1 }
	}

	pww_coer_eval_patron = yes
	pww_coer_eval_force = yes
	pww_coer_quarantine_tick = yes
	pww_coer_join_tick = yes
	pww_coer_eval_levers = yes
	pww_coer_eval_resist = yes
	pww_coer_strangle_tick = yes
	pww_coer_eval_state = yes
	pww_coer_save_reasons = yes
	pww_coer_force_upkeep = yes

	set_variable = { global.pww_cd_prev_net^pww_ce_id = global.pww_cd_net^pww_ce_id }
	set_variable = { global.pww_cd_p^pww_ce_id = pww_ce_p }
	set_variable = { global.pww_cd_r^pww_ce_id = pww_ce_r }
	set_variable = { global.pww_cd_net^pww_ce_id = pww_ce_net }
	set_variable = { global.pww_cd_lever^pww_ce_id = pww_ce_lever }
	add_to_variable = { global.pww_coer_rev = 1 }

	if = {
		limit = { has_global_flag = pww_coer_debug }
		log = "[GetDateText] PWW_COER eval id=[?pww_ce_id] [?pww_ce_cl.GetTag]->[?pww_ce_tg.GetTag] state=[?global.pww_cd_tstate^pww_ce_id] P=[?pww_ce_p|1] R=[?pww_ce_r|1] net=[?pww_ce_net|1] lever=[?pww_ce_lever|1] force=[?global.pww_cd_force_pts^pww_ce_id|1] raw=[?global.pww_cd_fraw^pww_ce_id|1] ramp=[?global.pww_cd_framp^pww_ce_id|0] ships=[?pww_ce_ships|0]/[?pww_ce_tships|0] for=[?global.pww_cd_for0^pww_ce_id],[?global.pww_cd_for1^pww_ce_id],[?global.pww_cd_for2^pww_ce_id],[?global.pww_cd_for3^pww_ce_id] hold=[?global.pww_cd_hold0^pww_ce_id],[?global.pww_cd_hold1^pww_ce_id],[?global.pww_cd_hold2^pww_ce_id],[?global.pww_cd_hold3^pww_ce_id] route=[?global.pww_cd_route^pww_ce_id] quar=[?global.pww_cd_quar^pww_ce_id] strangle=[?global.pww_cd_strangle^pww_ce_id|0] channel=[?global.pww_cd_channel^pww_ce_id]"
	}

	# AI moves after the readout is current
	if = {
		limit = { check_variable = { global.pww_cd_status^pww_ce_id = 0 } }
		pww_coer_ai_target = yes
	}
	if = {
		limit = { check_variable = { global.pww_cd_status^pww_ce_id = 0 } }
		pww_coer_ai_claimant = yes
	}
}

# In: pww_ca_r (for-reason id), pww_ca_w (weight), pww_ca_lever (1 = a
# non-force lever). Adds to P (and lever points) and to the display list.
pww_coer_add_for = {
	if = {
		limit = { check_variable = { pww_ca_w > 0 } }
		add_to_temp_variable = { pww_ce_p = pww_ca_w }
		if = {
			limit = { check_variable = { pww_ca_lever = 1 } }
			add_to_temp_variable = { pww_ce_lever = pww_ca_w }
		}
		add_to_temp_array = { pww_ce_fid = pww_ca_r }
		add_to_temp_array = { pww_ce_fw = pww_ca_w }
	}
	set_temp_variable = { pww_ca_lever = 0 }
}

# In: pww_ca_r (holding-out reason id), pww_ca_w, pww_ca_show (1 = display
# only, not added to R).
pww_coer_add_hold = {
	if = {
		limit = { check_variable = { pww_ca_w > 0 } }
		if = {
			limit = { check_variable = { pww_ca_show = 0 } }
			add_to_temp_variable = { pww_ce_r = pww_ca_w }
		}
		add_to_temp_array = { pww_ce_hid = pww_ca_r }
		add_to_temp_array = { pww_ce_hw = pww_ca_w }
	}
	set_temp_variable = { pww_ca_show = 0 }
}

# Patron pledge (diplomacy lane), guarantors, factions. Sets pww_ce_pledge,
# pww_ce_guar, pww_ce_fac and the record's patron.
pww_coer_eval_patron = {
	set_temp_variable = { pww_ce_pledge = 0 }
	set_temp_variable = { pww_ce_guar = 0 }
	set_temp_variable = { pww_ce_fac = 0 }
	set_temp_variable = { pww_ce_pat = 0 }
	var:pww_ce_tg = {
		if = {
			limit = { has_variable = pww_patron_pledge_strength }
			set_temp_variable = { pww_ce_pledge = pww_patron_pledge_strength }
		}
		if = {
			limit = {
				has_variable = pww_patron_pledge_patron
				check_variable = { pww_ce_pledge > 0 }
			}
			set_temp_variable = { pww_ce_pat = pww_patron_pledge_patron }
		}
		random_country = {
			limit = {
				is_major = yes
				has_guaranteed = PREV
				NOT = { check_variable = { pww_ce_cl = THIS } }
			}
			set_temp_variable = { pww_ce_guar = 1 }
			if = {
				limit = { check_variable = { pww_ce_pat = 0 } }
				set_temp_variable = { pww_ce_pat = THIS }
			}
		}
		if = {
			limit = {
				is_in_faction = yes
				var:pww_ce_cl = { NOT = { is_in_faction_with = PREV } }
			}
			set_temp_variable = { pww_ce_fac = 1 }
		}
	}
	clamp_temp_variable = { var = pww_ce_pledge min = 0 max = 100 }
	# No pledge patron or guarantor: a situation demand names the
	# situation's patron seat (a target that called a patron calls it)
	if = {
		limit = {
			check_variable = { pww_ce_pat = 0 }
			has_global_flag = pww_sit_loaded
			set_temp_variable = { pww_cep_sit = global.pww_cd_sit^pww_ce_id }
			check_variable = { pww_cep_sit > -1 }
			check_variable = { global.pww_sit_has_patron^pww_cep_sit = 1 }
		}
		set_temp_variable = { pww_ce_pat = global.pww_sit_patron^pww_cep_sit }
	}
	if = {
		limit = { NOT = { check_variable = { pww_ce_pat = 0 } } }
		set_variable = { global.pww_cd_patron^pww_ce_id = pww_ce_pat }
	}
	if = {
		limit = { check_variable = { pww_ce_pledge > global.pww_cd_pledge_max^pww_ce_id } }
		set_variable = { global.pww_cd_pledge_max^pww_ce_id = pww_ce_pledge }
	}
}

# Committed force: measure, ramp, points, reasons, leak. Also measures the
# naval balance for a quarantine (pww_ce_ships, pww_ce_tships, pww_ce_eff).
pww_coer_eval_force = {
	set_temp_variable = { pww_ce_ships = 0 }
	set_temp_variable = { pww_ce_tships = 0 }
	set_temp_variable = { pww_ce_eff = 0 }
	set_temp_variable = { pww_ce_raw = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { pww_ce_ft = 1 }
				check_variable = { pww_ce_quar > 0 }
			}
		}
		var:pww_ce_cl = { pww_coer_measure_ships = yes }
		set_temp_variable = { pww_ce_ships = pww_cms_n }
		var:pww_ce_tg = { pww_coer_measure_ships = yes }
		set_temp_variable = { pww_ce_tships = pww_cms_n }
		set_temp_variable = { pww_ce_half = pww_ce_tships }
		multiply_temp_variable = { pww_ce_half = 0.5 }
		set_temp_variable = { pww_ce_eff = pww_ce_ships }
		subtract_from_temp_variable = { pww_ce_eff = pww_ce_half }
		clamp_temp_variable = { var = pww_ce_eff min = 0 max = 1000 }
	}
	if = {
		limit = { check_variable = { pww_ce_ft = 1 } }
		set_temp_variable = { pww_ce_raw = pww_ce_eff }
		multiply_temp_variable = { pww_ce_raw = 3 }
		clamp_temp_variable = { var = pww_ce_raw min = 0 max = 60 }
		set_temp_variable = { pww_ce_rdays = global.pww_coer_c_ramp_naval }
	}
	else_if = {
		limit = { check_variable = { pww_ce_ft = 2 } }
		var:pww_ce_cl = { pww_coer_measure_planes = yes }
		set_temp_variable = { pww_ce_ships = pww_cmp_n }
		set_temp_variable = { pww_ce_raw = pww_cmp_n }
		divide_temp_variable = { pww_ce_raw = 3 }
		clamp_temp_variable = { var = pww_ce_raw min = 0 max = 45 }
		set_temp_variable = { pww_ce_rdays = global.pww_coer_c_ramp_air }
	}
	else_if = {
		limit = { check_variable = { pww_ce_ft = 3 } }
		var:pww_ce_cl = { pww_coer_measure_border = yes }
		set_temp_variable = { pww_ce_ships = pww_cmb_n }
		var:pww_ce_tg = { set_temp_variable = { pww_ce_tships = num_divisions } }
		set_temp_variable = { pww_ce_half = pww_ce_tships }
		multiply_temp_variable = { pww_ce_half = 0.3 }
		set_temp_variable = { pww_ce_raw = pww_cmb_n }
		subtract_from_temp_variable = { pww_ce_raw = pww_ce_half }
		multiply_temp_variable = { pww_ce_raw = 2.5 }
		clamp_temp_variable = { var = pww_ce_raw min = 0 max = 55 }
		set_temp_variable = { pww_ce_rdays = global.pww_coer_c_ramp_army }
	}
	if = {
		limit = { check_variable = { pww_ce_ft > 0 } }
		set_temp_variable = { pww_ce_ramp = global.pww_cd_framp^pww_ce_id }
		set_temp_variable = { pww_ce_step = 700 }
		divide_temp_variable = { pww_ce_step = pww_ce_rdays }
		if = {
			limit = { check_variable = { pww_ce_raw > 0 } }
			add_to_temp_variable = { pww_ce_ramp = pww_ce_step }
			# A loud tool in use: a quiet demand will leak
			if = {
				limit = {
					check_variable = { pww_ce_ch = 0 }
					check_variable = { global.pww_cd_leak_day^pww_ce_id = 0 }
				}
				pww_coer_schedule_leak = yes
			}
		}
		else = {
			multiply_temp_variable = { pww_ce_step = 2 }
			subtract_from_temp_variable = { pww_ce_ramp = pww_ce_step }
		}
		clamp_temp_variable = { var = pww_ce_ramp min = 0 max = 100 }
		set_variable = { global.pww_cd_framp^pww_ce_id = pww_ce_ramp }
		set_variable = { global.pww_cd_fraw^pww_ce_id = pww_ce_raw }
		set_temp_variable = { pww_ce_fpts = pww_ce_raw }
		multiply_temp_variable = { pww_ce_fpts = pww_ce_ramp }
		divide_temp_variable = { pww_ce_fpts = 100 }
		set_variable = { global.pww_cd_force_pts^pww_ce_id = pww_ce_fpts }
		set_temp_variable = { pww_ca_w = pww_ce_fpts }
		# Reason by type and ramp
		if = {
			limit = { check_variable = { pww_ce_ft = 1 } }
			set_temp_variable = { pww_ca_r = 2 }
			if = {
				limit = { check_variable = { pww_ce_ramp > 99.9 } }
				set_temp_variable = { pww_ca_r = 1 }
			}
		}
		else_if = {
			limit = { check_variable = { pww_ce_ft = 2 } }
			set_temp_variable = { pww_ca_r = 4 }
			if = {
				limit = { var:pww_ce_tg = { var:pww_ce_cl = { is_neighbor_of = PREV } } }
				set_temp_variable = { pww_ca_r = 3 }
			}
		}
		else = {
			set_temp_variable = { pww_ca_r = 6 }
			if = {
				limit = { check_variable = { pww_ce_ramp > 99.9 } }
				set_temp_variable = { pww_ca_r = 5 }
			}
		}
		pww_coer_add_for = yes
		# Holding-out side of the force
		if = {
			limit = {
				check_variable = { pww_ce_raw > 0 }
				check_variable = { pww_ce_ramp < 50 }
			}
			set_temp_variable = { pww_ca_r = 17 }
			set_temp_variable = { pww_ca_w = 6 }
			set_temp_variable = { pww_ca_show = 1 }
			pww_coer_add_hold = yes
		}
		if = {
			limit = {
				check_variable = { pww_ce_ft = 1 }
				OR = {
					check_variable = { pww_ce_raw = 0 }
					check_variable = { pww_ce_half > pww_ce_eff }
				}
			}
			set_temp_variable = { pww_ca_r = 18 }
			set_temp_variable = { pww_ca_w = 6 }
			pww_coer_add_hold = yes
		}
		if = {
			limit = {
				check_variable = { pww_ce_ft = 2 }
				check_variable = { pww_ce_raw = 0 }
			}

pww_coer_eval_state = {
	set_temp_variable = { pww_ce_net = pww_ce_p }
	subtract_from_temp_variable = { pww_ce_net = pww_ce_r }
	# Desired state from thresholds
	set_temp_variable = { pww_ces_d = 0 }
	if = { limit = { check_variable = { pww_ce_net > global.pww_coer_c_t4 } } set_temp_variable = { pww_ces_d = 4 } }
	else_if = { limit = { check_variable = { pww_ce_net > global.pww_coer_c_t3 } } set_temp_variable = { pww_ces_d = 3 } }
	else_if = { limit = { check_variable = { pww_ce_net > global.pww_coer_c_t2 } } set_temp_variable = { pww_ces_d = 2 } }
	else_if = { limit = { check_variable = { pww_ce_net > global.pww_coer_c_t1 } } set_temp_variable = { pww_ces_d = 1 } }
	# Hysteresis on the way down
	if = {
		limit = { check_variable = { pww_ces_d < pww_ce_s } }
		set_temp_variable = { pww_ces_h = pww_ce_net }
		add_to_temp_variable = { pww_ces_h = global.pww_coer_c_hyst }
		set_temp_variable = { pww_ces_d2 = 0 }
		if = { limit = { check_variable = { pww_ces_h > global.pww_coer_c_t4 } } set_temp_variable = { pww_ces_d2 = 4 } }
		else_if = { limit = { check_variable = { pww_ces_h > global.pww_coer_c_t3 } } set_temp_variable = { pww_ces_d2 = 3 } }
		else_if = { limit = { check_variable = { pww_ces_h > global.pww_coer_c_t2 } } set_temp_variable = { pww_ces_d2 = 2 } }
		else_if = { limit = { check_variable = { pww_ces_h > global.pww_coer_c_t1 } } set_temp_variable = { pww_ces_d2 = 1 } }
		if = {
			limit = { NOT = { check_variable = { pww_ces_d2 < pww_ce_s } } }
			set_temp_variable = { pww_ces_d = pww_ce_s }
		}
	}
	# Force alone tops out at Wavering
	if = {
		limit = {
			check_variable = { pww_ces_d > 2 }
			check_variable = { pww_ce_lever < global.pww_coer_c_lever_min }
		}
		set_temp_variable = { pww_ces_d = 2 }
		set_temp_variable = { pww_ca_r = 16 }
		set_temp_variable = { pww_ca_w = 30 }
		set_temp_variable = { pww_ca_show = 1 }
		pww_coer_add_hold = yes
		if = {
			limit = { has_global_flag = pww_coer_debug }
			log = "[GetDateText] PWW_COER force_cap id=[?pww_ce_id] net=[?pww_ce_net|1] lever=[?pww_ce_lever|1]"
		}
	}
	# One step per evaluation (always through Cornered)
	set_temp_variable = { pww_ces_up = pww_ce_s }
	add_to_temp_variable = { pww_ces_up = 1 }
	set_temp_variable = { pww_ces_dn = pww_ce_s }
	subtract_from_temp_variable = { pww_ces_dn = 1 }
	if = { limit = { check_variable = { pww_ces_d > pww_ces_up } } set_temp_variable = { pww_ces_d = pww_ces_up } }
	if = { limit = { check_variable = { pww_ces_d < pww_ces_dn } } set_temp_variable = { pww_ces_d = pww_ces_dn } }

	# Routes to war (computed from Wavering up)
	set_temp_variable = { pww_ces_route = 0 }
	if = {
		limit = { check_variable = { pww_ce_s > 1 } }
		pww_coer_eval_route = yes
	}
	set_variable = { global.pww_cd_route^pww_ce_id = pww_ces_route }

	# Cornered -> Ready: time in Cornered, and an exit when a route is open
	set_temp_variable = { pww_ces_cdays = 0 }
	if = {
		limit = { check_variable = { global.pww_cd_corner_day^pww_ce_id > 0 } }
		set_temp_variable = { pww_ces_cdays = global.num_days }
		subtract_from_temp_variable = { pww_ces_cdays = global.pww_cd_corner_day^pww_ce_id }
	}
	set_temp_variable = { pww_ces_exit = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { pww_ce_fig > 0 }
				check_variable = { pww_ce_tfig = 1 }
			}
		}
		set_temp_variable = { pww_ces_exit = 1 }
	}
	if = {
		limit = {
			check_variable = { pww_ce_s = 3 }
			check_variable = { pww_ces_d = 4 }
			OR = {
				check_variable = { pww_ces_cdays < global.pww_coer_c_corner_days }
				AND = {
					check_variable = { pww_ces_route > 0 }
					check_variable = { pww_ces_exit = 0 }
				}
			}
		}
		set_temp_variable = { pww_ces_d = 3 }
	}

	# Warning and intel lines
	pww_coer_eval_warnings = yes

	# Apply a shift
	if = {
		limit = { NOT = { check_variable = { pww_ces_d = pww_ce_s } } }
		pww_coer_apply_shift = yes
	}

	# War route fires after war_days in Cornered with no exit on the table
	if = {
		limit = {
			check_variable = { pww_ces_d = 3 }
			check_variable = { pww_ces_route > 0 }
			check_variable = { pww_ces_exit = 0 }
			check_variable = { global.pww_cd_corner_day^pww_ce_id > 0 }
			NOT = { check_variable = { pww_ces_cdays < global.pww_coer_c_war_days } }
		}
		pww_coer_fire_war_route = yes
	}
}

# pww_ce_s -> pww_ces_d with its cause.
pww_coer_apply_shift = {
	set_temp_variable = { pww_cas_side = 0 }
	if = {
		limit = { check_variable = { pww_ces_d < pww_ce_s } }
		set_temp_variable = { pww_cas_side = 1 }
	}
	# Cause: the latest move in the same direction (last 14 days), else the
	# strongest reason on the side that moved it.
	set_temp_variable = { pww_cas_cause = 0 }
	set_temp_variable = { pww_cas_age = global.num_days }
	subtract_from_temp_variable = { pww_cas_age = global.pww_cd_evt_day^pww_ce_id }
	if = {
		limit = {
			check_variable = { global.pww_cd_evt^pww_ce_id > 0 }
			check_variable = { pww_cas_age < 15 }
			check_variable = { global.pww_cd_evt_side^pww_ce_id = pww_cas_side }
		}
		set_temp_variable = { pww_cas_cause = global.pww_cd_evt^pww_ce_id }
		set_variable = { global.pww_cd_evt^pww_ce_id = 0 }
	}
	else_if = {
		limit = {
			check_variable = { pww_cas_side = 0 }
			check_variable = { pww_ce_fid^num > 0 }
		}
		find_highest_in_array = { array = pww_ce_fw value = pww_cas_v index = pww_cas_i }
		set_temp_variable = { pww_cas_cause = pww_ce_fid^pww_cas_i }
	}
	else_if = {
		limit = {
			check_variable = { pww_cas_side = 1 }
			check_variable = { pww_ce_hid^num > 0 }
		}
		find_highest_in_array = { array = pww_ce_hw value = pww_cas_v index = pww_cas_i }
		set_temp_variable = { pww_cas_cause = pww_ce_hid^pww_cas_i }
	}
	set_variable = { global.pww_cd_prev^pww_ce_id = pww_ce_s }
	set_variable = { global.pww_cd_tstate^pww_ce_id = pww_ces_d }
	set_variable = { global.pww_cd_cause^pww_ce_id = pww_cas_cause }
	set_variable = { global.pww_cd_cause_side^pww_ce_id = pww_cas_side }
	set_variable = { global.pww_cd_shift_day^pww_ce_id = global.num_days }
	set_temp_variable = { pww_cqp_kind = 2 }
	set_temp_variable = { pww_cqp_dem = pww_ce_id }
	set_temp_variable = { pww_cqp_val = pww_ces_d }
	pww_coer_queue_push = yes
	if = {
		limit = { has_global_flag = pww_coer_debug }
		log = "[GetDateText] PWW_COER shift id=[?pww_ce_id] [?pww_ce_s] -> [?pww_ces_d] cause=[?pww_cas_cause] side=[?pww_cas_side]"
	}
	# Cornered bookkeeping
	if = {
		limit = { check_variable = { pww_ces_d = 3 } check_variable = { pww_ce_s < 3 } }
		set_variable = { global.pww_cd_corner_day^pww_ce_id = global.num_days }
		set_temp_variable = { pww_cqp_kind = 5 }
		set_temp_variable = { pww_cqp_dem = pww_ce_id }
		set_temp_variable = { pww_cqp_val = pww_ces_route }
		pww_coer_queue_push = yes
		if = {
			limit = {
				NOT = { has_global_flag = pww_coer_no_popups }
				var:pww_ce_cl = { is_ai = no }
			}
			set_temp_variable = { pww_cpu_id = pww_ce_id }
			var:pww_ce_tg = { var:pww_ce_cl = { pww_coer_popup_prep = yes country_event = { id = pww_coer.3 } } }
		}
	}
	if = {
		limit = { check_variable = { pww_ces_d < 3 } }
		set_variable = { global.pww_cd_corner_day^pww_ce_id = 0 }
	}
	pww_coer_notify_sit = yes
}

# The target, cornered with a route open and no exit offered, turns to war:
# a war goal on the claimant plus AI war strategies. No declaration is
# scripted; a human target decides for itself.
pww_coer_fire_war_route = {
	set_variable = { global.pww_cd_route^pww_ce_id = pww_ces_route }
	var:pww_ce_cl = {
		var:pww_ce_tg = {
			if = {
				limit = { NOT = { has_wargoal_against = PREV } }
				create_wargoal = { type = topple_government target = PREV expire = 0 }
			}
			add_ai_strategy = { type = prepare_for_war id = PREV value = 200 }
			add_ai_strategy = { type = antagonize id = PREV value = 100 }
			add_ai_strategy = { type = declare_war id = PREV value = 100 }
		}
	}
	set_temp_variable = { pww_cqp_kind = 6 }
	set_temp_variable = { pww_cqp_dem = pww_ce_id }
	set_temp_variable = { pww_cqp_val = pww_ces_route }
	pww_coer_queue_push = yes
	if = {
		limit = { has_global_flag = pww_coer_debug }
		log = "[GetDateText] PWW_COER war_route id=[?pww_ce_id] route=[?pww_ces_route]"
	}
	set_temp_variable = { pww_ce_end_status = 4 }
	pww_coer_end_demand = yes
}

# Writes the top four reasons per side into the record.
pww_coer_save_reasons = {
	set_temp_variable = { pww_csr_n = 0 }
	for_loop_effect = {
		start = 0
		end = 4
		value = pww_csr_k
		set_temp_variable = { pww_csr_id = 0 }
		if = {
			limit = { check_variable = { pww_ce_fw^num > 0 } }
			find_highest_in_array = { array = pww_ce_fw value = pww_csr_v index = pww_csr_i }
			if = {
				limit = { check_variable = { pww_csr_v > 0 } }
				set_temp_variable = { pww_csr_id = pww_ce_fid^pww_csr_i }
				set_temp_variable = { pww_ce_fw^pww_csr_i = -1 }
				add_to_temp_variable = { pww_csr_n = 1 }
			}
		}
		if = { limit = { check_variable = { pww_csr_k = 0 } } set_variable = { global.pww_cd_for0^pww_ce_id = pww_csr_id } }
		else_if = { limit = { check_variable = { pww_csr_k = 1 } } set_variable = { global.pww_cd_for1^pww_ce_id = pww_csr_id } }
		else_if = { limit = { check_variable = { pww_csr_k = 2 } } set_variable = { global.pww_cd_for2^pww_ce_id = pww_csr_id } }
		else = { set_variable = { global.pww_cd_for3^pww_ce_id = pww_csr_id } }
	}
	set_temp_variable = { pww_csr_total = pww_ce_fid^num }
	set_variable = { global.pww_cd_nfor^pww_ce_id = pww_csr_total }
	set_temp_variable = { pww_csr_n = 0 }
	for_loop_effect = {
		start = 0
		end = 4
		value = pww_csr_k
		set_temp_variable = { pww_csr_id = 0 }
		if = {
			limit = { check_variable = { pww_ce_hw^num > 0 } }
			find_highest_in_array = { array = pww_ce_hw value = pww_csr_v index = pww_csr_i }
			if = {
				limit = { check_variable = { pww_csr_v > 0 } }
				set_temp_variable = { pww_csr_id = pww_ce_hid^pww_csr_i }
				set_temp_variable = { pww_ce_hw^pww_csr_i = -1 }
				add_to_temp_variable = { pww_csr_n = 1 }
			}
		}
		if = { limit = { check_variable = { pww_csr_k = 0 } } set_variable = { global.pww_cd_hold0^pww_ce_id = pww_csr_id } }
		else_if = { limit = { check_variable = { pww_csr_k = 1 } } set_variable = { global.pww_cd_hold1^pww_ce_id = pww_csr_id } }
		else_if = { limit = { check_variable = { pww_csr_k = 2 } } set_variable = { global.pww_cd_hold2^pww_ce_id = pww_csr_id } }
		else = { set_variable = { global.pww_cd_hold3^pww_ce_id = pww_csr_id } }
	}
	set_temp_variable = { pww_csr_total = pww_ce_hid^num }
	set_variable = { global.pww_cd_nhold^pww_ce_id = pww_csr_total }
}

# Weekly upkeep of the committed force (political power, command power,
# fuel). Charged while committed, present or not.
pww_coer_force_upkeep = {
	if = {
		limit = { check_variable = { global.pww_cd_ftype^pww_ce_id > 0 } }
		var:pww_ce_cl = {
			add_political_power = -1
			if = {
				limit = { check_variable = { pww_ce_ft = 3 } }
				add_command_power = -3
			}
			else = {
				add_command_power = -2
			}
			if = {
				limit = { check_variable = { pww_ce_ft = 1 } }
				if = {
					limit = { check_variable = { pww_ce_ships > 19 } }
					add_fuel = -1000
				}
				else_if = {
					limit = { check_variable = { pww_ce_ships > 9 } }
					add_fuel = -500
				}
				else_if = {
					limit = { check_variable = { pww_ce_ships > 0 } }
					add_fuel = -250
				}
			}
			else_if = {
				limit = {
					check_variable = { pww_ce_ft = 2 }
					check_variable = { pww_ce_ships > 0 }
				}
				add_fuel = -300
			}
		}
	}
}

# The AI target answers the pressure it is under.
pww_coer_ai_target = {
	if = {
		limit = {
			var:pww_ce_tg = {
				is_ai = yes
				NOT = { has_country_flag = pww_coer_no_ai_response }
			}
		}
		set_temp_variable = { pww_cat_s = global.pww_cd_tstate^pww_ce_id }
		set_temp_variable = { pww_in_demand_id = pww_ce_id }
		set_temp_variable = { pww_in_response = -1 }
		if = {
			limit = { check_variable = { pww_cat_s = 4 } }
			set_temp_variable = { pww_in_response = 0 }
		}
		else_if = {
			limit = { check_variable = { pww_cat_s = 3 } }
			if = {
				limit = {
					check_variable = { pww_ces_route > 0 }
					check_variable = { pww_ces_exit = 0 }
					check_variable = { global.pww_cd_hardened^pww_ce_id = 0 }
				}
				set_temp_variable = { pww_in_response = 5 }
			}
			else_if = {
				limit = {
					check_variable = { pww_ces_route = 0 }
					check_variable = { global.pww_cd_lot_done^pww_ce_id = 0 }
				}
				set_temp_variable = { pww_in_response = 1 }
			}
			else_if = {
				limit = {
					check_variable = { pww_ces_route = 0 }
					check_variable = { global.pww_cd_tfig_done^pww_ce_id = 0 }
					check_variable = { pww_ce_fig = 0 }
				}
				set_temp_variable = { pww_in_response = 3 }
				set_temp_variable = { pww_in_fig_leaf = 2 }
			}
		}
		else_if = {
			limit = { check_variable = { pww_cat_s = 2 } }
			if = {
				limit = {
					check_variable = { global.pww_cd_patron_until^pww_ce_id = 0 }
					OR = {
						check_variable = { pww_ce_pledge > 0 }
						check_variable = { pww_ce_guar = 1 }
						check_variable = { pww_ce_fac = 1 }
					}
				}
				set_temp_variable = { pww_in_response = 4 }
			}
			else_if = {
				limit = {
					check_variable = { pww_ce_ch = 0 }
					check_variable = { global.pww_cd_stall^pww_ce_id = 0 }
				}
				set_temp_variable = { pww_in_response = 7 }
			}
		}
		else_if = {
			limit = {
				check_variable = { pww_ce_ch = 1 }
				check_variable = { pww_ce_tdem = 0 }
				check_variable = { pww_ce_tstab > 0.5 }
				check_variable = { pww_ce_age > 13 }
				check_variable = { global.pww_cd_hardened^pww_ce_id = 0 }
			}
			set_temp_variable = { pww_in_response = 5 }
		}
		if = {
			limit = { check_variable = { pww_in_response > -1 } }
			pww_coer_target_respond = yes
		}
	}
}

# The AI claimant escalates along the ladder.
pww_coer_ai_claimant = {
	if = {
		limit = {
			var:pww_ce_cl = {
				is_ai = yes
				NOT = { has_country_flag = pww_coer_no_ai_escalation }
			}
		}
		set_temp_variable = { pww_in_demand_id = pww_ce_id }
		set_temp_variable = { pww_cac_s = global.pww_cd_tstate^pww_ce_id }
		# A committed force that registers nothing after four weeks (an AI
		# fleet stays in its home ports in peacetime) is released: it only
		# costs upkeep. The AI does not commit again for 120 days.
		if = {
			limit = {
				check_variable = { global.pww_cd_ftype^pww_ce_id > 0 }
				NOT = { check_variable = { global.pww_cd_fraw^pww_ce_id > 0 } }
				set_temp_variable = { pww_cac_fa = global.num_days }
				subtract_from_temp_variable = { pww_cac_fa = global.pww_cd_fsince^pww_ce_id }
				check_variable = { pww_cac_fa > 27 }
			}
			pww_coer_release_force = yes
			var:pww_ce_cl = { set_country_flag = { flag = pww_coer_ai_force_idle days = 120 value = 1 } }
			if = {
				limit = { has_global_flag = pww_coer_debug }
				log = "[GetDateText] PWW_COER ai releases idle force id=[?pww_ce_id]"
			}
		}
		# Commit a force after two weeks
		if = {
			limit = {
				check_variable = { global.pww_cd_ftype^pww_ce_id = 0 }
				check_variable = { pww_ce_age > 13 }
				var:pww_ce_cl = { NOT = { has_country_flag = pww_coer_ai_force_idle } }
			}
			set_temp_variable = { pww_in_force_type = 0 }
			if = {
				limit = {
					check_variable = { global.pww_cd_sea0^pww_ce_id > 0 }
					var:pww_ce_cl = { check_variable = { num_ships > 9 } }
				}
				set_temp_variable = { pww_in_force_type = 1 }
			}
			else_if = {
				limit = { var:pww_ce_tg = { var:pww_ce_cl = { is_neighbor_of = PREV } } }
				set_temp_variable = { pww_in_force_type = 3 }
			}
			else_if = {
				limit = { var:pww_ce_cl = { check_variable = { num_deployed_planes > 49 } } }
				set_temp_variable = { pww_in_force_type = 2 }
			}
			if = {
				limit = { check_variable = { pww_in_force_type > 0 } }
				pww_coer_commit_force = yes
			}
		}
		# Stuck at Wavering or below without a lever, or a desperate claimant
		# stuck at Defiant/Strained: cutoff, else embargo
		if = {
			limit = {
				check_variable = { pww_ce_age > 41 }
				check_variable = { pww_cac_s < 3 }
				OR = {
					check_variable = { pww_ce_lever < global.pww_coer_c_lever_min }
					AND = {
						check_variable = { pww_cac_s < 2 }
						var:pww_ce_cl = { check_variable = { pww_desp_max > 69.9 } }
					}
				}
			}
			set_temp_variable = { pww_in_cut_mode = 0 }
			set_temp_variable = { pww_in_res = -1 }
			pww_coer_cutoff = yes
			if = {
				limit = {
					check_variable = { pww_out_result = 0 }
					check_variable = { pww_ce_emb = 0 }
				}
				pww_coer_embargo = yes
			}
		}
		# Quarantine once the squadron holds the approaches
		if = {
			limit = {
				check_variable = { pww_ce_age > 69 }
				check_variable = { pww_cac_s < 3 }
				check_variable = { global.pww_cd_quar^pww_ce_id = 0 }
				check_variable = { global.pww_cd_ftype^pww_ce_id = 1 }
				check_variable = { pww_ce_eff > 4.9 }
			}
			pww_coer_declare_quarantine = yes
		}
		# Cornered with a war route open: take the face-saving exit
		if = {
			limit = {
				check_variable = { pww_cac_s = 3 }
				check_variable = { pww_ces_route > 0 }
				check_variable = { pww_ce_fig = 0 }
			}
			set_temp_variable = { pww_in_response = 2 }
			set_temp_variable = { pww_in_fig_leaf = 4 }
			pww_coer_claimant_respond = yes
		}
	}
}
