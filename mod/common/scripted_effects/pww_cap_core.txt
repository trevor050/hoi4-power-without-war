###############################################################################
# Power Without War: production capacity (supply that answers demand), core
#
# Spec: docs/design/2026-11-26-production-capacity.md. Per-resource blocks are
# GENERATED (tools/pww_cap_gen.py -> pww_cap_res.txt,
# pww_cap_dynamic_modifiers.txt, pww_cap_l_english.yml). This file and
# on_actions/pww_cap_on_actions.txt are hand-written.
#
# MODEL (all seven resources)
#   Per country and resource an output factor pww_cap_<res> (1 = vanilla,
#   c_floor 0.5 .. c_ceiling 1.3), applied by the dynamic modifier
#   pww_cap_output as local_resources_<res>_factor = (factor - 1) x (1 +
#   local_resources_factor) (the per-resource factor adds to the technology
#   factor; c_comp = 0 drops the scaling).
#   Tick: monthly per country on core's bucket day (pww_bucket =
#   global.pww_day_mod30), at least 25 days apart.
#   Engine facts: resource_produced is net of costs and includes inflows;
#   resource_exported is the pool ON OFFER (export share x produced), not
#   sales (world chromium 1936: 1,057 offered, 16 imported); resource@ =
#   produced - consumed - pool + imports.
#   World market (every c_world_days): global.pww_cap_rho_<res> = world
#   imports / world pool (the share of the open market buyers take).
#   Signals per resource, from extraction (produced + own costs - own
#   inflows), at least c_min_out a day:
#     natural balance = resource@ + (1 - e) x (cartel hold + storage cost -
#                       reserve draw - dump), e = pool / produced
#     unsold = natural balance - storage intake (pww_stock_in_<res>)
#     spare  = unsold - c_margin (15%) x extraction   -> pww_cap_spare_<res>
#     short  = natural balance less vanilla imports and contract purchases
#              (x (1 - e)) < -max(c_short_min, c_short_share x consumed)
#     world floor = max(c_floor, c_rho_mult x rho), at most 1: nobody idles
#              below twice the share of the pool the world buys, so vanilla
#              buyers keep their imports (seller-level sales are not
#              observable)
#     demand = rho > c_pool_full (tight world market), contract buyers are
#              taking the working stock (unsold under c_low_stock x margin
#              with contract sales), a buyer turned away within
#              c_turned_days (country var pww_cap_turned_<res> = num_days,
#              optional hook, nobody writes it yet), oil: fuel under 50%,
#              test hook pww_cap_test_demand_<res>
#     sold out = no spare and (demand or short)
#   Idle (factor - c_idle, 2%/month, to the world floor) when spare persists
#     beyond a deadband (c_idle_band, 5% of extraction: a seller at balance
#     neither idles nor expands)
#     AND storage takes nothing more (reserve full or not filling; oil: fuel
#     tanks over 95%) AND the system allows it.
#   Below 1: reopen c_reopen (4%/month, any system) when sold out or below
#     the world floor. At 1 or more, sold out: invest to c_ceiling: market
#     c_invest_market (1%) for any buyer, planned c_invest_planned (1%) and
#     command c_invest_command (1.5%) only when the country itself is short.
#   Economic system pww_cap_sys, re-read every tick from the ruling party:
#     0 market (democracies, non-aligned), 1 planned (communism: never
#     idles), 2 command (fascism: never idles a resource it uses, i.e.
#     consumed > c_mil_use, or is short of; idles others like a market).
#     Test hook: country var pww_cap_test_sys overrides it.
#   Cartels: the member's hold (pww_cartel_hold_<res>) counts as unsold, so
#     withheld output idles over time; the cartel's integrator then lowers the
#     hold as output falls, and under this lane the cartel keeps no fallback
#     pool (pww_cartel_quota.txt). The quota is unchanged.
#
# PUBLIC INTERFACE (country scope)
#   global flag pww_cap_loaded
#   vars pww_cap_<res>          output factor (1 = vanilla), 0.5..1.3
#        pww_cap_trend_<res>    -1 idling, 0 steady, 1 reopening, 2 expanding
#                               (this month's step)
#        pww_cap_spare_<res>    spare units a day at the last tick (unsold
#                               beyond the margin; negative = inside it)
#        pww_cap_sys            0 market, 1 planned, 2 command
#        pww_cap_last_tick      num_days of the last tick
#   input pww_cap_turned_<res>  num_days a buyer was refused for lack of
#                               supply (optional hook for other lanes)
#   effect pww_cap_reset = yes  every factor back to 1 (tests)
#   loc  pww_cap_sys_<0..2>, pww_cap_trend_<n1|0|1|2>
#   DEBUG global flag pww_cap_debug logs PWW_CAP lines.
#
# TEMP VARIABLES: pww_cpk_* (internal).
###############################################################################

# Any scope. Idempotent. Tunables are written when the version changes, so
# tests can override them after init and saves pick up a new version.
pww_cap_init = {
	set_global_flag = pww_cap_loaded
	# Day-1 readings are not settled (consumption is not computed yet): the
	# scheduled tick waits c_start_days after the lane first loads.
	if = {
		limit = { NOT = { has_variable = global.pww_cap_first_day } }
		set_variable = { global.pww_cap_first_day = global.num_days }
	}
	if = {
		limit = { NOT = { check_variable = { global.pww_cap_version = 4 } } }
		set_variable = { global.pww_cap_version = 4 }
		set_variable = { global.pww_cap_c_margin = 0.15 }
		set_variable = { global.pww_cap_c_idle = 0.02 }
		set_variable = { global.pww_cap_c_idle_band = 0.05 }
		set_variable = { global.pww_cap_c_reopen = 0.04 }
		set_variable = { global.pww_cap_c_invest_market = 0.01 }
		set_variable = { global.pww_cap_c_invest_planned = 0.01 }
		set_variable = { global.pww_cap_c_invest_command = 0.015 }
		set_variable = { global.pww_cap_c_floor = 0.5 }
		set_variable = { global.pww_cap_c_ceiling = 1.3 }
		set_variable = { global.pww_cap_c_min_out = 2 }
		set_variable = { global.pww_cap_c_short_share = 0.02 }
		set_variable = { global.pww_cap_c_short_min = 0.5 }
		set_variable = { global.pww_cap_c_pool_full = 0.8 }
		set_variable = { global.pww_cap_c_rho_mult = 2 }
		set_variable = { global.pww_cap_c_world_days = 10 }
		set_variable = { global.pww_cap_c_low_stock = 0.5 }
		set_variable = { global.pww_cap_c_turned_days = 45 }
		set_variable = { global.pww_cap_c_mil_use = 0.5 }
		set_variable = { global.pww_cap_c_comp = 1 }
		set_variable = { global.pww_cap_c_period = 25 }
		set_variable = { global.pww_cap_c_start_days = 14 }
		set_variable = { global.pww_cap_c_on = 1 }
	}
}

# Country scope. on_daily entry point.
pww_cap_on_daily = {
	if = {
		limit = { NOT = { has_global_flag = pww_cap_loaded } }
		pww_cap_init = yes
	}
	if = {
		limit = { has_global_flag = pww_initialized }
		set_temp_variable = { pww_cpk_since = global.num_days }
		subtract_from_temp_variable = { pww_cpk_since = global.pww_cap_world_day }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = global.pww_cap_world_day }
					check_variable = { pww_cpk_since > global.pww_cap_c_world_days }
				}
			}
			pww_cap_world_refresh = yes
		}
	}
	if = {
		limit = {
			has_global_flag = pww_initialized
			check_variable = { global.pww_cap_c_on = 1 }
			has_variable = pww_bucket
			check_variable = { pww_bucket = global.pww_day_mod30 }
			num_of_controlled_states > 0
		}
		set_temp_variable = { pww_cpk_age = global.num_days }
		subtract_from_temp_variable = { pww_cpk_age = global.pww_cap_first_day }
		set_temp_variable = { pww_cpk_since = global.num_days }
		subtract_from_temp_variable = { pww_cpk_since = pww_cap_last_tick }
		if = {
			limit = {
				check_variable = { pww_cpk_age > global.pww_cap_c_start_days }
				OR = {
					NOT = { has_variable = pww_cap_last_tick }
					check_variable = { pww_cpk_since > global.pww_cap_c_period }
				}
			}
			pww_cap_country_tick = yes
		}
	}
}

# Country scope. The monthly tick (also callable from tests).
pww_cap_country_tick = {
	set_variable = { pww_cap_last_tick = global.num_days }
	if = {
		limit = { NOT = { has_variable = global.pww_cap_world_day } }
		pww_cap_world_refresh = yes
	}
	pww_cap_read_system = yes
	set_temp_variable = { pww_cpk_lrf = 1 }
	if = {
		limit = { check_variable = { global.pww_cap_c_comp = 1 } }
		set_temp_variable = { pww_cpk_t = modifier@local_resources_factor }
		add_to_temp_variable = { pww_cpk_lrf = pww_cpk_t }
		clamp_temp_variable = { var = pww_cpk_lrf min = 0.1 max = 10 }
	}
	pww_cap_tick_all = yes
	# The modifier appears once some output departs from vanilla, and stays.
	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = pww_cap_output } }
			OR = {
				NOT = { check_variable = { pww_cap_m_oil = 0 } }
				NOT = { check_variable = { pww_cap_m_aluminium = 0 } }
				NOT = { check_variable = { pww_cap_m_rubber = 0 } }
				NOT = { check_variable = { pww_cap_m_tungsten = 0 } }
				NOT = { check_variable = { pww_cap_m_steel = 0 } }
				NOT = { check_variable = { pww_cap_m_chromium = 0 } }
				NOT = { check_variable = { pww_cap_m_coal = 0 } }
			}
		}
		add_dynamic_modifier = { modifier = pww_cap_output }
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = pww_cap_output } }
		force_update_dynamic_modifier = yes
	}
	if = {
		limit = { has_global_flag = pww_cap_debug }
		log = "[GetDateText] PWW_CAP [This.GetTag] tick sys=[?pww_cap_sys|0] lrf=[?pww_cpk_lrf|2] oil=[?pww_cap_oil|2] alu=[?pww_cap_aluminium|2] rub=[?pww_cap_rubber|2] tun=[?pww_cap_tungsten|2] ste=[?pww_cap_steel|2] chr=[?pww_cap_chromium|2] coa=[?pww_cap_coal|2]"
	}
}

# Country scope. pww_cap_sys from the ruling party (test hook first).
pww_cap_read_system = {
	if = {
		limit = { has_variable = pww_cap_test_sys }
		set_variable = { pww_cap_sys = pww_cap_test_sys }
	}
	else_if = {
		limit = { has_government = communism }
		set_variable = { pww_cap_sys = 1 }
	}
	else_if = {
		limit = { has_government = fascism }
		set_variable = { pww_cap_sys = 2 }
	}
	else = {
		set_variable = { pww_cap_sys = 0 }
	}
}

# Country scope. Public: every factor back to vanilla now (tests).
pww_cap_reset = {
	pww_cap_reset_all = yes
	if = {
		limit = { has_dynamic_modifier = { modifier = pww_cap_output } }
		force_update_dynamic_modifier = yes
	}
}
