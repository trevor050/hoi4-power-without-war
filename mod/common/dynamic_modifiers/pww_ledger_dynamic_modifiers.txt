###############################################################################
# Power Without War: agreements ledger dynamic modifier
#
# One modifier per country carries the sum of every active agreement's flows.
# Values are country variables written by pww_ledger_recompute_flows
# (common/scripted_effects/pww_ledger_flows.txt). Pattern follows vanilla
# INS_great_depression (TAOG_dynamic_modifiers.txt) and
# GER_industrial_relocation_resources (wuw_dynamic_modifiers.txt).
#
#   pww_lf_in_<res>    resources received (buyer side)
#   pww_lf_out_<res>   resources delivered (seller side, consumption)
#   pww_lf_civ_use     civilian factories paid out (payer side)
#   pww_lf_pp          net political power per day (credit minus debit)
#
# Civilian factories received are NOT in this modifier: they are offsite
# buildings (pww_ledger_apply_civ_receipt).
###############################################################################

pww_ledger_flows = {
	enable = { always = yes }
	icon = GFX_idea_export_focus

	country_resource_oil = pww_lf_in_oil
	country_resource_aluminium = pww_lf_in_aluminium
	country_resource_rubber = pww_lf_in_rubber
	country_resource_tungsten = pww_lf_in_tungsten
	country_resource_steel = pww_lf_in_steel
	country_resource_chromium = pww_lf_in_chromium
	country_resource_coal = pww_lf_in_coal

	country_resource_cost_oil = pww_lf_out_oil
	country_resource_cost_aluminium = pww_lf_out_aluminium
	country_resource_cost_rubber = pww_lf_out_rubber
	country_resource_cost_tungsten = pww_lf_out_tungsten
	country_resource_cost_steel = pww_lf_out_steel
	country_resource_cost_chromium = pww_lf_out_chromium
	country_resource_cost_coal = pww_lf_out_coal

	civilian_factory_use = pww_lf_civ_use
	political_power_gain = pww_lf_pp
}
