# Diplomacy test helpers: human-side trade-talks bundles (GER is the player)
# scored by the same evaluation the target's ai_acceptance reads (pww_do_e_*
# on the sender; the engine sums exactly those parts and accepts above zero).
#
# Test hooks: the hidden reserve and the frozen relations of a talk are
# overwritten (global.pww_dt_reserve / pww_dt_rel) where a case needs a
# controlled value; every other part comes from the live game.

# GER scope. in: pww_dtest_hg_tgt (country), pww_dtest_hg_res, pww_dtest_hg_dir
# (0 we buy, 1 we sell), pww_dtest_hg_price (0 30% below market, 1 at market,
# 2 35% above), optional pww_dtest_hg_units. Opens (or resumes) talks and
# tables a one-year fixed bundle: 30% of the seller's output of the resource
# (or the given units) against whole civilian
# factories worth price x market (rounded down below market, up above it,
# to the nearest at market; at least 1). The evaluation sees the real count.
pww_dtest_hg_draft = {
	set_temp_variable = { pww_dk_target = pww_dtest_hg_tgt }
	set_temp_variable = { pww_dk_act = 0 }
	pww_diplo_offer_begin = yes
	pww_diplo_bundle_clear = yes
	set_variable = { pww_do_len = 1 }
	set_variable = { pww_do_bind = 1 }
	set_temp_variable = { pww_dg_res = pww_dtest_hg_res }
	set_temp_variable = { pww_dba_x = pww_dtest_hg_res }
	set_temp_variable = { pww_dba_kind = 3 }
	if = {
		limit = { check_variable = { pww_dtest_hg_dir = 0 } }
		var:pww_dtest_hg_tgt = { pww_diplo_read_res = yes }
		set_temp_variable = { pww_dba_side = 1 }
	}
	else = {
		pww_diplo_read_res = yes
		set_temp_variable = { pww_dba_side = 0 }
	}
	set_temp_variable = { pww_dtest_hg_u = pww_dg_prod }
	multiply_temp_variable = { pww_dtest_hg_u = 0.3 }
	round_temp_variable = pww_dtest_hg_u
	# Optional fixed quantity (pww_dtest_hg_units > 0)
	if = {
		limit = { check_variable = { pww_dtest_hg_units > 0 } }
		set_temp_variable = { pww_dtest_hg_u = pww_dtest_hg_units }
	}
	if = {
		limit = { check_variable = { pww_dtest_hg_u < 1 } }
		set_temp_variable = { pww_dtest_hg_u = 1 }
	}
	pww_diplo_bundle_add = yes
	set_variable = { pww_dob_a^pww_dba_new = pww_dtest_hg_u }
	# Payment
	set_temp_variable = { pww_dtest_hg_pf = 1 }
	if = {
		limit = { check_variable = { pww_dtest_hg_price = 0 } }
		set_temp_variable = { pww_dtest_hg_pf = 0.7 }
	}
	else_if = {
		limit = { check_variable = { pww_dtest_hg_price = 2 } }
		set_temp_variable = { pww_dtest_hg_pf = 1.35 }
	}
	set_temp_variable = { pww_dtest_hg_civ = pww_dtest_hg_u }
	divide_temp_variable = { pww_dtest_hg_civ = 8 }
	multiply_temp_variable = { pww_dtest_hg_civ = pww_dtest_hg_pf }
	if = {
		limit = { check_variable = { pww_dtest_hg_price = 0 } }
		subtract_from_temp_variable = { pww_dtest_hg_civ = 0.499 }
	}
	else_if = {
		limit = { check_variable = { pww_dtest_hg_price = 2 } }
		add_to_temp_variable = { pww_dtest_hg_civ = 0.499 }
	}
	round_temp_variable = pww_dtest_hg_civ
	if = {
		limit = { check_variable = { pww_dtest_hg_civ < 1 } }
		set_temp_variable = { pww_dtest_hg_civ = 1 }
	}
	set_temp_variable = { pww_dba_side = 1 }
	if = {
		limit = { check_variable = { pww_dtest_hg_dir = 0 } }
		set_temp_variable = { pww_dba_side = 0 }
	}
	set_temp_variable = { pww_dba_kind = 1 }
	set_temp_variable = { pww_dba_x = 0 }
	pww_diplo_bundle_add = yes
	set_variable = { pww_dob_a^pww_dba_new = pww_dtest_hg_civ }
	pww_diplo_offer_changed = yes
}

# GER scope, after a draft. Counts how many of the 24 possible seeded
# reserves (2..25) accept the current draft. out: pww_dtest_hg_n.
# Restores the talk's own reserve afterwards.
pww_dtest_hg_sweep = {
	set_temp_variable = { pww_dtest_hg_t = pww_do_talk }
	set_temp_variable = { pww_dtest_hg_keep = global.pww_dt_reserve^pww_dtest_hg_t }
	set_temp_variable = { pww_dtest_hg_n = 0 }
	for_loop_effect = {
		start = 2
		end = 26
		value = pww_dtest_hg_r
		set_variable = { global.pww_dt_reserve^pww_dtest_hg_t = pww_dtest_hg_r }
		pww_diplo_eval_bundle = yes
		if = {
			limit = { check_variable = { pww_do_ok = 1 } }
			add_to_temp_variable = { pww_dtest_hg_n = 1 }
		}
	}
	set_variable = { global.pww_dt_reserve^pww_dtest_hg_t = pww_dtest_hg_keep }
	pww_diplo_offer_changed = yes
}

# GER scope. Like pww_dtest_hg_sweep, but counts the reserves at which the

		pww_diplo_eval_bundle = yes
		set_temp_variable = { pww_dtest_hg_rt = pww_do_e_total }

# GER scope, after an evaluation. Case I: checks that what the screen shows is
# the acceptance model's market baseline. out:
#   pww_dtest_hg_rowok  1 when every row's value is its raw market value at the
#                       form's rate (political power 0.9, equipment 0.85) and
#                       the displayed row figure is that value rounded
#   pww_dtest_hg_s0/s1  row values summed per side (must equal m0 / m1)
#   pww_dtest_hg_sumok  1 when the sums and the displayed totals match m0 / m1
#   pww_dtest_hg_tm     terms from the shown row values and the target's appetite:
#                       (sum of m x p on our side - on theirs) / base x 100
#                       (equals pww_do_e_terms while the seller's desperation is
#                       0; with every appetite at 1 it is (m0 - m1) / base)
pww_dtest_hg_check_market = {
	set_temp_variable = { pww_dtest_hg_rowok = 1 }
	set_temp_variable = { pww_dtest_hg_s0 = 0 }
	set_temp_variable = { pww_dtest_hg_s1 = 0 }
	set_temp_variable = { pww_dtest_hg_q0 = 0 }
	set_temp_variable = { pww_dtest_hg_q1 = 0 }
	for_loop_effect = {
		start = 0
		end = 8
		value = pww_dtest_hg_i
		if = {
			limit = { check_variable = { pww_dob_k^pww_dtest_hg_i > 0 } }
			set_temp_variable = { pww_dtest_hg_fm = 1 }
			if = {
				limit = { check_variable = { pww_dob_k^pww_dtest_hg_i = 2 } }
				set_temp_variable = { pww_dtest_hg_fm = 0.9 }
			}
			if = {
				limit = { check_variable = { pww_dob_k^pww_dtest_hg_i = 4 } }
				set_temp_variable = { pww_dtest_hg_fm = 0.85 }
			}
			set_temp_variable = { pww_dtest_hg_w = pww_dob_v^pww_dtest_hg_i }
			multiply_temp_variable = { pww_dtest_hg_w = pww_dtest_hg_fm }
			subtract_from_temp_variable = { pww_dtest_hg_w = pww_dob_m^pww_dtest_hg_i }
			if = {
				limit = {
					OR = {
						check_variable = { pww_dtest_hg_w > 0.002 }
						check_variable = { pww_dtest_hg_w < -0.002 }
					}
				}
				set_temp_variable = { pww_dtest_hg_rowok = 0 }
			}
			set_temp_variable = { pww_dtest_hg_q = pww_dob_m^pww_dtest_hg_i }
			multiply_temp_variable = { pww_dtest_hg_q = pww_dob_p^pww_dtest_hg_i }
			if = {
				limit = { check_variable = { pww_dtest_hg_i < 4 } }
				add_to_temp_variable = { pww_dtest_hg_s0 = pww_dob_m^pww_dtest_hg_i }
				add_to_temp_variable = { pww_dtest_hg_q0 = pww_dtest_hg_q }
			}
			else = {
				add_to_temp_variable = { pww_dtest_hg_s1 = pww_dob_m^pww_dtest_hg_i }
				add_to_temp_variable = { pww_dtest_hg_q1 = pww_dtest_hg_q }
			}
		}
	}
	# Displayed row figures (generated per-slot mirror) within rounding.
	set_temp_variable = { pww_dtest_hg_d = pww_dbv_0 }
	subtract_from_temp_variable = { pww_dtest_hg_d = pww_dob_m^0 }
	if = { limit = { OR = { check_variable = { pww_dtest_hg_d > 0.051 } check_variable = { pww_dtest_hg_d < -0.051 } } } set_temp_variable = { pww_dtest_hg_rowok = 0 } }
	set_temp_variable = { pww_dtest_hg_d = pww_dbv_4 }
	subtract_from_temp_variable = { pww_dtest_hg_d = pww_dob_m^4 }
	if = { limit = { OR = { check_variable = { pww_dtest_hg_d > 0.051 } check_variable = { pww_dtest_hg_d < -0.051 } } } set_temp_variable = { pww_dtest_hg_rowok = 0 } }
	set_temp_variable = { pww_dtest_hg_sumok = 1 }
	set_temp_variable = { pww_dtest_hg_d = pww_dtest_hg_s0 }
	subtract_from_temp_variable = { pww_dtest_hg_d = pww_do_mk_m0 }
	if = { limit = { OR = { check_variable = { pww_dtest_hg_d > 0.002 } check_variable = { pww_dtest_hg_d < -0.002 } } } set_temp_variable = { pww_dtest_hg_sumok = 0 } }
	set_temp_variable = { pww_dtest_hg_d = pww_dtest_hg_s1 }
	subtract_from_temp_variable = { pww_dtest_hg_d = pww_do_mk_m1 }
	if = { limit = { OR = { check_variable = { pww_dtest_hg_d > 0.002 } check_variable = { pww_dtest_hg_d < -0.002 } } } set_temp_variable = { pww_dtest_hg_sumok = 0 } }
	set_temp_variable = { pww_dtest_hg_d = pww_dbm_t0 }
	subtract_from_temp_variable = { pww_dtest_hg_d = pww_do_mk_m0 }
	if = { limit = { OR = { check_variable = { pww_dtest_hg_d > 0.051 } check_variable = { pww_dtest_hg_d < -0.051 } } } set_temp_variable = { pww_dtest_hg_sumok = 0 } }
	set_temp_variable = { pww_dtest_hg_d = pww_dbm_t1 }
	subtract_from_temp_variable = { pww_dtest_hg_d = pww_do_mk_m1 }
	if = { limit = { OR = { check_variable = { pww_dtest_hg_d > 0.051 } check_variable = { pww_dtest_hg_d < -0.051 } } } set_temp_variable = { pww_dtest_hg_sumok = 0 } }
	# Terms from the rows and the appetite
	set_temp_variable = { pww_dtest_hg_b = pww_do_mk_base }
	if = {
		limit = { check_variable = { pww_dtest_hg_b < 0.25 } }
		set_temp_variable = { pww_dtest_hg_b = 0.25 }
	}
	set_temp_variable = { pww_dtest_hg_tm = pww_dtest_hg_q0 }
	subtract_from_temp_variable = { pww_dtest_hg_tm = pww_dtest_hg_q1 }
	divide_temp_variable = { pww_dtest_hg_tm = pww_dtest_hg_b }
	multiply_temp_variable = { pww_dtest_hg_tm = 100 }
	clamp_temp_variable = { var = pww_dtest_hg_tm min = -100 max = 100 }
	round_temp_variable = pww_dtest_hg_tm
	set_temp_variable = { pww_dtest_hg_td = pww_dtest_hg_tm }

# Any scope. Sets the target's market position for every resource through
# the diplomacy test hooks. in: pww_dtest_hg_tgt, pww_dtest_hg_uns (unsold
# units), pww_dtest_hg_exp (units it already exports); pww_dtest_hg_uns -1
# clears both hooks (the live game figures again).
pww_dtest_hg_market = {
	var:pww_dtest_hg_tgt = {
		if = {
			limit = { check_variable = { pww_dtest_hg_uns < 0 } }
			clear_variable = pww_diplo_test_unsold
			clear_variable = pww_diplo_test_exported
		}
		else = {
			set_variable = { pww_diplo_test_unsold = pww_dtest_hg_uns }
			set_variable = { pww_diplo_test_exported = pww_dtest_hg_exp }
		}
	}
}
