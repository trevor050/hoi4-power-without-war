# GER scope. Reopens the talks with POR and checks nothing was reseeded.
pww_dtest_seed_check = {
	set_temp_variable = { pww_dk_target = POR }
	set_temp_variable = { pww_dk_act = 0 }
	pww_diplo_offer_begin = yes
	set_temp_variable = { t = pww_do_talk }
	log = "PWW_RESULT diplo_seed_reload.values INFO [GetDateText] talk=[?t|0] seed=[?global.pww_dt_seed^t|0] redline=[?global.pww_dt_redline^t|0] want=[?global.pww_dt_want^t|0] patience=[?global.pww_dt_patience^t|0] reserve=[?global.pww_dt_reserve^t|0] talks=[?global.pww_next_talk_id|0] salt=[?global.pww_diplo_salt|0]"
	if = {
		limit = {
			check_variable = { t = global.pww_dtest_talk }
			check_variable = { global.pww_next_talk_id = global.pww_dtest_n }
			check_variable = { global.pww_dt_redline^t = global.pww_dtest_rl }
			check_variable = { global.pww_dt_want^t = global.pww_dtest_want }
			check_variable = { global.pww_dt_patience^t = global.pww_dtest_pat }
			check_variable = { global.pww_dt_reserve^t = global.pww_dtest_res }
			check_variable = { global.pww_dt_seed^t = global.pww_dtest_seed }
		}
		log = "PWW_RESULT diplo_seed_reload.stable PASS [GetDateText] same talk, no reseed"
	}
	else = {
		log = "PWW_RESULT diplo_seed_reload.stable FAIL [GetDateText] talk=[?t|0] vs [?global.pww_dtest_talk|0] talks=[?global.pww_next_talk_id|0] vs [?global.pww_dtest_n|0]"
	}
	# The seed is a pure function of the saved salt, the pair's core buckets
	# and the talk id, so it can be recomputed and must match.
	set_temp_variable = { s = global.pww_diplo_salt }
	set_temp_variable = { b = pww_bucket }
	multiply_temp_variable = { b = 17 }
	add_to_temp_variable = { s = b }
	set_temp_variable = { b = POR.pww_bucket }
	multiply_temp_variable = { b = 31 }
	add_to_temp_variable = { s = b }
	set_temp_variable = { b = t }
	multiply_temp_variable = { b = 7 }
	add_to_temp_variable = { s = b }
	modulo_temp_variable = { var = s value = 97 }
	if = {
		limit = { check_variable = { s = global.pww_dt_seed^t } }
		log = "PWW_RESULT diplo_seed_reload.recompute PASS seed=[?s|0]"
	}
	else = {
		log = "PWW_RESULT diplo_seed_reload.recompute FAIL recomputed=[?s|0] stored=[?global.pww_dt_seed^t|0]"
	}
	log = "PWW_DONE diplo_seed_reload"
}
