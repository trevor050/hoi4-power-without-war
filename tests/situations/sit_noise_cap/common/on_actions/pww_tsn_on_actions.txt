# Situations test (b): noise cap. Start tag POR (human). Six generic
# situations with POR as holder: four go live, two queue. An AI-only
# situation is never capped. Resolving and cancelling promote the queue.
# Run: python tools/run_hoi4.py --test tests/situations/sit_noise_cap
on_actions = {
	on_daily = {
		effect = {
			if = {
				limit = { tag = GER }
				add_to_variable = { global.pww_tsn_day = 1 }

				if = {
					limit = { check_variable = { global.pww_tsn_day = 3 } }
					set_global_flag = pww_sit_debug
					POR = { set_temp_variable = { pww_in_holder = THIS } }
					set_temp_variable = { pww_in_template = 0 }
					set_temp_variable = { pww_in_resource = 3 }
					set_temp_variable = { pww_in_days = 300 }
					SPR = { set_temp_variable = { pww_in_claimant = THIS } }
					set_temp_variable = { pww_in_no_patron = 1 }
					pww_sit_open = yes
					ENG = { set_temp_variable = { pww_in_claimant = THIS } }
					set_temp_variable = { pww_in_no_patron = 1 }
					pww_sit_open = yes
					FRA = { set_temp_variable = { pww_in_claimant = THIS } }
					set_temp_variable = { pww_in_no_patron = 1 }
					pww_sit_open = yes
					ITA = { set_temp_variable = { pww_in_claimant = THIS } }
					set_temp_variable = { pww_in_no_patron = 1 }
					pww_sit_open = yes
					GER = { set_temp_variable = { pww_in_claimant = THIS } }
					set_temp_variable = { pww_in_no_patron = 1 }
					pww_sit_open = yes
					USA = { set_temp_variable = { pww_in_claimant = THIS } }
					set_temp_variable = { pww_in_no_patron = 1 }
					pww_sit_open = yes
					if = {
						limit = {
							check_variable = { global.pww_next_sit_id = 6 }
							check_variable = { global.pww_sit_status^0 = 1 }
							check_variable = { global.pww_sit_status^1 = 1 }
							check_variable = { global.pww_sit_status^2 = 1 }
							check_variable = { global.pww_sit_status^3 = 1 }
							check_variable = { global.pww_sit_status^4 = 0 }
							check_variable = { global.pww_sit_status^5 = 0 }
							POR = { check_variable = { pww_my_situations^num = 4 } }
							check_variable = { global.pww_sit_live^num = 4 }
							check_variable = { global.pww_sit_dev^4 = 11 }
						}
						log = "PWW_RESULT sit_noise_cap.cap PASS live=[?global.pww_sit_live^num|0] por_rows=[?POR.pww_my_situations^num|0]"
					}
					else = {
						log = "PWW_RESULT sit_noise_cap.cap FAIL next=[?global.pww_next_sit_id|0] s0=[?global.pww_sit_status^0|0] s3=[?global.pww_sit_status^3|0] s4=[?global.pww_sit_status^4|0] s5=[?global.pww_sit_status^5|0] por_rows=[?POR.pww_my_situations^num|0]"
					}

					# AI-only situation: not capped
					GER = { set_temp_variable = { pww_in_claimant = THIS } }
					SWE = { set_temp_variable = { pww_in_holder = THIS } }
					ENG = { set_temp_variable = { pww_in_patron = THIS } }
					set_temp_variable = { pww_in_template = 0 }
					set_temp_variable = { pww_in_resource = 4 }
					set_temp_variable = { pww_in_days = 300 }
					set_temp_variable = { pww_in_no_patron = 0 }
					pww_sit_open = yes
					if = {
						limit = {
							check_variable = { pww_out_result = 1 }
							check_variable = { pww_out_queued = 0 }
							check_variable = { global.pww_sit_status^6 = 1 }
							check_variable = { global.pww_sit_human^6 = 0 }
						}
						log = "PWW_RESULT sit_noise_cap.ai_uncapped PASS id=[?pww_out_sit_id|0]"
					}
					else = {
						log = "PWW_RESULT sit_noise_cap.ai_uncapped FAIL result=[?pww_out_result|0] fail=[?pww_out_fail|0] queued=[?pww_out_queued|0]"
					}

					# Resolve one human-seated situation: the oldest queued goes live
					set_temp_variable = { pww_in_sit_id = 0 }
					set_temp_variable = { pww_in_outcome = 4 }
					pww_sit_resolve = yes
					if = {
						limit = {
							check_variable = { global.pww_sit_status^0 = 2 }
							check_variable = { global.pww_sit_status^4 = 1 }
							check_variable = { global.pww_sit_status^5 = 0 }
							POR = { check_variable = { pww_my_situations^num = 4 } }
							POR = { NOT = { is_in_array = { array = pww_my_situations value = 0 } } }
							POR = { is_in_array = { array = pww_my_situations value = 4 } }
						}
						log = "PWW_RESULT sit_noise_cap.promote_on_resolve PASS"
					}
					else = {
						log = "PWW_RESULT sit_noise_cap.promote_on_resolve FAIL s0=[?global.pww_sit_status^0|0] s4=[?global.pww_sit_status^4|0] s5=[?global.pww_sit_status^5|0] por_rows=[?POR.pww_my_situations^num|0]"
					}

					# Resolve applies once
					set_temp_variable = { pww_in_sit_id = 0 }
					set_temp_variable = { pww_in_outcome = 2 }
					pww_sit_resolve = yes
					if = {
						limit = {
							check_variable = { pww_out_result = 0 }
							check_variable = { global.pww_sit_outcome^0 = 4 }
						}
						log = "PWW_RESULT sit_noise_cap.resolve_once PASS"
					}
					else = {
						log = "PWW_RESULT sit_noise_cap.resolve_once FAIL outcome=[?global.pww_sit_outcome^0|0]"
					}

					# Cancel one: the last queued goes live
					set_temp_variable = { pww_in_sit_id = 1 }
					pww_sit_cancel = yes
					if = {
						limit = {
							check_variable = { global.pww_sit_status^1 = 3 }
							check_variable = { global.pww_sit_status^5 = 1 }
							POR = { check_variable = { pww_my_situations^num = 4 } }
						}
						log = "PWW_RESULT sit_noise_cap.promote_on_cancel PASS"
					}
					else = {
						log = "PWW_RESULT sit_noise_cap.promote_on_cancel FAIL s1=[?global.pww_sit_status^1|0] s5=[?global.pww_sit_status^5|0]"
					}
				}

				# Daily clock keeps days_left current
				if = {
					limit = { check_variable = { global.pww_tsn_day = 8 } }
					set_temp_variable = { pww_tsn_exp = global.pww_sit_decision_day^2 }
					subtract_from_temp_variable = { pww_tsn_exp = global.num_days }
					set_temp_variable = { pww_tsn_lo = pww_tsn_exp }
					subtract_from_temp_variable = { pww_tsn_lo = 1 }
					set_temp_variable = { pww_tsn_hi = pww_tsn_exp }
					add_to_temp_variable = { pww_tsn_hi = 1 }
					if = {
						limit = {
							check_variable = { global.pww_sit_days_left^2 > pww_tsn_lo }
							check_variable = { global.pww_sit_days_left^2 < pww_tsn_hi }
							check_variable = { global.pww_sit_days_left^2 < 300 }
						}
						log = "PWW_RESULT sit_noise_cap.clock PASS left=[?global.pww_sit_days_left^2|0]"
					}
					else = {
						log = "PWW_RESULT sit_noise_cap.clock FAIL left=[?global.pww_sit_days_left^2|0] expected=[?pww_tsn_exp|0]"
					}
					pww_sit_debug_dump = yes
					log = "PWW_DONE sit_noise_cap"
				}
			}
		}
	}
}
